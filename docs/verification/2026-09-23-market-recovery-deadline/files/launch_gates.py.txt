"""Start two bounded gate controllers on separate clean, fixed checkouts."""
import json
from pathlib import Path
import socket
import subprocess

ROOT = Path(__file__).resolve().parent
PY = '/Users/a77/finance-workspace-private/.venv-workbench/bin/python'
candidate = json.loads((ROOT / 'candidate.json').read_text())
rev = candidate['revision']
assert subprocess.check_output(['git', '-C', str(ROOT / 'candidate'), 'ls-remote', 'gitea', 'refs/heads/main'], text=True).split()[0] == candidate['base']
for port in [19871, 19874]:
    with socket.socket() as sock:
        sock.bind(('127.0.0.1', port))
commands = {
    'full-gate': ('candidate', 7200, ['bash', 'scripts/run_main_gate.sh', '--pytest-args',
        f'-q -p no:cacheprovider --basetemp={ROOT / "full-pytest-tmp"} --junitxml={ROOT / "full-gate/pytest.xml"} --durations=20']),
    'frontend-gate': ('frontend-candidate', 1800, [PY, '-B', 'scripts/run_frontend_gate.py',
        '--tree', str(ROOT / 'frontend-candidate'), '--expect-revision', rev,
        '--output', str(ROOT / 'frontend-receipt'), '--workbench-port', '19871', '--re06-port', '19874']),
}
record = {}
for name, (tree, seconds, command) in commands.items():
    argv = [PY, '-B', str(ROOT / 'run_step.py'), '--tree', str(ROOT / tree), '--name', name,
            '--seconds', str(seconds), '--clean', '--', *command]
    with (ROOT / f'{name}-controller.log').open('xb') as log:
        proc = subprocess.Popen(argv, stdout=log, stderr=subprocess.STDOUT,
                                stdin=subprocess.DEVNULL, start_new_session=True)
    record[name] = {'pid': proc.pid, 'command': argv}
with (ROOT / 'launched.json').open('x') as stream:
    json.dump(record, stream, indent=2)
print(json.dumps(record))
