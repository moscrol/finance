"""Replay immutable prior-review probes; this is not a new reviewer verdict."""
import hashlib
import importlib.util
import json
from pathlib import Path
import shutil

ROOT = Path(__file__).resolve().parent
OLD = ROOT.parent / 'glm-50330cf'
spec = importlib.util.spec_from_file_location('prior_probe_runner', OLD / 'run.py')
runner = importlib.util.module_from_spec(spec)
spec.loader.exec_module(runner)
identity = json.loads((ROOT / 'candidate.json').read_text())
runner.ROOT = ROOT / 'recovery-replay'
runner.TREE = ROOT / 'candidate'
runner.REV = identity['revision']
runner.BASE = identity['base']
runner.ROOT.mkdir(exist_ok=False)
provenance = {'new_model_requests': 0, 'new_reviewer_verdict': None,
              'revision': runner.REV, 'original_revision': '50330cf4fa435292bbd555ccc817e62ca5cc45b4',
              'runner_path': str(OLD / 'run.py'),
              'runner_sha256': hashlib.sha256((OLD / 'run.py').read_bytes()).hexdigest(), 'probes': {}}
for claim in ['F1', 'F2', 'F3']:
    origin = OLD / 'zhipu-v1' / f'{claim}-explore/delivery'
    manifest = json.loads((origin / 'manifest.json').read_text())
    assert all(hashlib.sha256((origin / name).read_bytes()).hexdigest() == digest
               for name, digest in manifest.items())
    shutil.copytree(origin, runner.ROOT / f'{claim}-explore/delivery')
    provenance['probes'][claim] = {'path': str(origin), 'sha256': manifest['probe.py']}
    runner.execute(claim)
with (runner.ROOT / 'provenance.json').open('x') as stream:
    json.dump(provenance, stream, indent=2)
