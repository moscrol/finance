"""Remove only this finished frontend run's ignored, reproducible caches."""
from datetime import datetime, timezone
import json
import os
from pathlib import Path
import shutil
import subprocess

ROOT = Path(__file__).resolve().parent
TREE = ROOT / 'frontend-candidate'
identity = json.loads((ROOT / 'candidate.json').read_text())
for name in ['frontend-gate', 'frontend-two-diagnostic']:
    execution = json.loads((ROOT / name / 'execution.json').read_text())
    assert execution['complete'] and execution['after']['revision'] == identity['revision']
    try:
        os.kill(execution['pid'], 0)
    except ProcessLookupError:
        pass
    else:
        raise RuntimeError(f'{name} PID is live or reused; leaving caches untouched')
assert subprocess.check_output(['git', '-C', str(TREE), 'rev-parse', 'HEAD'], text=True).strip() == identity['revision']
assert not subprocess.check_output(['git', '-C', str(TREE), 'status', '--porcelain'], text=True).strip()
record = {'at': datetime.now(timezone.utc).isoformat(), 'revision': identity['revision'],
          'free_before': shutil.disk_usage(ROOT).free, 'removed': [],
          'free_space_delta_is_not_attributable': True, 'logs_and_reports_removed': False}
for name in ['node_modules', '.next']:
    path = TREE / 'intelligence/webapp' / name
    assert path.parent.resolve().is_relative_to(TREE.resolve())
    assert not path.is_symlink()
    subprocess.run(['git', '-C', str(TREE), 'check-ignore', '--quiet', str(path)], check=True)
    if path.exists():
        shutil.rmtree(path)
        record['removed'].append(str(path))
assert not subprocess.check_output(['git', '-C', str(TREE), 'status', '--porcelain'], text=True).strip()
record['free_after'] = shutil.disk_usage(ROOT).free
with (ROOT / 'frontend-cache-cleanup.json').open('x') as stream:
    json.dump(record, stream, indent=2)
print(json.dumps(record))
