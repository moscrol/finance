+def _data_checks(check, prod: Path, clone: Path, pq: Path,

+                f"SELECT COUNT(*) FROM (SELECT {sel} FROM {table} {where} "
+                f"EXCEPT ALL SELECT {sel} FROM prod.{table} {where})"
