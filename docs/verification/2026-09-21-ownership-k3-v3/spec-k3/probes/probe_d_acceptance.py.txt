def run_accept(clone, out, **over):
    args = [sys.executable, str(SCRIPT), "--production", str(art["baseline"]),
            "--clone", str(clone), "--parquet", str(art["pq"]),
            "--run-apply", fx.APPLY_RUN, "--run-verify", fx.VERIFY_RUN,
            "--expected-revision", over.get("rev", fx.E2E_REV),
            "--expected-production-sha256", art["base_sha"],
            "--output", str(out)]
    return subprocess.run(args, capture_output=True, text=True, timeout=110, env=ENV)


def fresh_clone(tag):
    c = tmp / f"clone-{tag}.duckdb"
    shutil.copy(str(art["clone"]), str(c))
    for mode, rid in (("apply", fx.APPLY_RUN), ("verify", fx.VERIFY_RUN)):
        shutil.copy(str(art["clone"]) + f".repair-backfill-execution.{rid}.json",
                    str(c) + f".repair-backfill-execution.{rid}.json")
    return c

def mutate_receipts(tag, fn):
    c = fresh_clone(tag)
    for mode, rid in (("apply", fx.APPLY_RUN), ("verify", fx.VERIFY_RUN)):
        p = Path(str(c) + f".repair-backfill-execution.{rid}.json")
        d = json.loads(p.read_text())
        fn(mode, d)
        p.write_text(json.dumps(d, ensure_ascii=False))
        # keep external child report in sync with embedded one
        cp = Path(d["child_report_path"])
        if cp.exists():
            cp.write_text(json.dumps(d["child_report"], ensure_ascii=False))
    return c
