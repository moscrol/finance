#!/usr/bin/env bash
# K3 independent probe A: shell gate receipt ownership/readback (R1/R2).
set -uo pipefail
OUT=/Users/a77/.finance-runtime/reviews/ownership-k3-v3-20260921/spec-k3/probes
CHK=/Users/a77/fwp-wt-ownership-spec-v3-0921
PY=/Users/a77/finance-workspace-private/.venv-workbench/bin/python
LOG=$OUT/probe_a_gate.log
: > "$LOG"

REPO=$OUT/repo-a
rm -rf "$REPO"; mkdir -p "$REPO"
cd "$REPO"
git init -q -b main
git config user.name k3; git config user.email k3@probe.test; git config commit.gpgsign false
cp "$CHK/conftest.py" conftest.py
cat > test-environment.json <<EOF
{"interpreter": "$PY", "required_modules": [], "code_path_prefixes": ["test_", "conftest.py"]}
EOF
cat > test_pass.py <<'EOF'
def test_ok():
    assert True
EOF
cat > .gitignore <<'EOF'
__pycache__/
.pytest_cache/
.ruff_cache/
EOF
git add -A; git commit -qm fixture
HEAD=$(git rev-parse HEAD)

run_gate() { # extra env then args
  env -i HOME="$OUT/home" PATH="$PATH" FWP_TEST_RECEIPT_DIR="$OUT/receipts-a" \
      FWP_TEST_RECEIPT=1 PYTHONDONTWRITEBYTECODE=1 "$@" 2>&1
}
note() { echo "=== $*" | tee -a "$LOG"; }

note "A1 legit clean run (passing test)"
run_gate bash "$CHK/scripts/run_main_gate.sh" --pytest-args "-q -p no:cacheprovider test_pass.py" >>"$LOG"
echo "A1 rc=$?" | tee -a "$LOG"
RCPT=$(ls "$OUT"/receipts-a/gate-*/pytest.json 2>/dev/null | head -1)
echo "A1 receipt=$RCPT" | tee -a "$LOG"
[ -n "$RCPT" ] && "$PY" -c "
import json,sys
d=json.load(open('$RCPT'))
print('A1 receipt identity:', d['revision']=='$HEAD', 'dirty=',d['dirty'], 'tree_ok=', d['tree']=='$REPO', 'counts=',d['counts'], 'exit=',d['exit_status'])" | tee -a "$LOG"

note "A2 failing run -> rc1, readback keeps failure"
cat > test_fail.py <<'EOF'
def test_bad():
    assert False
EOF
run_gate bash "$CHK/scripts/run_main_gate.sh" --pytest-args "-q -p no:cacheprovider test_pass.py test_fail.py" >>"$LOG"
echo "A2 exec rc=$? (expect 1)" | tee -a "$LOG"
RCPT2=$(ls -t "$OUT"/receipts-a/gate-*/pytest.json | head -1)
run_gate bash "$CHK/scripts/run_main_gate.sh" --receipt "$RCPT2" >>"$LOG"
echo "A2 readback rc=$? (expect 1)" | tee -a "$LOG"
git rm -q test_fail.py; git commit -qm remove-fail

note "A3 readback mutations (all must refuse rc!=0)"
"$PY" - "$RCPT" "$OUT" <<'PYEOF'
import json, subprocess, sys, copy
rcpt_path, out = sys.argv[1], sys.argv[2]
base = json.load(open(rcpt_path))
muts = {
  "wrong_revision": {"revision": "0"*40},
  "foreign_tree": {"tree": "/foreign/tree"},
  "tree_none": {"tree": None},
  "tree_int": {"tree": 123},
  "dirty_true": {"dirty": True},
  "dirty_total_1": {"worktree_dirty_total": 1},
  "dirty_total_missing": {"worktree_dirty_total": None},
  "counts_bool": {"counts": {"passed": True, "failed": 0, "error": 0, "skipped": 0}},
  "counts_negative": {"counts": {"passed": -1, "failed": 0, "error": 0, "skipped": 0}},
  "counts_zero_exec": {"counts": {"passed": 0, "failed": 0, "error": 0, "skipped": 3}},
  "counts_str": {"counts": {"passed": "1", "failed": 0, "error": 0, "skipped": 0}},
  "exit_interrupted": {"exit_status": 2},
  "exit_none": {"exit_status": None},
  "failedids_len_mismatch": {"failed_ids": ["x::y"]},
  "failedids_empty_str": {"failed_ids": [""]},
  "dep_bypassed": {"dependency_gate_bypassed": True},
  "dep_missing": {"dependency_gate_bypassed": None},
  "wrong_interpreter": {"interpreter": "/usr/bin/false"},
  "status_counts_disagree": {"exit_status": 1},
}
env = {"HOME": out+"/home", "PATH": "/usr/bin:/bin:/usr/local/bin:/opt/homebrew/bin",
       "FWP_TEST_RECEIPT_DIR": out+"/receipts-a"}
for name, m in muts.items():
    d = copy.deepcopy(base); d.update(m)
    p = f"{out}/mut-{name}.json"; json.dump(d, open(p, "w"))
    r = subprocess.run(["bash", "/Users/a77/fwp-wt-ownership-spec-v3-0921/scripts/run_main_gate.sh",
                        "--receipt", p], cwd="/Users/a77/fwp-wt-ownership-spec-v3-0921",
                       capture_output=True, text=True, env=env)
    verdict = "OK-REFUSED" if r.returncode != 0 else "!!!ACCEPTED"
    print(f"A3 {name}: rc={r.returncode} {verdict} :: {(r.stderr or r.stdout).strip().splitlines()[-1][:110]}")
PYEOF

note "A3b baseline semantics"
# same red set vs baseline -> 0 ; new red -> 3 ; baseline target mismatch -> !=0
"$PY" - "$OUT" <<'PYEOF'
import json, subprocess, sys
out = sys.argv[1]
env = {"HOME": out+"/home", "PATH": "/usr/bin:/bin:/usr/local/bin:/opt/homebrew/bin",
       "FWP_TEST_RECEIPT_DIR": out+"/receipts-a"}
repo = out+"/repo-a"
head = subprocess.run(["git","rev-parse","HEAD"],cwd=repo,capture_output=True,text=True).stdout.strip()
def rc(**kw):
    d = {"revision": head, "tree": repo, "interpreter": sys.executable,
         "dirty": False, "worktree_dirty_total": 0, "dependency_gate_bypassed": False,
         "target": "t.py", "counts": {"passed": 1, "failed": 1, "error": 0, "skipped": 0},
         "failed_ids": ["t.py::test_x"], "exit_status": 1}
    d.update(kw); return d
cur = rc()
json.dump(cur, open(out+"/b-cur.json","w"))
json.dump(cur, open(out+"/b-base-same.json","w"))
nb = rc(failed_ids=["t.py::test_x","t.py::test_y"], counts={"passed":0,"failed":2,"error":0,"skipped":0})
json.dump(nb, open(out+"/b-cur-newred.json","w"))
tg = rc(target="other.py"); json.dump(tg, open(out+"/b-base-target.json","w"))
dr = rc(dirty=True); json.dump(dr, open(out+"/b-base-dirty.json","w"))
G="/Users/a77/fwp-wt-ownership-spec-v3-0921/scripts/run_main_gate.sh"
for name, c, b in [("same_red", "b-cur.json", "b-base-same.json"),
                   ("new_red", "b-cur-newred.json", "b-base-same.json"),
                   ("target_mismatch", "b-cur.json", "b-base-target.json"),
                   ("dirty_baseline", "b-cur.json", "b-base-dirty.json")]:
    r = subprocess.run(["bash", G, "--receipt", f"{out}/{c}", "--baseline", f"{out}/{b}"],
                       cwd=repo, capture_output=True, text=True, env=env)
    print(f"A3b {name}: rc={r.returncode} :: {(r.stdout+r.stderr).strip().splitlines()[-1][:100]}")
PYEOF

note "A4 allow-dirty combinations"
echo "x" >> "$REPO/conftest.py"   # dirty the tree (code path)
run_gate bash "$CHK/scripts/run_main_gate.sh" --pytest-args "-q -p no:cacheprovider test_pass.py" >>"$LOG"
echo "A4 dirty-no-flag rc=$? (expect 2)" | tee -a "$LOG"
run_gate bash "$CHK/scripts/run_main_gate.sh" --allow-dirty --pytest-args "-q -p no:cacheprovider test_pass.py" >>"$LOG"
echo "A4 dirty-allow rc=$? (expect 0)" | tee -a "$LOG"
RCPT3=$(ls -t "$OUT"/receipts-a/gate-*/pytest.json | head -1)
"$PY" -c "import json;d=json.load(open('$RCPT3'));print('A4 dirty receipt dirty=',d['dirty'],'total=',d['worktree_dirty_total'])" | tee -a "$LOG"
run_gate bash "$CHK/scripts/run_main_gate.sh" --receipt "$RCPT3" >>"$LOG"
echo "A4 dirty receipt readback no-flag rc=$? (expect 4)" | tee -a "$LOG"
run_gate bash "$CHK/scripts/run_main_gate.sh" --receipt "$RCPT3" --allow-dirty >>"$LOG"
echo "A4 dirty receipt readback allow rc=$? (expect 0)" | tee -a "$LOG"
git checkout -- conftest.py

note "A5 receipt immutability: second writer cannot overwrite owned path"
OWN="$OUT/owned/pytest.json"; mkdir -p "$OUT/owned"
FWP_TEST_RECEIPT=1 FWP_TEST_RECEIPT_DIR="$OUT/owned" FWP_TEST_RECEIPT_PATH="$OWN" \
  "$PY" -m pytest -q -p no:cacheprovider test_pass.py >/dev/null 2>&1
S1=$(shasum -a 256 "$OWN" | cut -d' ' -f1)
FWP_TEST_RECEIPT=1 FWP_TEST_RECEIPT_DIR="$OUT/owned" FWP_TEST_RECEIPT_PATH="$OWN" \
  "$PY" -m pytest -q -p no:cacheprovider test_pass.py 2>&1 | tail -2 | tee -a "$LOG"
S2=$(shasum -a 256 "$OWN" | cut -d' ' -f1)
echo "A5 same-file-after-second-run: $([ "$S1" = "$S2" ] && echo UNCHANGED || echo OVERWRITTEN)" | tee -a "$LOG"

note "A6 receipt disabled -> gate refuses, no latest.json fallback"
mkdir -p "$OUT/receipts-a6"
echo '{"revision":"stale"}' > "$OUT/receipts-a6/latest.json"
run_gate FWP_TEST_RECEIPT=0 FWP_TEST_RECEIPT_DIR="$OUT/receipts-a6" \
  bash "$CHK/scripts/run_main_gate.sh" --pytest-args "-q -p no:cacheprovider test_pass.py" >>"$LOG"
echo "A6 rc=$? (expect 4)" | tee -a "$LOG"

note "A7 nested pytest cannot claim parent path"
cat > test_nested.py <<'EOF'
import os, subprocess, sys
def test_nested():
    r = subprocess.run([sys.executable, "-m", "pytest", "--collect-only", "-q"],
                       capture_output=True, text=True, timeout=60)
    assert r.returncode == 0, r.stdout + r.stderr
EOF
git add test_nested.py; git commit -qm nested
run_gate bash "$CHK/scripts/run_main_gate.sh" --pytest-args "-q -p no:cacheprovider test_nested.py" >>"$LOG"
echo "A7 rc=$? (expect 0)" | tee -a "$LOG"
N=$(ls "$OUT"/receipts-a/gate-*/pytest.json | wc -l | tr -d ' ')
echo "A7 receipts total=$N (each run exactly one)" | tee -a "$LOG"
git rm -q test_nested.py; git commit -qm cleanup
echo "PROBE-A-DONE" | tee -a "$LOG"
