#!/usr/bin/env python3
"""Probe C4 (behavioral): llm_refine wrapper must honor the shared deadline.

Case A: deadline=0.5s < the 1s minimum-viable gate -> pre-flight rejection,
        NO HTTP request to the fixture.
Case B: deadline=1.5s (passes the gate), timeout=10 slice, fixture answers at
        ~2.9s. If the wrapper forwards the shared deadline the call must be cut
        near 1.5s and return None; if forwarding were missing the slice (10s)
        would let the ~2.9s fixture succeed.
"""
import json
import sys
import threading
import time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

sys.path.insert(0, "/Users/a77/.finance-runtime/reviews/pr868-k3-qc-20260923-1607/candidate/finance-workspace-private")
from intelligence.services import llm_refine  # noqa: E402

HOST, PORT = "127.0.0.1", 26003
server_hit = []


class Handler(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def do_POST(self):
        server_hit.append(time.monotonic())
        time.sleep(2.9)
        body = json.dumps({
            "choices": [{"message": {"content": "SLOW-FIXTURE-ANSWER"},
                         "finish_reason": "stop"}],
            "usage": {"total_tokens": 7},
        }).encode()
        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, *a):
        pass


def run(provider, deadline):
    t0 = time.monotonic()
    with llm_refine.provider_override(provider):
        result, reason = llm_refine.synthesize_messages(
            [{"role": "user", "content": "hi"}], timeout=10,
            deadline=deadline, max_tokens=64)
    return result, reason, time.monotonic() - t0


def main():
    srv = ThreadingHTTPServer((HOST, PORT), Handler)
    threading.Thread(target=srv.serve_forever, daemon=True).start()
    time.sleep(0.2)
    provider = llm_refine.LLMProvider(
        name="probe", api_key="k", model="m", base_url=f"http://{HOST}:{PORT}/v1")

    # Case A: below viability gate -> no request at all
    hits_before = len(server_hit)
    result, reason, dt = run(provider, llm_refine.Deadline(time.monotonic() + 0.5))
    print(f"A: result={result!r} reason={reason!r} dt={dt:.2f} hits+={len(server_hit)-hits_before}")
    assert result is None and len(server_hit) == hits_before, (
        "0.5s shared deadline should reject pre-flight without an HTTP request")

    # Case B: gate passed, fixture slower than deadline -> cut mid-flight
    t0 = time.monotonic()
    result, reason, dt = run(provider, llm_refine.Deadline(t0 + 1.5))
    print(f"B: result={result!r} reason={reason!r} dt={dt:.2f} hits={len(server_hit)}")
    assert len(server_hit) == 2 or len(server_hit) >= 2, "fixture was not called in case B"
    assert result is None, f"wrapper accepted the late fixture answer: {result!r}"
    assert dt < 2.5, (
        f"shared deadline not enforced mid-flight: survived {dt:.2f}s "
        "with 1.5s deadline / 10s slice / 2.9s fixture")
    assert dt >= 1.4, f"cut too early ({dt:.2f}s) -- gate or fixture timing changed"
    srv.shutdown()
    print("PROBE_C4_WRAPPER_SHARED_DEADLINE: PASS")


if __name__ == "__main__":
    main()
