#!/usr/bin/env python3
"""Probe C4 (behavioral): llm_refine wrapper must honor the shared deadline.

Server on 127.0.0.1:26003 waits ~2.9s, then returns a valid chat.completions
body. We call llm_refine.synthesize_messages with timeout=10 (a per-call slice
far larger than the fixture) but deadline=Deadline(now+0.5). If the wrapper
truly forwards the shared deadline, the call must fail near 0.5s with a
deadline-style reason. If forwarding is missing, the call would succeed at
~2.9s (or fail at 10s) -- the behavioral discriminator the claim demands.
"""
import json
import sys
import threading
import time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

sys.path.insert(0, "/Users/a77/.finance-runtime/reviews/pr868-k3-qc-20260923-1607/candidate/finance-workspace-private")
from intelligence.services import llm_refine  # noqa: E402

HOST, PORT = "127.0.0.1", 26003
server_hit = []


class Handler(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def do_POST(self):
        server_hit.append(time.monotonic())
        time.sleep(2.9)
        body = json.dumps({
            "choices": [{"message": {"content": "SLOW-FIXTURE-ANSWER"},
                         "finish_reason": "stop"}],
            "usage": {"total_tokens": 7},
        }).encode()
        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, *a):
        pass


def main():
    srv = ThreadingHTTPServer((HOST, PORT), Handler)
    threading.Thread(target=srv.serve_forever, daemon=True).start()

    provider = llm_refine.LLMProvider(
        name="probe", api_key="k", model="m",
        base_url=f"http://{HOST}:{PORT}/v1")
    messages = [{"role": "user", "content": "hi"}]
    dl = llm_refine.Deadline(time.monotonic() + 0.5)
    t0 = time.monotonic()
    with llm_refine.provider_override(provider):
        result, reason = llm_refine.synthesize_messages(
            messages, timeout=10, deadline=dl, max_tokens=64)
    elapsed = time.monotonic() - t0
    print(f"result={result!r} reason={reason!r} elapsed={elapsed:.2f}s "
          f"server_hit={len(server_hit)}")
    assert server_hit, "no HTTP request reached the fixture"
    assert result is None, f"wrapper accepted the late fixture answer: {result!r}"
    assert elapsed < 2.0, (
        f"shared deadline not enforced: call survived {elapsed:.2f}s "
        "with a 0.5s deadline and a 10s slice"
    )
    print("PROBE_C4_WRAPPER_SHARED_DEADLINE: PASS")


if __name__ == "__main__":
    main()
