import shutil

for name in ['review.mjs', 'run_control.py', 'run_stage.py', 'sandbox_preflight.mjs']:
    p = ROOT / 'spec' / name
    p.write_text(p.read_text().replace('fomo', 'glm-direct-review').replace('private K3', 'private GLM').replace('isolated K3', 'isolated GLM'))
    shutil.copyfile(p, ROOT / 'quality' / name)

    config.update(provider='glm-direct-review', upstream_port=443,
                  upstream_url='https://open.bigmodel.cn/api/coding/paas/v4')
