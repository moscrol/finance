class HTTPStatusError(RuntimeError):
    def __init__(self, status: int):
        super().__init__(f"HTTP status {status}")
        self.status = status


def dump(path: Path, value: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def now() -> str:
    return datetime.now(timezone.utc).isoformat()


def request(base: str, path: str, payload=None, *, method: str | None = None, timeout: float = 30):
    data = json.dumps(payload, ensure_ascii=False).encode() if payload is not None else None
    req = urllib.request.Request(
        base + path,
        data=data,
        headers={"Content-Type": "application/json"},
        method=method or ("POST" if data is not None else "GET"),
    )
    try:
        with urllib.request.urlopen(req, timeout=timeout) as response:
            raw = response.read()
    except urllib.error.HTTPError as exc:
        raise HTTPStatusError(exc.code) from None
    if not raw:
        return {}
    try:
        return json.loads(raw)
    except json.JSONDecodeError:
        return {"raw_response_bytes": len(raw)}


def identity(repo: Path) -> dict[str, str]:
    return {
        "revision": subprocess.check_output(
            ["git", "rev-parse", "HEAD"], cwd=repo, text=True
        ).strip(),
        "status": subprocess.check_output(
            ["git", "status", "--porcelain"], cwd=repo, text=True
        ).strip(),
    }


def production_identity(payload: dict) -> dict:
    runtime = payload.get("runtime") or {}
    return {
        key: runtime.get(key)
        for key in (
            "source_revision",
            "source_dirty",
            "code_root",
            "loaded_code_root",
            "loaded_tree_fingerprint",
            "repo_tree_fingerprint",
            "dependency_fingerprint",
            "python_prefix",
            "code_matches_repo",
            "users_dir",
            "agent_runtime",
        )
    }


def sidecar_identity(health: dict, repo: Path) -> dict:
    runtime = health.get("runtime") or {}
    expected_root = str(repo.resolve())
    actual = {
        "source_revision": runtime.get("source_revision"),
        "source_dirty": runtime.get("source_dirty"),
        "code_root": runtime.get("code_root"),
        "loaded_code_root": runtime.get("loaded_code_root"),
        "loaded_tree_fingerprint": runtime.get("loaded_tree_fingerprint"),
        "repo_tree_fingerprint": runtime.get("repo_tree_fingerprint"),
        "dependency_fingerprint": runtime.get("dependency_fingerprint"),
        "python_prefix": runtime.get("python_prefix"),
        "python_version": runtime.get("python_version"),
    }
    if actual["source_revision"] != CANDIDATE:
        raise RuntimeError("sidecar revision mismatch")
    if actual["source_dirty"] is not False:
        raise RuntimeError("sidecar source tree is dirty")
    if actual["code_root"] != expected_root or actual["loaded_code_root"] != str(repo / "intelligence"):
        raise RuntimeError("sidecar loaded the wrong code root")
    if actual["loaded_tree_fingerprint"] != actual["repo_tree_fingerprint"]:
        raise RuntimeError("sidecar code fingerprint does not match repository fingerprint")
    if actual["python_prefix"] != str(PYTHON.parent.parent):
        raise RuntimeError("sidecar Python prefix mismatch")
    return actual


def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def verify_manifest(preparation: Path) -> dict:
    manifest = json.loads((preparation / "frozen-data-manifest.json").read_text())
    mismatches = []
    checked = 0
    for relative, expected in manifest.items():
        path = preparation / "data" / relative
        if not path.is_file():
            mismatches.append({"path": relative, "reason": "missing"})
            continue
        checked += 1
        actual_bytes = path.stat().st_size
        actual_sha = sha256(path)
        if actual_bytes != expected["bytes"] or actual_sha != expected["sha256"]:
            mismatches.append({
                "path": relative,
                "expected_bytes": expected["bytes"],
                "actual_bytes": actual_bytes,
                "expected_sha256": expected["sha256"],
                "actual_sha256": actual_sha,
            })
    return {"checked": checked, "manifest_entries": len(manifest), "mismatches": mismatches}


def kill_process(process: subprocess.Popen | None) -> None:
    if process is None or process.poll() is not None:
        return
    try:
        os.killpg(process.pid, signal.SIGTERM)
        process.wait(timeout=20)
    except subprocess.TimeoutExpired:
        os.killpg(process.pid, signal.SIGKILL)
        process.wait(timeout=10)


def wait_for_health(base: str, process: subprocess.Popen, out: Path, repo: Path, user: str) -> dict:
    deadline = time.monotonic() + 420
    last_live = {}
    while time.monotonic() < deadline:
        if process.poll() is not None:
            raise RuntimeError("sidecar exited during startup")
        try:
            last_live = request(base, "/api/health", timeout=10)
            sidecar_identity(last_live, repo)
            try:
                ready = request(base, f"/api/health/ready?user={urllib.parse.quote(user)}", timeout=30)
            except HTTPStatusError as exc:
                if exc.status != 503:
                    raise
                time.sleep(2)
                continue
            if ready.get("status") == "ready":
                dump(out / "health.json", last_live)
                dump(out / "readiness.json", ready)
                return last_live
        except (OSError, urllib.error.URLError):
            pass
        time.sleep(2)
    raise RuntimeError("sidecar readiness timed out")


def run_question(base: str, out: Path, user: str, question: dict, process: subprocess.Popen) -> dict:
    question_dir = out / question["id"]
    question_dir.mkdir()
    dump(question_dir / "question.json", question)
    if process.poll() is not None:
        raise RuntimeError("sidecar exited before submit")
    conversation = request(
        base,
        "/api/conversations",
        {"user": user, "title": f"L6 natural acceptance {question['id']}"},
    )
    dump(question_dir / "conversation.json", conversation)
    conversation_id = conversation["conversation_id"]
    submitted_at = now()
    with (question_dir / "submission-reservation.json").open("x") as stream:
        json.dump({"reserved": 1, "resubmissions": 0, "followups": 0, "at": submitted_at}, stream)
    try:
        submitted = request(
            base,
            f"/api/conversations/{conversation_id}/messages",
            {
                "user": user,
                "content": question["text"],
                "skill_mode": "hybrid",
                "perspective_mode": "neutral",
            },
        )
    except HTTPStatusError as exc:
        dump(question_dir / "submission-error.json", {"status": exc.status, "at": submitted_at})
        raise
    dump(question_dir / "submission.json", submitted)
    run_id = submitted["run_id"]
    message_id = submitted["assistant_message_id"]
    dump(question_dir / "submission-count.json", {"accepted_submissions": 1, "resubmissions": 0, "followups": 0})
    started = time.monotonic()
    deadline = started + 900
    with (question_dir / "polls.jsonl").open("w", encoding="utf-8") as polls:
        while time.monotonic() < deadline:
            state = request(f"http://127.0.0.1:{SHIM_PORT}", "/_status")
            if state.get("stopped_for"):
                result = {"id": question["id"], "run_id": run_id, "status": "blocked_provider", "reason": state["stopped_for"]}
                dump(question_dir / "result.json", result)
                return result
            run = request(base, f"/api/runs/{run_id}?user={urllib.parse.quote(user)}")
            polls.write(json.dumps(run, ensure_ascii=False) + "\n")
            polls.flush()
            dump(question_dir / "run.json", run)
            status = run.get("status")
            if status in {"failed", "cancelled"}:
                result = {
                    "id": question["id"],
                    "run_id": run_id,
                    "status": status,
                    "elapsed_seconds": round(time.monotonic() - started, 2),
                }
                dump(question_dir / "result.json", result)
                return result
            if status == "completed" and not run.get("delivery_pending"):
                messages = request(
                    base,
                    f"/api/conversations/{conversation_id}/messages?user={urllib.parse.quote(user)}",
                )
                dump(question_dir / "messages.json", messages)
                items = messages.get("messages", []) if isinstance(messages, dict) else messages
                answer = next((item for item in items if item.get("message_id") == message_id), None)
                if answer and answer.get("content"):
                    dump(question_dir / "message.json", answer)
                    (question_dir / "answer.md").write_text(answer["content"], encoding="utf-8")
                    source = out / "users" / user / "runs" / run_id
                    if source.is_dir():
                        subprocess.run(["cp", "-R", str(source), str(question_dir / "raw-run")], check=True)
                    result = {
                        "id": question["id"],
                        "run_id": run_id,
                        "status": status,
                        "elapsed_seconds": round(time.monotonic() - started, 2),
                    }
                    dump(question_dir / "result.json", result)
                    return result
            time.sleep(2)
    result = {
        "id": question["id"],
        "run_id": run_id,
        "status": "deadline_exceeded",
        "elapsed_seconds": round(time.monotonic() - started, 2),
    }
    dump(question_dir / "result.json", result)
    return result


def scan_for_secret(root: Path, secret: str) -> list[str]:
    matches = []
    needle = secret.encode()
    for path in root.rglob("*"):
        if not path.is_file():
            continue
        with path.open("rb") as stream:
            while block := stream.read(1024 * 1024):
                if needle in block:
                    matches.append(str(path.relative_to(root)))
                    break
    return matches
