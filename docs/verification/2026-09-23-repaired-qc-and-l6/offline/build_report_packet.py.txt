"""Build the same-axis report packet from raw allowed evidence, not reasoning."""
import hashlib
import json
from pathlib import Path
import sys

ROOT = Path('/Users/a77/.finance-runtime/reviews/pr868-glm-qc-repaired-20260923-2115')
axis = sys.argv[1]
assert axis in ('spec', 'quality')
base = ROOT / axis
for phase in ('explore', 'execute'):
    assert json.loads((base / phase / 'execution.json').read_text())['status'] == 'STAGE_COMPLETE'
    assert json.loads((base / f'{phase}.controller.json').read_text())['status'] == 'COMPLETE'
parts = ['# Same-axis immutable evidence packet\n\nNo model reasoning or other-axis conclusion is included. Host packaging is not a verdict.\n']
manifest = {}
def include(path):
    raw = path.read_bytes()
    key = str(path.relative_to(base))
    manifest[key] = {'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw)}
    parts.append('\n## ' + key + '\n\n' + raw.decode() + '\n')
include(base / 'inputs/claims.md')
include(base / 'config.json')
for phase in ('explore', 'execute'):
    include(base / phase / 'parsed.json')
    record = json.loads((base / phase / 'execution.json').read_text())
    summary = {key: record.get(key) for key in ('stage','status','revision','baseline','requests','elapsed_seconds','errors','report_structural','inputs_unchanged','candidate_before','candidate_after','model_identities','probe_sha256')}
    parts.append('\n## '+phase+' controller summary (derived whitelist)\n'+json.dumps(summary, ensure_ascii=False)+'\n')
for path in sorted((base / 'work/probes').glob('*.py')):
    include(path)
for folder in sorted((base / 'execute/commands').iterdir()):
    if folder.name.endswith('-deliver_stage'):
        continue
    for name in ('request.json', 'result.json', 'output.log'):
        if (folder / name).is_file():
            include(folder / name)
body = '\n'.join(parts)
assert len(body.encode()) <= 160000, len(body.encode())
with (base / 'report-packet.md').open('x') as stream:
    stream.write(body)
with (base / 'report-packet-manifest.json').open('x') as stream:
    json.dump({'axis': axis, 'inputs': manifest, 'reasoning_excluded': True, 'truncated_files': [],
               'packet_sha256': hashlib.sha256(body.encode()).hexdigest(), 'packet_bytes': len(body.encode())}, stream, indent=2)
print(json.dumps({'axis': axis, 'bytes': len(body.encode()), 'files': len(manifest)}))
