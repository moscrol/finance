import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { test } from 'node:test';
import bootstrap from './credential-bootstrap.mjs';

const root = path.dirname(fileURLToPath(import.meta.url));

test('normalize models.json for extension and audit bounded streamed requests without credential copies', () => {
  const out = fs.mkdtempSync(path.join(root, 'bootstrap-selftest-'));
  process.env.REVIEW_OUTPUT_DIR = out;
  const hooks = new Map();
  let registered;
  bootstrap({ registerProvider: (name, config) => { registered = { name, config }; },
    on: (name, handler) => hooks.set(name, handler) });
  assert.equal(registered.name, 'fomo');
  const model = registered.config.models[0];
  assert.equal(model.id, 'glm-5.3');
  assert.deepEqual(model.input, ['text']);
  assert.deepEqual(Object.keys(model.cost).sort(), ['cacheRead', 'cacheWrite', 'input', 'output']);
  assert.equal(model.compat.supportsDeveloperRole, false);
  assert.equal(registered.config.baseUrl, 'http://127.0.0.1:54340/v1');
  const ctx = { model: { provider: 'fomo', id: 'glm-5.3' } };
  for (let i = 0; i < 4; i++) {
    const payload = hooks.get('before_provider_request')({ payload: {
      model: 'glm-5.3', stream: true, temperature: 1, tools: [{ type: 'function' }],
    } }, ctx);
    assert.equal(Object.hasOwn(payload, 'temperature'), false);
    const admissions = fs.readFileSync(path.join(out, 'request-admissions.jsonl'), 'utf8').trim().split('\n');
    assert.equal(admissions.length, i + 1);
    assert.equal(JSON.parse(admissions.at(-1)).admission, i + 1);
    hooks.get('after_provider_response')({ status: 200 });
  }
  const originalExit = process.exit;
  process.exit = code => { throw new Error(`EXIT_${code}`); };
  try {
    assert.throws(() => hooks.get('before_provider_request')({ payload: {
      model: 'glm-5.3', stream: true,
    } }, ctx), /EXIT_75/);
  } finally {
    process.exit = originalExit;
  }
  for (const name of fs.readdirSync(out)) {
    assert.equal(fs.readFileSync(path.join(out, name), 'utf8').includes(registered.config.apiKey), false);
  }
  fs.rmSync(out, { recursive: true });
});
