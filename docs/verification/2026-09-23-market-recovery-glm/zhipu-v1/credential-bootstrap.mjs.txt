import fs from 'node:fs';
import path from 'node:path';
import childProcess from 'node:child_process';
import { fileURLToPath } from 'node:url';

const ROOT = path.dirname(fileURLToPath(import.meta.url));
const MAX_REQUESTS = 4;
const BUDGET_SECONDS = 600;

export default function (pi) {
  const out = fs.realpathSync(process.env.REVIEW_OUTPUT_DIR || '/invalid-output');
  if (!out.startsWith(ROOT + path.sep)) throw new Error('Invalid review output root');
  const started = Date.now();
  const file = path.join(out, 'credential.json');
  const record = { status: 'RESERVED', provider: 'zhipu', model: 'glm-5.3',
    started_at: new Date(started).toISOString(), credential_copied: false,
    credential_source: 'Keychain service finance-workbench-glm, account a77',
    runtime_credential_memory_only: true, prices_verified: false };
  fs.writeFileSync(file, JSON.stringify(record, null, 2) + '\n', { flag: 'wx', mode: 0o600 });
  try {
    const config = JSON.parse(fs.readFileSync(path.join(ROOT, 'provider-template.json'), 'utf8'));
    if (config.baseUrl !== 'https://open.bigmodel.cn/api/coding/paas/v4' ||
        config.api !== 'openai-completions' || config.models.length !== 1 ||
        config.models[0].id !== 'glm-5.3') throw new Error('Provider route mismatch');
    const token = childProcess.execFileSync('/usr/bin/security', [
      'find-generic-password', '-a', 'a77', '-s', 'finance-workbench-glm', '-w',
    ], { encoding: 'utf8', timeout: 20000, maxBuffer: 65536,
      stdio: ['ignore', 'pipe', 'pipe'] }).trim();
    if (token.length < 10) throw new Error('Missing credential');
    pi.registerProvider('zhipu', { ...config, apiKey: token });
    record.status = 'READY';
    fs.writeFileSync(file, JSON.stringify(record, null, 2) + '\n');
    let admitted = 0;
    let failed = false;
    pi.on('before_provider_request', (event, ctx) => {
      try {
        if (ctx.model?.provider !== 'zhipu' || ctx.model?.id !== 'glm-5.3' ||
            event.payload?.model !== 'glm-5.3' || event.payload?.stream !== true ||
            admitted >= MAX_REQUESTS || failed || Date.now() - started >= BUDGET_SECONDS * 1000) {
          throw new Error('Request admission denied');
        }
        const payload = { ...event.payload };
        delete payload.temperature;
        fs.appendFileSync(path.join(out, 'request-admissions.jsonl'), JSON.stringify({
          admission: admitted + 1, timestamp: new Date().toISOString(), provider: 'zhipu',
          model: 'glm-5.3', stream: payload.stream, tools: payload.tools?.length || 0,
          has_temperature: Object.hasOwn(payload, 'temperature'),
        }) + '\n', { mode: 0o600 });
        admitted += 1;
        return payload;
      } catch {
        process.stderr.write('Review request admission blocked; no retry or fallback.\n');
        process.exit(75);
      }
    });
    pi.on('after_provider_response', event => {
      if (event.status >= 400) failed = true;
      fs.appendFileSync(path.join(out, 'http-responses.jsonl'), JSON.stringify({
        admission: admitted, status: event.status, timestamp: new Date().toISOString(),
      }) + '\n');
    });
  } catch {
    record.status = 'BLOCKED';
    fs.writeFileSync(file, JSON.stringify(record, null, 2) + '\n');
    process.stderr.write('Zhipu bootstrap failed before dispatch; credentials omitted.\n');
    process.exit(70);
  }
}
