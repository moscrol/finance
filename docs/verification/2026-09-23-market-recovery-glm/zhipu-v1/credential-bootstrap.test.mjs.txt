import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import test from 'node:test';
import bootstrap from './credential-bootstrap.mjs';

const root = path.dirname(fileURLToPath(import.meta.url));
test('direct Zhipu model contract, request admission and credential non-persistence', () => {
  const out = fs.mkdtempSync(path.join(root, 'bootstrap-selftest-'));
  const oldOutput = process.env.REVIEW_OUTPUT_DIR;
  const oldExit = process.exit;
  try {
    process.env.REVIEW_OUTPUT_DIR = out;
    let registered;
    const hooks = {};
    bootstrap({ registerProvider: (name, config) => { assert.equal(name, 'zhipu'); registered = config; },
      on: (name, handler) => { hooks[name] = handler; } });
    assert.deepEqual(registered.models[0].input, ['text']);
    assert.equal(registered.models[0].compat.supportsDeveloperRole, false);
    assert.equal(registered.models[0].compat.maxTokensField, 'max_tokens');
    assert.equal(registered.baseUrl, 'https://open.bigmodel.cn/api/coding/paas/v4');
    const context = { model: { provider: 'zhipu', id: 'glm-5.3' } };
    const event = { payload: { model: 'glm-5.3', stream: true, temperature: 0.4, tools: [{}] } };
    for (let i = 0; i < 4; i++) {
      assert.equal(Object.hasOwn(hooks.before_provider_request(event, context), 'temperature'), false);
    }
    process.exit = code => { throw new Error(`blocked:${code}`); };
    assert.throws(() => hooks.before_provider_request(event, context), /blocked:75/);
    assert.equal(fs.readFileSync(path.join(out, 'request-admissions.jsonl'), 'utf8').trim().split('\n').length, 4);
    for (const name of fs.readdirSync(out)) {
      assert.equal(fs.readFileSync(path.join(out, name), 'utf8').includes(registered.apiKey), false);
    }
    assert.equal(fs.existsSync(path.join(out, 'http-responses.jsonl')), false);
  } finally {
    process.exit = oldExit;
    if (oldOutput === undefined) delete process.env.REVIEW_OUTPUT_DIR;
    else process.env.REVIEW_OUTPUT_DIR = oldOutput;
    fs.rmSync(out, { recursive: true });
  }
});
