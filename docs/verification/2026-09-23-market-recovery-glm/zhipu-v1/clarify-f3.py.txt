"""Request a fresh bounded F3 report with evidence-strength constraints; preserve the first report."""
from pathlib import Path
import run

original_make_out = run.make_out

def make_out(name):
    assert name == 'F3-report'
    return original_make_out('F3-report-clarified')

run.make_out = make_out
packet_source = run.ROOT / 'F3-report/packet.md'
packet = packet_source.read_text() + '''

HOST COVERAGE-AUDIT CONSTRAINTS (verify against the supplied probe; no author verdict):
- The fixture pre-seeds the target day, and _plan manually creates a dict. No probe calls build_bridge_day on an empty day and then inserts intervening rows. Describe this as a stale-plan simulation, not a witnessed build-then-stale workflow.
- The tests do not instrument transaction boundaries or interleave two connections. Separate source-level reasoning about BEGIN/recheck order from dynamic evidence. Do not mark all lines of the transaction/rollback implementation as dynamically verified.
- SELECT * / fetchall equality proves final logical SQL values only. Do not call it byte-for-byte preservation, even with a logical qualifier.
- No injected exception occurs after DELETE or partial INSERT. Partial-write rollback remains unverified by this independent probe.
- The final stale-plan check after the successful literal-True replacement compares only target-day rows; the other-day comparison in that test occurs before this last refusal. Describe exactly when the other date was checked.
Report the complete F3 scope again, independently, with verified/not_verified separation. No desired verdict is supplied. The first report is immutable and is not supplied to you.
'''
run.save(run.ROOT / 'F3-clarification-provenance.json', {
    'wrapper_sha256': run.sha(Path(__file__)), 'source_packet_sha256': run.sha(packet_source),
    'original_report_preserved': True, 'original_report_provided_to_reviewer': False,
    'reason': 'Distinguish simulated staleness and static transaction ordering from witnessed execution; prohibit byte-level and partial-rollback overclaims.',
    'scope': 'Same fixed candidate, unchanged independent probe and execution results; no new product test.'})
run.model_stage('F3', 'report', packet)
