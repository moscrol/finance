import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { createHash } from 'node:crypto';

const ROOT = path.dirname(fileURLToPath(import.meta.url));
const digest = value => createHash('sha256').update(value).digest('hex');

export default function (pi) {
  const out = fs.realpathSync(process.env.REVIEW_OUTPUT_DIR || '/invalid-output');
  if (!out.startsWith(ROOT + path.sep)) throw new Error('Invalid review output root');
  const started = Date.now();
  const file = path.join(out, 'credential.json');
  const record = { status: 'RESERVED', provider: 'fomo', model: 'glm-5.3',
    started_at: new Date(started).toISOString(), credential_copied: false,
    runtime_credential_memory_only: true, prices_verified: false };
  fs.writeFileSync(file, JSON.stringify(record, null, 2) + '\n', { flag: 'wx', mode: 0o600 });
  try {
    const template = JSON.parse(fs.readFileSync(path.join(ROOT, 'provider-template.json'), 'utf8'));
    const config = JSON.parse(fs.readFileSync(template.config_source, 'utf8')).providers.fomo;
    if (digest(JSON.stringify(config)) !== template.provider_config_sha256 ||
        config.baseUrl !== template.baseUrl || config.api !== 'openai-completions' ||
        template.model !== 'glm-5.3' || template.provider !== 'fomo') {
      throw new Error('Provider configuration changed');
    }
    if (typeof config.apiKey !== 'string' || !config.apiKey || config.apiKey.startsWith('!')) {
      throw new Error('Expected existing literal credential');
    }
    const model = config.models.find(item => item.id === template.model);
    if (!model || model.reasoning !== false) throw new Error('Model configuration mismatch');
    // Extension registration requires defaults that models.json supplies implicitly.
    const normalized = { ...model, input: model.input ?? ['text'],
      cost: model.cost ?? { input: 0, output: 0, cacheRead: 0, cacheWrite: 0 },
      compat: { ...config.compat, ...model.compat } };
    pi.registerProvider('fomo', { ...config, models: [normalized] });
    Object.assign(record, { status: 'READY', provider_config_sha256: template.provider_config_sha256 });
    fs.writeFileSync(file, JSON.stringify(record, null, 2) + '\n');
    let admitted = 0;
    let providerFailed = false;
    pi.on('before_provider_request', (event, ctx) => {
      try {
        if (ctx.model?.provider !== 'fomo' || ctx.model?.id !== 'glm-5.3' ||
            event.payload?.model !== 'glm-5.3' || event.payload?.stream !== true ||
            admitted >= template.max_requests_per_stage || providerFailed ||
            Date.now() - started >= template.deadline_seconds * 1000) {
          throw new Error('Request admission denied');
        }
        const payload = { ...event.payload };
        delete payload.temperature;
        fs.appendFileSync(path.join(out, 'request-admissions.jsonl'), JSON.stringify({
          admission: admitted + 1, timestamp: new Date().toISOString(),
          provider: 'fomo', model: 'glm-5.3', stream: payload.stream,
          tools: payload.tools?.length || 0,
          has_temperature: Object.hasOwn(payload, 'temperature'),
        }) + '\n', { mode: 0o600 });
        admitted += 1;
        return payload;
      } catch {
        process.stderr.write('Review request admission blocked; no retry or fallback.\n');
        process.exit(75);
      }
    });
    pi.on('after_provider_response', event => {
      if (event.status >= 400) providerFailed = true;
      fs.appendFileSync(path.join(out, 'http-responses.jsonl'), JSON.stringify({
        admission: admitted, status: event.status, timestamp: new Date().toISOString(),
      }) + '\n');
    });
  } catch {
    record.status = 'BLOCKED';
    fs.writeFileSync(file, JSON.stringify(record, null, 2) + '\n');
    process.stderr.write('GLM bootstrap failed before dispatch; credentials omitted.\n');
    process.exit(70);
  }
}
