CODE, NAME = "999999.SZ", "探针股份"

CAL = [d.isoformat() for d in [x for x in _WD if x.weekday() < 5]][:20]


def _ms(d):
    return int((datetime.fromisoformat(d).replace(tzinfo=timezone.utc)
                + timedelta(hours=12)).timestamp() * 1000)


def _ins(con, t, row):
    con.execute(f"INSERT INTO {t} ({', '.join(row)}) VALUES "
                f"({', '.join('?' * len(row))})", list(row.values()))


def _close(i):
    return round(10.0 + 0.5 * i, 2)


def build(tmp, extra_hithink=(), drop_gap_date=None, qfq_only=False):
    db, pq = tmp / "p.db", tmp / "f.parquet"
    con = duckdb.connect(str(db))
    init_db(con)
    for d in [PREV] + CAL:
        _ins(con, "fact_market_daily", {"trade_date": d})
    _ins(con, "fact_stock_daily_hithink", {
        "trade_date": PREV, "stock_ts_code": CODE, "open": 9.4, "high": 9.7,
        "low": 9.3, "close": 9.5, "volume": 5000.0, "turnover": 47500.0,
        "adjusted": "none", "source": "hithink:daily-k",
        "updated_at": datetime(2026, 9, 10, 22, 8)})
    for i, d in enumerate(CAL[:18]):  # PREV..CAL[17]; seg_a_end = CAL[17]
        if drop_gap_date == d and qfq_only:
            _ins(con, "fact_stock_daily_hithink", {
                "trade_date": d, "stock_ts_code": CODE, "open": 1.0, "high": 1.2,
                "low": 0.9, "close": 1.1, "volume": 100.0, "turnover": 111.0,
                "adjusted": "qfq", "source": "hithink:daily-k",
                "updated_at": datetime(2026, 9, 10, 22, 8)})
            continue
        if drop_gap_date == d:
            continue
        _ins(con, "fact_stock_daily_hithink", {
            "trade_date": d, "stock_ts_code": CODE, "open": _close(i) - 0.1,
            "high": _close(i) + 0.2, "low": _close(i) - 0.2, "close": _close(i),
            "volume": float((i + 1) * 10000), "turnover": round(_close(i) * (i + 1) * 1e4, 2),
            "adjusted": "none", "source": "hithink:daily-k",
            "updated_at": datetime(2026, 9, 10, 22, 8)})
    for d, i in extra_hithink:  # C2: later/adjusted rows beyond upper bound
        _ins(con, "fact_stock_daily_hithink", {
            "trade_date": d, "stock_ts_code": CODE, "open": _close(i) - 0.1,
            "high": _close(i) + 0.2, "low": _close(i) - 0.2, "close": _close(i),
            "volume": float((i + 1) * 10000), "turnover": round(_close(i) * (i + 1) * 1e4, 2),
            "adjusted": "none", "source": "hithink:daily-k",
            "updated_at": datetime(2026, 9, 11, 22, 8)})

    def main(i, d, shell=False):
        r = {"trade_date": d, "stock_ts_code": CODE, "stock_name": NAME,
             "close": _close(i), "pre_close": _close(i - 1), "pct_chg": 5.0,
             "amount": 1.0, "turnover": None, "source": "fupanhui:fallback",
             "updated_at": datetime(2026, 9, 1, 8), "open": _close(i) - 0.1,
             "high": _close(i) + 0.2, "low": _close(i) - 0.2,
             "volume": float((i + 1) * 100)}
        if shell:
            for k in ("open", "high", "low", "close", "pre_close", "pct_chg",
                      "amount", "volume"):
                r[k] = None
        return r

    _ins(con, "fact_stock_daily", main(0, CAL[0]))
    _ins(con, "fact_stock_daily", main(1, CAL[1], shell=True))
    _ins(con, "fact_stock_daily", main(2, CAL[2]))
    _ins(con, "fact_stock_daily", main(19, CAL[19]))
    rows = ", ".join(
        f"('{CODE}','CNY','1d','none',{_ms(CAL[i])},{_close(i)-0.1},{_close(i)+0.2},"
        f"{_close(i)-0.2},{_close(i)},{(i+1)*10000}.0,"
        f"{round(_close(i)*(i+1)*1e4,2)})" for i in (17, 18))
    con.execute(f"COPY (SELECT * FROM (VALUES {rows}) AS v(thscode,currency,"
                f"interval,adjusted,date_ms,open_price,high_price,low_price,"
                f"close_price,volume,turnover)) TO '{pq}' (FORMAT PARQUET)")
    con.close()
    spec = BackfillSpec(
        code=CODE, name=NAME, window_start=CAL[0], main_fill_end=CAL[18],
        window_end=CAL[19], shell_date=CAL[1], gap_parallel=tuple(CAL[3:18]),
        gap_parquet=(CAL[18],), expected_total_rows=20,
        stale_technical_dates=(), stale_window_keys=(), pinned_0911={},
        pinned_technical_0911={}, pinned_windows_0911=(),
        expected_technical_count=0, expected_window_counts={5: 0, 10: 0, 20: 0, 60: 0},
        parquet_sha256=hashlib.sha256(pq.read_bytes()).hexdigest(),
        spec_version="probe-v1")
    return db, pq, spec


def test_c2_later_history_and_adjusted_rows_do_not_refuse(tmp_path):
    db, pq, spec = build(tmp_path, extra_hithink=[(CAL[18], 18)])
    con = duckdb.connect(str(db), read_only=True)
    out = _guard(con, spec, Path(pq))  # later valid history beyond max(gap_parallel)
    assert out["mode"] == "apply"
    assert out["prev_day"] == PREV
    # retained snapshot: exactly the pre-existing main rows (D0, D2, D19)
    assert [str(r[0]) for r in out["retained_rows"]] == [CAL[0], CAL[2], CAL[19]]
    con.close()

# ── C3: oracle arithmetic, half-up (not banker's), Decimal-not-float path ──
def test_c3_oracle_pct_half_up():
    # (21.0/6.0-1)*100 = 250.0 exact
    assert _oracle_pct(21.0, 6.0) == 250.0
    # half-up edge: 10.005->10.01 not 10.00
    assert _oracle_pct(_d2(10.005), 100.0) == round(
        (Decimal("10.005") / 100 - 1) * 100, 2)
    # pre_close quantized to 0.01 BEFORE division
    v = _oracle_pct(20.0, 10.004)  # pre->10.00 -> 100.0, not 99.96
    assert abs(v - 100.0) < 1e-9


def _d2(x):
    return float(Decimal(str(x)).quantize(Decimal("0.01"), ROUND_HALF_UP))


def test_c3_oracle_amount_volume():
    assert abs(_oracle_amount(123456789.0) - 1.23456789) < 1e-12
    # half-up: 150000000.005/1e8 = 1.5000000005 -> 1.5 (5th decimal 0)
    assert abs(_oracle_amount(150000000.01) - 1.5) < 1e-9
    # volume /100 half-up: 12345.5/100=123.455 -> 124 (int quantize HALF_UP)
    assert _oracle_volume(12345.5) == 123.0
    assert _oracle_volume(9999.4) == 100.0  # 99.994 -> 100
