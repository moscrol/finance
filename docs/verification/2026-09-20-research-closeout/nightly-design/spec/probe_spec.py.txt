def load(path, name):
    spec = importlib.util.spec_from_file_location(name, ROOT / path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def integrated_sync():
    module = load(SYNC_PATH, 'simulated_integrated_review_sync')
    # This is a bounded callable-level integration probe, not a whole merged tree:
    # take the exact main leaf list and helpers, keeping candidate dependencies.
    tree = ast.parse(INTEGRATED_SOURCE)
    selected = [node for node in tree.body if (
        isinstance(node, ast.FunctionDef) and node.name in {'sync_hithink_step', 'build_local_plan'}
    ) or (
        isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == 'HITHINK_STEPS' for t in node.targets)
    )]
    assert len(selected) == 3
    exec(compile(ast.Module(body=selected, type_ignores=[]), module.__file__, 'exec'), module.__dict__)
    return module

def child_flow(case, *, integrated=False, key=True, fail_label=None):
    with tempfile.TemporaryDirectory(prefix=case + '-', dir=OUT) as temp:
        work = Path(temp)
        target = work / 'sample.duckdb.staging'
        with duckdb.connect(str(target)) as con:
            db.init_db(con)
            con.execute("INSERT INTO fact_stock_daily (trade_date, stock_ts_code, stock_name, close, pre_close, amount, turnover, source) VALUES ('2026-09-15','000001.SZ','样本',10,9,.01,1,'sina:stock_zh_a_daily')")
        args, _ = captured_inputs(work)
        sync = integrated_sync() if integrated else load(SYNC_PATH, 'candidate_review_sync')
        history = load('skills/duckdb-backfill/scripts/backfill_stock_daily_sina.py', 'candidate_history')
        calls = []

        def run_step(label, argv, timeout):
            calls.append({'label': label, 'argv': argv})
            failed = label == fail_label
            return dict(label=label, status='fail' if failed else 'ok', code=2 if failed else 0, elapsed=0)

        with ExitStack() as stack:
            stack.enter_context(patch.dict(os.environ, {'FINANCE_WS': str(work), 'MARKET_FEATURE_STORE_DB': str(target), 'MARKET_FEATURE_STORE_RUN_ID': case, 'REVIEW_SYNC_PLAN': 'local'}))
            stack.enter_context(patch.object(db, 'DB_PATH', target))
            stack.enter_context(patch.object(db, 'DB_DIR', work))
            stack.enter_context(patch.object(sync, 'run_step', run_step))
            stack.enter_context(patch.object(hithink_client, 'has_api_key', lambda: key))
            stack.enter_context(patch.object(recovery, 'load_script', lambda path: history if 'backfill_stock_daily_sina.py' in path else sync))
            error, rc = None, None
            try:
                rc = recovery.child(args)
            except RuntimeError as exc:
                error = str(exc)
            status_path = Path(str(target) + '.status.json')
            status = json.loads(status_path.read_text()) if status_path.exists() else None
            result = dict(case=case, integrated_plan=integrated, key_stub=key, failed_leaf=fail_label,
                          rc=rc, error=error, wrote_success_status=status is not None, calls=calls)
            if error:
                assert status is None
            else:
                assert rc == 0 and status['run_id'] == case and status['ok'] is True
                for call in calls:
                    if call['label'] == 'index-daily':
                        assert '--no-fupanhui-fallback' in call['argv']
            return result


def dated_replay():
    qa = load('skills/duckdb-backfill/scripts/qa_backfill_align.py', 'qa_replay')
    result = []
    with tempfile.TemporaryDirectory(prefix='dated-replay-', dir=OUT) as temp:
        work = Path(temp)
        args, raw = captured_inputs(work)
        recovery.validate_inputs(args.history_day, args.snapshot_day, args.history, args.snapshot)
        snapshot_bytes = args.snapshot.read_bytes()
        (OUT / 'dated-replay-snapshot.json').write_bytes(snapshot_bytes)
        (OUT / 'dated-replay-history.jsonl').write_bytes(args.history.read_bytes())
        snapshot_sha256 = hashlib.sha256(snapshot_bytes).hexdigest()
        for actual_now in (datetime(2026, 9, 17, 22), datetime(2026, 9, 18, 0, 1)):
            target = work / f'replay-{actual_now.day}.duckdb.staging'

            class Clock(datetime):
                @classmethod
                def now(cls, tz=None):
                    return actual_now

            with patch.object(db, 'DB_PATH', target), patch.object(db, 'DB_DIR', work), patch.object(em, 'datetime', Clock), patch.object(em, 'fetch_snapshot', return_value=raw):
                replay = em.sync_fact_stock_daily_snapshot(args.snapshot_day)
                rep = qa.Report()
                with duckdb.connect(str(target), read_only=True) as con:
                    qa.check_latest_snapshot_sources(con, rep, args.snapshot_day)
                failed = [item for item in rep.items if item['level'] == 'FAIL']
                assert args.snapshot.read_bytes() == snapshot_bytes
                result.append(dict(replay_clock=actual_now.isoformat(), input_date_validated=True,
                                   snapshot_sha256=snapshot_sha256,
                                   actual_snapshot_date=replay['snapshot_trade_date'], failures=failed,
                                   snapshot_stitch_argv=recovery.recovery_stitch_command(types.SimpleNamespace(CLI=['python', '-m', 'market_feature_store.cli']), args.snapshot_day, args.history_day)))
    assert result[0]['failures'] == []
    assert len(result[1]['failures']) == 1
    return result
