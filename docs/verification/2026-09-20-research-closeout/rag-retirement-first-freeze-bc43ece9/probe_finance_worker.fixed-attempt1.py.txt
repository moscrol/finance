from __future__ import annotations

HERE = Path(__file__).resolve().parent
TREE = Path('/Users/a77/kb-wt-guarded-maintenance-0920')
FINANCE = Path('/Users/a77/fwp-wt-rag-retirement-closeout-0920')
EXPECTED = '3a21010323eababb54ce8519093fd60dbacb9451'
FEXPECTED = '3f5d0690cec45bb906d758f4165d1558eda78a26'
OUT = HERE / 'worker-fixed'
OUT.mkdir()


def git_identity(path):
    head = subprocess.check_output(['git', '-C', str(path), 'rev-parse', 'HEAD'], text=True).strip()
    dirty = subprocess.check_output(['git', '-C', str(path), 'status', '--porcelain'], text=True)
    return {'sha': head, 'dirty': dirty}


before = {'kb': git_identity(TREE), 'finance': git_identity(FINANCE)}
assert before == {'kb': {'sha': EXPECTED, 'dirty': ''}, 'finance': {'sha': FEXPECTED, 'dirty': ''}}
for key in list(os.environ):
    if key.startswith(('RAG_', 'KB_RAG_')) or key in {'KB_VAULT', 'CONCEPT_VAULT', 'ENTITY_VAULT', 'VECTOR_INDEX_DIR', 'PYTHONPATH'}:
        os.environ.pop(key)
os.environ.update(PYTHONDONTWRITEBYTECODE='1', FINANCE_WS=str(FINANCE),
                  KB_ACCESS_LOG_PATH=str(OUT / 'access-log.jsonl'),
                  HF_HUB_OFFLINE='1', TRANSFORMERS_OFFLINE='1', RAG_WORKER_KEEPALIVE_SECONDS='0')
sys.path[:0] = [str(TREE / 'skills/lib'), str(FINANCE)]
from rag.embedder import get_embedder
from rag.generation import activate, approve, consumer_environment, resolve_startup, verify_manifest
from rag.maintenance import prepare
from rag.store import build
from intelligence.services import kb_rag, rag_worker

wiki = OUT / 'source/wiki'
(wiki / 'sources').mkdir(parents=True)
(wiki.parent / 'raw').mkdir()
root = OUT / 'generations-root'
standard, full = OUT / 'legacy-standard', OUT / 'legacy-full'


def source(epoch):
    (wiki / 'sources/order.md').write_text('---\nevidence_layer: L3\nfact_hardness: hard_fact\n'
        'source_type: official_disclosure\npublish_time: 2026-09-01\navailable_time: 2026-09-01\n'
        'sources: [scratch-filing]\n---\n# Order\n\nepoch' + epoch + ' signed order confirmed.\n')
    (wiki.parent / 'raw/original.md').write_text('# Original\n\nraw' + epoch + ' isolated source text.\n')


def prepare_epoch(epoch):
    probes = {
        'standard': [{'query': 'epoch' + epoch, 'expected_paths': ['wiki/sources/order.md']},
                     {'query': 'epoch' + epoch, 'filters': {'evidence_layer': 'L1'}, 'expect_empty': True}],
        'full': [{'query': 'raw' + epoch, 'expected_paths': ['raw/original.md']},
                 {'query': 'epoch' + epoch, 'filters': {'evidence_layer': 'L1'}, 'expect_empty': True}],
    }
    path = prepare(root, generation=epoch, vault=wiki, standard=standard, full=full, probes=probes)
    approve(path, approved_by='independent scratch review')
    activate(path)
    return path


all_workers = []
processes = []
rows = []


def workers_for(path):
    binding = resolve_startup(root=root)
    assert binding['manifest'] == str(path)
    # The launch environment must remove an unrelated higher-priority finance index override.
    os.environ['VECTOR_INDEX_DIR'] = str(OUT / 'unrelated-index')
    env = consumer_environment(binding)
    assert 'VECTOR_INDEX_DIR' not in env
    os.environ.clear()
    os.environ.update(env)
    workers = [rag_worker._worker_for(binding['environment']['KB_RAG_PYTHON'],
        Path(binding['environment']['KB_RAG_CODE_ROOT']), path.parent / kind,
        Path(binding['environment']['KB_VAULT'])) for kind in ('standard', 'full')]
    all_workers.extend(workers)
    return workers


def query(worker, word, extra=()):
    return worker.query(['query', word, '--mode', 'bm25', '--json', *extra], timeout=20)


def check_epoch(workers, epoch, stage):
    for kind, worker, word, expected_path in zip(('standard', 'full'), workers,
            ('epoch' + epoch, 'raw' + epoch), ('wiki/sources/order.md', 'raw/original.md')):
        warm = worker.prewarm(['query', word, '--mode', 'bm25', '--json'], timeout=20)
        assert warm.returncode == 0 and warm.model_load_count == 1, warm
        if worker._process not in processes:
            processes.append(worker._process)
        first = query(worker, word)
        second = query(worker, word)
        assert first.returncode == second.returncode == 0, (first.stderr, second.stderr)
        hits = json.loads(second.stdout)
        assert expected_path in {item['file_path'] for item in hits}, hits
        assert word in second.stdout and second.model_load_count == 1, second
        negative = query(worker, word, ['--evidence-layer', 'L1'])
        assert negative.returncode == 0 and json.loads(negative.stdout) == [], negative
        rows.append({'stage': stage, 'kind': kind, 'epoch': epoch,
                     'pid': worker._process.pid, 'model_load_count': second.model_load_count,
                     'hits': hits, 'negative_hits': [], 'status': worker.status()})


try:
    source('alpha')
    for path, include_raw in ((standard, False), (full, True)):
        build(get_embedder('hash'), wiki, include_raw=include_raw, max_files=0).save(path)
    old = prepare_epoch('alpha')
    old_hash = hashlib.sha256((old.parent / 'source/wiki/sources/order.md').read_bytes()).hexdigest()
    old_workers = workers_for(old)
    check_epoch(old_workers, 'alpha', 'initial-warm')
    assert rag_worker.status()['state'] == 'ready' and rag_worker.status()['active'] == 2
    source('beta')
    newer = prepare_epoch('beta')
    assert old_hash == hashlib.sha256((old.parent / 'source/wiki/sources/order.md').read_bytes()).hexdigest()
    assert verify_manifest(old)['generation'] == 'alpha'
    for kind, worker in zip(('standard', 'full'), old_workers):
        before_first_query = worker.status()
        assert before_first_query['state'] == 'failed' and before_first_query['active'] is False
        assert before_first_query['generation_status'] == 'retired'
        assert before_first_query['generation_reason'] == 'current_generation_changed'
        assert worker._process is not None and worker._process.poll() is None
        rows.append({'stage': 'retired-before-first-query-fixed', 'kind': kind,
                     'status': before_first_query})
    aggregate = rag_worker.status()
    assert aggregate['state'] == 'failed' and aggregate['active'] == 0
    consumer = kb_rag.retrieve('epochalpha', Path(os.environ['KB_VAULT']),
                               mode='bm25', timeout=20)
    assert consumer.telemetry.status == 'error'
    assert consumer.telemetry.fallback_reason == 'persistent_worker_generation_unavailable'
    rows.append({'stage': 'retired-consumer-no-cli-fallback',
                 'status': consumer.telemetry.status,
                 'fallback_reason': consumer.telemetry.fallback_reason})
    for kind, worker in zip(('standard', 'full'), old_workers):
        try:
            query(worker, 'epochalpha')
            raise AssertionError('retired worker query unexpectedly succeeded')
        except RagGenerationUnavailable as retired:
            assert retired.reason == 'current_generation_changed'
        rows.append({'stage': 'retired-after-switch', 'kind': kind,
                     'status': worker.status()})
    new_workers = workers_for(newer)
    check_epoch(new_workers, 'beta', 'new-generation-with-old-registered')
    assert rag_worker.status()['state'] == 'failed'
    rag_worker.close_all()
    new_workers = workers_for(newer)
    check_epoch(new_workers, 'beta', 'new-generation-after-close-all')
    assert rag_worker.status()['state'] == 'ready' and rag_worker.status()['active'] == 2
    activate(old)
    assert rag_worker.status()['state'] == 'failed'
    rag_worker.close_all()
    rollback_workers = workers_for(old)
    check_epoch(rollback_workers, 'alpha', 'explicit-rollback-warm')
    assert rag_worker.status()['state'] == 'ready' and rag_worker.status()['active'] == 2
finally:
    rag_worker.close_all()
    for worker in all_workers:
        worker.close()
    assert all(process is not None and process.poll() is not None for process in processes)
after = {'kb': git_identity(TREE), 'finance': git_identity(FINANCE)}
assert after == before
report = {'passed': True, 'before': before, 'after': after,
          'probe_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
          'rows': rows, 'real_finance_worker': True, 'changed_v2_source': True,
          'closed_workers': len(all_workers), 'production_service': False,
          'closed_processes': len(processes),
          'network': False, 'model': 'hash/offline; BM25 queries'}
(OUT / 'results.json').write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({'passed': True, 'real_worker_rows': len(rows),
                  'closed_workers': len(all_workers), 'evidence': str(OUT / 'results.json')}))
