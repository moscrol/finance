def git(*args):
    return subprocess.check_output(['git', '-C', str(ROOT), *args])

def original(ref, path):
    data = git('show', f'{ref}:{path}')
    identities[f'{ref}:{path}'] = hashlib.sha256(data).hexdigest()
    return json.loads(data)

identities = {}

before = git('status', '--porcelain').decode()
assert not before

expected = {item['path']: item['sha256'] for item in manifest['files']}
members = {}

    relative = f'components/postcheck-{theme}-limit-members.json'
    raw = (ARCHIVE / relative).read_bytes()

    identities[str(ARCHIVE / relative)] = digest
    result = json.loads(raw)['result']
    assert not result.get('gaps')
    members[theme] = [dict(part.split('=', 1) for part in item['detail'].split('；'))
