"""Redirect only test receipt and temporary-output destinations, retaining repo gates."""
from pathlib import Path
import sys

ROOT = Path(__file__).resolve().parent
SOURCE = Path('/Users/a77/fwp-wt-history-closeout-0920')
sys.path.insert(0, str(SOURCE))
import pytest

name, *args = sys.argv[1:]
(ROOT / 'temporary').mkdir(parents=True, exist_ok=True)
(ROOT / 'pytest-cache').mkdir(parents=True, exist_ok=True)
class ReceiptDestination:
    @pytest.hookimpl(trylast=True)
    def pytest_configure(self, config):
        matched = []
        for plugin in config.pluginmanager.get_plugins():
            file = getattr(plugin, '__file__', None)
            if file and Path(file).resolve() == SOURCE / 'conftest.py':
                plugin._RECEIPT_DIR = ROOT / 'receipts' / name
                matched.append(file)
        assert len(matched) == 1, matched

raise SystemExit(pytest.main([
    '-q', '-rA', '-o', 'cache_dir=' + str(ROOT / 'pytest-cache' / name),
    '--basetemp=' + str(ROOT / 'temporary' / name),
    '--junitxml=' + str(ROOT / (name + '.junit.xml')),
    *args,
], plugins=[ReceiptDestination()]))
