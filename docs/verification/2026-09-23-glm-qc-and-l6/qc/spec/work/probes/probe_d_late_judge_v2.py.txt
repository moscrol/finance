# -*- coding: utf-8 -*-
"""独立探针 D v2：C3 迟到判官 payload 拒收(不运行于本阶段)。

v2 修正: _JudgeCall 与 SemanticEpisodeOutcome 均有必填字段
(report/unavailable/correlated; verified/status/public_answer/judge_status)。
边界: 真实 worker + 本地假 HTTP(127.0.0.1:26005, ~2.9s 迟到响应),
共享判官预算 0.8s。
D1 迟到 payload 拒收: 判官走非流式 chat 路径, 0.8s 共享 deadline ->
   ~0.8s 内 LLMDeadlineExceeded; 台账 status=failed 且 reason 归
   timeout/deadline 一类; payload 内容从未被接受。
D2 根剩余反映真实墙钟: elapsed ~0.8s 且 root remaining ~0。
D3 判官时钟: _attach_judge_clock 复制 timeout_asked/remaining_seconds_at_entry;
   judge_status="unavailable"(=report_received=False, unavailable=True 语义)
   时 pending judge_request 保留。
"""
import http.server
import json
import sys
import threading
import time

sys.path.insert(0, ".")
from intelligence.services import llm_refine as R
from intelligence.services import episode_semantic_verifier as V

PORT = 26005


class Handler(http.server.BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def do_POST(self):
        self.rfile.read(int(self.headers.get("Content-Length", 0) or 0))
        time.sleep(2.9)  # 迟到 payload
        body = json.dumps({"choices": [{"message": {"content": "LATE_VERDICT"},
                                        "finish_reason": "stop"}]}).encode()
        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, *a):
        pass


def _stub_outcome():
    verified = object()
    return V.SemanticEpisodeOutcome(
        verified=verified, status="judge_failed", public_answer="",
        judge_status="unavailable")


def main():
    results = []

    def check(name, cond, detail=""):
        results.append((name, bool(cond), detail))
        print(("PASS" if cond else "FAIL"), name, detail)

    srv = http.server.ThreadingHTTPServer(("127.0.0.1", PORT), Handler)
    threading.Thread(target=srv.serve_forever, daemon=True).start()
    time.sleep(0.3)

    prov = R.LLMProvider(name="fake", api_key="k",
                         base_url=f"http://127.0.0.1:{PORT}", model="m")
    msgs = [{"role": "user", "content": "judge"}]

    judge_deadline = R.Deadline(time.monotonic() + 0.8)
    t0 = time.monotonic()
    accepted = {"value": None}
    with R.call_ledger_scope() as ledger:
        try:
            out = R._post_chat(prov, msgs, 10.0, deadline=judge_deadline)
            accepted["value"] = out
            check("D1_late_rejected", False, f"payload accepted: {out[:12]!r}")
        except R.LLMDeadlineExceeded:
            el = time.monotonic() - t0
            check("D1_late_rejected", el < 2.0 and accepted["value"] is None,
                  f"elapsed={el:.2f}s (须~0.8s 非2.9s)")
        except Exception as e:
            check("D1_late_rejected", False, f"wrong exc {type(e).__name__}")
        failed = [r for r in ledger.records if r.status == "failed"]
        check("D1_recorded_failed", len(failed) == 1,
              f"n_failed={len(failed)}")
        if failed:
            reason = failed[0].reason or ""
            check("D1_reason_timeout_like",
                  any(k in reason.lower() for k in
                      ("timeout", "deadline", "llmdeadline")),
                  f"reason={reason!r}")

    # D2 根剩余反映真实墙钟
    el = time.monotonic() - t0
    rem = judge_deadline.remaining()
    check("D2_root_remaining_wallclock",
          abs(el - 0.8) < 0.7 and rem < 0.05,
          f"elapsed={el:.2f}s remaining={rem:.3f}")

    # D3 判官时钟附着 (report=None => report_received=False; unavailable=True)
    try:
        base = _stub_outcome()
    except TypeError as e:
        check("D3_clock_attached", False, f"outcome ctor: {e}")
        fails = [n for n, ok, _ in results if not ok]
        print(f"SUMMARY probe_d: {len(results)-len(fails)}/{len(results)} passed")
        sys.exit(1)
    jc = V._JudgeCall(
        report=None, unavailable=True, correlated=False,
        timeout_asked=0.8, timeout_configured=10.0,
        remaining_seconds_at_entry=0.8, judge_attempt_index=1,
        request={"messages": msgs})
    out = V._attach_judge_clock(base, jc)
    check("D3_clock_attached",
          out.timeout_asked == 0.8 and out.timeout_configured == 10.0
          and out.remaining_seconds_at_entry == 0.8
          and out.judge_attempt_index == 1,
          f"asked={out.timeout_asked} entry={out.remaining_seconds_at_entry}")
    check("D3_unavailable_keeps_request",
          out.judge_status == "unavailable" and out.judge_request is not None,
          "pending request preserved for unavailable judge")

    fails = [n for n, ok, _ in results if not ok]
    print(f"SUMMARY probe_d: {len(results)-len(fails)}/{len(results)} passed")
    sys.exit(1 if fails else 0)


if __name__ == "__main__":
    main()
