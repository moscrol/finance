import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import { fileURLToPath } from 'node:url';
import { install } from './review.mjs';
const root = path.dirname(fileURLToPath(import.meta.url));
const out = path.join(root, process.argv[2] || 'caps-preflight');
fs.mkdirSync(out);
fs.mkdirSync(path.join(out, 'commands'));
const tools = {};
const now = Date.now();
const token = 'fake.' + Buffer.from(JSON.stringify({plan: 'plus', exp: now / 1000 + 3000, plan_exp: now / 1000 + 3000})).toString('base64url') + '.fake';
install({registerProvider() {}, on() {}, registerTool(t) {tools[t.name] = t;}}, {out, token, clock: () => now});
const config = JSON.parse(fs.readFileSync(path.join(root, 'config.json')));
const work = path.join(root, 'work');
const target = path.join(work, 'cap-fixture.txt');
const body = Array.from({length: 250}, (_, i) => String(i + 1).padStart(3, '0') + ' x\n').join('');
await tools.write.execute('write-lines', {path: target, content: body});
let result = await tools.read.execute('read-lines', {path: target, offset: 1, limit: 99999});
let text = result.content[0].text;
assert.match(text, /120 x\n/);
assert.doesNotMatch(text, /121 x\n/);
assert.match(text, /next offset=121/);
const wide = Array.from({length: 150}, (_, i) => String(i + 1).padStart(3, '0') + 'x'.repeat(96) + '\n').join('');
await tools.write.execute('write-bytes', {path: target, content: wide});
result = await tools.read.execute('read-bytes', {path: target, offset: 1, limit: 120});
text = result.content[0].text;
assert.match(text, /next offset=61/);
assert.equal(Buffer.byteLength(text.split('\n[truncated;')[0]), 6000);
result = await tools.read.execute('continue-bytes', {path: target, offset: 61, limit: 1});
assert.match(result.content[0].text, /^061x/);
await tools.write.execute('write-giant-line', {path: target, content: 'x'.repeat(7000) + '\n'});
result = await tools.read.execute('read-giant-line', {path: target, offset: 1, limit: 1});
assert.match(result.content[0].text, /Single line exceeds 6000 bytes/);
result = await tools.bash.execute('bash-cap', {command: config.python + ' -B -c "print(chr(120)*10000)"', timeout: 20});
assert.equal(result.content[0].text.slice(0, 6000), 'x'.repeat(6000));
assert.match(result.content[0].text, /Response capped/);
const commandDir = fs.readdirSync(path.join(out, 'commands')).sort().at(-1);
assert.equal(fs.readFileSync(path.join(out, 'commands', commandDir, 'output.log'), 'utf8'), 'x'.repeat(10000) + '\n');
const plan = JSON.parse(fs.readFileSync(path.join(root, 'group-plan.json')));
const claims = plan.groups.flatMap(group => group.claims);
assert.equal(claims.length, 10);
assert.deepEqual([...new Set(claims)].sort(), Array.from({length: 10}, (_, i) => 'C' + (i + 1)).sort());
const receipt = {status: 'PASS', real_model_requests: 0, checks: ['read line cap 120', 'read byte cap 6000', 'continuation offset exact', 'oversized single line explicitly refused', 'bash response capped with raw output unchanged', 'C1-C10 grouped exactly once']};
fs.writeFileSync(path.join(out, 'receipt.json'), JSON.stringify(receipt, null, 2));
console.log(JSON.stringify(receipt));
