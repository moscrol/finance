def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def check_stage(root,name):
    out=root/name
    record=json.loads((out/'execution.json').read_text())
    assert record['complete'] and record['revision']==REV and record['baseline']==BASE
    assert record['inputs_unchanged']
    for path,value in record['inputs_sha256'].items():
        assert sha(root/path)==value,(name,path)
    for path,value in record['artifact_sha256'].items():
        assert sha(out/path)==value,(name,path)
    admissions=[json.loads(line) for line in (out/'request-admissions.jsonl').read_text().splitlines()]
    assert len(admissions)==record['requests']
    return {'raw_status':record['status'],'requests':len(admissions),'elapsed_seconds':record['elapsed_seconds'],
            'controller_timed_out':record['timed_out'],'response_statuses':dict(Counter(r['status'] for r in record['responses'])),
            'terminal_report':(out/'REPORT.md').exists(),'errors':record['errors']}
