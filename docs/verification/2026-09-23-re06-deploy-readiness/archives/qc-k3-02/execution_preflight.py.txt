"""Local controller checks only; never counted as independent reviewer tests."""
import ast
import hashlib
import importlib.util
import json
from pathlib import Path
import tempfile

ROOT=Path(__file__).resolve().parent
for name in ['run_probe.py','check_execution.py','stage_admission.py']:
    ast.parse((ROOT/name).read_text())
spec=importlib.util.spec_from_file_location('execution_preflight_target',ROOT/'run_probe.py')
module=importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
with tempfile.TemporaryDirectory(dir=ROOT/'work',prefix='controller-check-') as folder:
    path=Path(folder)/'junit.xml'
    path.write_text('<testsuites><testsuite tests="4"><testcase name="pass"/><testcase name="red"><failure/></testcase><testcase name="skip"><skipped/></testcase><testcase name="xfail"><skipped type="pytest.xfail"/></testcase></testsuite></testsuites>')
    counted=module.junit_counts(path,'collected 4 items\n')
    assert counted=={'tests':4,'passed':1,'failed':1,'errors':0,'skipped':1,'xfailed':1,'collected':4,'collected_source':'pytest_stdout'}
    assert module.junit_counts(path,'no collection line')['collected'] is None
    assert module.junit_counts(path,'',ui=True)['collected']==4
    path.write_text('<testsuites><testsuite tests="1"><testcase name="collection"><error/></testcase></testsuite></testsuites>')
    counted=module.junit_counts(path,'collected 0 items / 1 error\n')
    assert counted['collected']==0 and counted['tests']==1 and counted['errors']==1
    path.write_text('<testsuites><testsuite tests="1"><testcase name="positive"><failure>assert 1 == 2</failure></testcase></testsuite></testsuites>')
    counted=module.junit_counts(path,'collected 1 item\n')
    assert counted['failed']==1 and counted['passed']==0 and counted['collected']==1
receipt={'status':'PASS','real_model_requests':0,'pytest_invocations':0,'candidate_tests':0,
         'independent_qc_positive_control':False,
         'checks':['source AST','mixed JUnit counters','missing collection is unknown, not zero','collection error is not one collected test','Vitest declared count','expected-red parser'],
         'source_sha256':{name:hashlib.sha256((ROOT/name).read_bytes()).hexdigest() for name in ['run_probe.py','check_execution.py','stage_admission.py']}}
with (ROOT/'execution-preflight.json').open('x') as stream: json.dump(receipt,stream,indent=2)
print(json.dumps(receipt))
