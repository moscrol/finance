"""Record exactly one reviewer-selected invocation; no candidate verdict is inferred."""
import argparse
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import re
import signal
import subprocess
import sys
import time
import xml.etree.ElementTree as ET

ROOT=Path(__file__).resolve().parent
CONFIG=json.loads((ROOT/'config.json').read_text())
TREE=Path(CONFIG['tree'])
PYTHON=CONFIG['python']
AUTHOR={
    'controller':[],
    'e2':['test_e2_local_question_delivery.py'],
    'timer':['test_re06_activity_consent_gate_effect.py','test_research_evolution_i11_consent.py'],
    'consent':['test_re06_consent_fold_shared.py','test_re06_consent_gate_toctou.py','test_research_evolution_i11_consent.py'],
}

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def junit_counts(path, stdout, ui=False):
    if not path.exists(): return None
    root=ET.parse(path).getroot()
    cases=list(root.iter('testcase'))
    result={'tests':len(cases),'passed':0,'failed':0,'errors':0,'skipped':0,'xfailed':0}
    for case in cases:
        if case.find('error') is not None: result['errors']+=1
        elif case.find('failure') is not None: result['failed']+=1
        elif case.find('skipped') is not None:
            result['xfailed' if case.find('skipped').get('type')=='pytest.xfail' else 'skipped']+=1
        else: result['passed']+=1
    collected=re.search(r'(?m)^collected (\d+) items?(?:\s|$)',stdout)
    result['collected']=int(collected.group(1)) if collected else (sum(int(s.get('tests','0')) for s in root.iter('testsuite')) if ui else None)
    result['collected_source']='vitest_junit' if ui else 'pytest_stdout'
    return result

def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('group',choices=AUTHOR)
    parser.add_argument('role',choices=['positive_control','reviewer','author','ui_reviewer','ui_author','controller_preflight','ui_controller_preflight'])
    parser.add_argument('--probe',type=Path)
    args=parser.parse_args()
    work=ROOT/'work'/args.group
    frozen=ROOT/'inputs'/('frozen-explore-'+args.group+'-v2')
    assert (args.group=='controller')==(args.role in {'controller_preflight','ui_controller_preflight'})
    ui=args.role.startswith('ui_')
    assert not ui or args.group in {'timer','controller'}
    assert args.probe is None or args.role in {'reviewer','ui_reviewer'}
    if args.group=='controller':
        sources=[work/'probes'/('controller.test.tsx' if ui else 'test_controller.py')]
    elif args.role=='author':
        sources=[TREE/'intelligence/tests'/name for name in AUTHOR[args.group]]
    elif args.role=='ui_author':
        sources=[TREE/'intelligence/webapp/src/components/ResearchActivityControl.test.tsx',TREE/'intelligence/webapp/src/researchActivity.test.ts']
    else:
        filename={'positive_control':'test_positive_control.py','reviewer':'test_reviewer.py','ui_reviewer':'activity-reviewer.test.tsx'}[args.role]
        if args.probe:
            probe=args.probe.resolve()
            assert probe.is_relative_to((work/'probes').resolve()) and probe.name!=filename
            sources=[probe]
        else: sources=[frozen/'probes'/filename]
    assert all(p.is_file() and not p.is_symlink() for p in sources)
    out=work/'runs'/(args.role+'-'+str(time.time_ns()))
    out.mkdir(parents=True)
    junit=out/'junit.xml'; log=out/'stdout.log.txt'
    env=dict(os.environ)
    env.update(PYTHONPATH=str(TREE),PYTHONDONTWRITEBYTECODE='1',FWP_TEST_RECEIPT='0',CI='1',NO_COLOR='1',FORCE_COLOR='0')
    if ui:
        env['REVIEW_UI_KIND']='author' if args.role=='ui_author' else 'reviewer'
        if args.role!='ui_author': env['REVIEW_UI_PROBE']=str(sources[0])
        command=['/opt/homebrew/bin/node',str(TREE/'intelligence/webapp/node_modules/vitest/vitest.mjs'),'run',
                 '--config',str(ROOT/'inputs/ui.vitest-v2.config.mjs'),'--configLoader','native',
                 '--reporter=default','--reporter=junit','--outputFile='+str(junit)]
    else:
        command=[PYTHON,'-m','pytest','-o','addopts=','-ra','--color=no','-p','no:cacheprovider',
                 '--junitxml='+str(junit),'--basetemp='+str(out/'tmp'),*[str(p) for p in sources]]
    command=['/usr/bin/sandbox-exec','-f',str(ROOT/'tools.sb'),*command]
    record={'revision':CONFIG['revision'],'baseline':CONFIG['baseline'],'group':args.group,'role':args.role,
            'started_at':datetime.now(timezone.utc).isoformat(),'complete':False,'command':command,
            'cwd':str(TREE),'sources_sha256':{str(p):sha(p) for p in sources},'source_kind':'host_controller' if args.group=='controller' else 'author_existing' if args.role in {'author','ui_author'} else 'independent_reviewer',
            'wrapper_sha256':sha(Path(__file__)),'ui_config_sha256':sha(ROOT/'inputs/ui.vitest-v2.config.mjs') if ui else None,
            'junit':str(junit),'log':str(log),'probe_variant':'host_preflight' if args.group=='controller' else 'additional_or_repaired' if args.probe else 'initial_frozen'}
    receipt=out/'receipt.json'
    receipt.write_text(json.dumps(record,indent=2)+'\n')
    rc=None; blocked=None; started=time.monotonic()
    process=None
    def interrupted(signum, _frame):
        if process is not None and process.poll() is None:
            os.killpg(process.pid,signal.SIGTERM)
            try: process.wait(timeout=5)
            except subprocess.TimeoutExpired:
                os.killpg(process.pid,signal.SIGKILL); process.wait()
        raise SystemExit(128+signum)
    signal.signal(signal.SIGTERM,interrupted)
    signal.signal(signal.SIGINT,interrupted)
    admission=subprocess.run([PYTHON,'-I',str(ROOT/'resource_pytest.py'),'--dry-run'],env=env,cwd=TREE,capture_output=True,text=True,timeout=10)
    record['resource_admission_stdout']=admission.stdout
    record['resource_inspection_boundary']='trusted_host_before_sandboxed_test_child'
    if admission.returncode:
        rc=admission.returncode; blocked='BLOCKED_RESOURCE_GATE' if rc==75 else 'BLOCKED_RESOURCE_INSPECTION'
        log.write_text(admission.stdout+admission.stderr)
    if rc is None:
        with log.open('xb') as stream:
            process=subprocess.Popen(command,cwd=TREE,env=env,stdout=stream,stderr=subprocess.STDOUT,start_new_session=True)
            record['pid']=process.pid
            receipt.write_text(json.dumps(record,indent=2)+'\n')
            try: rc=process.wait(timeout=90)
            except subprocess.TimeoutExpired:
                blocked='BLOCKED_PROBE_TIMEOUT'
                os.killpg(process.pid,signal.SIGTERM)
                try: rc=process.wait(timeout=5)
                except subprocess.TimeoutExpired:
                    os.killpg(process.pid,signal.SIGKILL); rc=process.wait()
    stdout=log.read_text(errors='replace')
    if rc==75: blocked='BLOCKED_RESOURCE_GATE'
    counts=junit_counts(junit,stdout,ui)
    record.update(complete=True,exit_code=rc,blocked=blocked,elapsed_seconds=round(time.monotonic()-started,3),
                  sources_unchanged=record['sources_sha256']=={str(p):sha(p) for p in sources},counts=counts,
                  log_sha256=sha(log),junit_sha256=sha(junit) if junit.exists() else None,
                  candidate_semantic_verdict=None)
    receipt.write_text(json.dumps(record,indent=2)+'\n')
    sys.stdout.write(stdout)
    print('\nEXECUTION_RECEIPT '+json.dumps({'path':str(receipt),'sha256':sha(receipt),'exit_code':rc,'counts':counts,'blocked':blocked}),flush=True)
    return rc

if __name__=='__main__':
    raise SystemExit(main())
