def now():
    return datetime.now(timezone.utc).isoformat()

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def identity():
    def git(*args):
        return subprocess.check_output(['git','-C',str(TREE),*args],text=True).strip()
    return {'revision':git('rev-parse','HEAD'),'status':git('status','--porcelain')}

def save():
    tmp=ROOT/'receipt.json.tmp'
    tmp.write_text(json.dumps(record,ensure_ascii=False,indent=2)+'\n')
    tmp.replace(ROOT/'receipt.json')

def admit(leaf):
    deadline=time.monotonic()+900
    while True:
        ps=subprocess.check_output(['ps','-axo','pid=,command='],text=True)
        pattern=re.compile(r'^\S*/(?:Python|python(?:3(?:\.\d+)?)?)\s+(?:-B\s+)?-m\s+pytest(?:\s|$)|^\S*/pytest(?:\s|$)')
        processes=[line for line in ps.splitlines() if len(line.strip().split(None,1))==2 and pattern.search(line.strip().split(None,1)[1])]
        load=list(os.getloadavg())
        free=shutil.disk_usage(ROOT).free/1024**3
        projected=len(processes)+(leaf=='python')
        allowed=load[0]<=8 and projected<=2 and free>=8
        record['resource_admissions'].append({'leaf':leaf,'at':now(),'load':load,'pytest_processes':len(processes),
              'projected_pytest_processes':projected,'free_disk_gib':round(free,3),'allowed':allowed})
        save()
        if allowed:
            assert identity()=={'revision':REVISION,'status':''}
            return True
        if time.monotonic()>=deadline:
            record['leaves'][leaf]={'status':'BLOCKED_RESOURCE_GATE'}
            record.update(status='BLOCKED_RESOURCE_GATE',finished_at=now())
            save()
            raise SystemExit(75)
        time.sleep(10)

def command(name,args,timeout=1800):
    log=ROOT/(name+'.log.txt')
    start=time.monotonic()
    stop_reason=None
    minimum_free=shutil.disk_usage(ROOT).free/1024**3
    with log.open('xb') as stream:
        proc=subprocess.Popen(args,cwd=TREE,env=ENV,stdout=stream,stderr=subprocess.STDOUT,start_new_session=True)
        record['active_command']={'name':name,'pid':proc.pid,'at':now(),'timeout_seconds':timeout}
        save()
        while proc.poll() is None:
            free=shutil.disk_usage(ROOT).free/1024**3
            minimum_free=min(minimum_free,free)
            if free<8:
                stop_reason='BLOCKED_DISK_FLOOR'
            elif time.monotonic()-start>=timeout:
                stop_reason='BLOCKED_PROCESS_TIMEOUT'
            if stop_reason:
                os.killpg(proc.pid,signal.SIGINT)
                try:
                    proc.wait(timeout=20)
                except subprocess.TimeoutExpired:
                    os.killpg(proc.pid,signal.SIGTERM)
                    try:
                        proc.wait(timeout=10)
                    except subprocess.TimeoutExpired:
                        os.killpg(proc.pid,signal.SIGKILL)
                        proc.wait()
                break
            time.sleep(2)
    result={'command':args,'exit_code':proc.returncode,'timed_out':stop_reason=='BLOCKED_PROCESS_TIMEOUT',
            'stop_reason':stop_reason,'minimum_free_disk_gib':round(minimum_free,3),
            'elapsed_seconds':round(time.monotonic()-start,3),'log':log.name,'log_sha256':sha(log),'log_bytes':log.stat().st_size}
    record.pop('active_command',None)
    save()
    print(json.dumps({'step':name,**result}),flush=True)
    return result

def status(checks):
    blocked=next((c['stop_reason'] for c in checks if c.get('stop_reason')),None)
    return blocked or ('PASS' if all(c['exit_code']==0 for c in checks) else 'FAIL')
