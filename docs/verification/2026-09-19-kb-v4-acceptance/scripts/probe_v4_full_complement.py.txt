def save(name, payload):
    (OUT / name).write_text(json.dumps(payload, ensure_ascii=False, indent=2))


def hashes():
    result = {}
    for name in ('.rag_index', '.rag_index_full'):
        for path in sorted((PAIR / name).iterdir()):
            if path.is_file():
                h = hashlib.sha256()
                with path.open('rb') as handle:
                    for block in iter(lambda: handle.read(1024 * 1024), b''):
                        h.update(block)
                result[path.relative_to(PAIR).as_posix()] = h.hexdigest()
    return result

def check_filters(tel, expected):
    assert tel.applied_filters == expected, asdict(tel)
    assert tel.filter_verification == ('verified' if expected else 'not_requested')
    if expected:
        assert tel.metadata_version == 1 and tel.receipt_received
        assert not tel.metadata_update_required


def record(label, result, name, mode, filters=None, empty=False, expected_path=None):
    save(f'{label}.json', asdict(result))
    tel = result.telemetry
    assert Path(tel.index_dir).resolve() == PAIR / name
    assert tel.effective_mode == mode and not tel.fallback_reason, asdict(tel)
    check_filters(tel, filters or {})
    if empty:
        assert not result.ok and not result.hits and tel.status == 'empty', asdict(result)
    else:
        assert result.ok and result.hits, (label, result.warning)
        for hit in result.hits:
            assert hit.index_freshness == 'fresh'
            assert not any(marker in hit.excerpt for marker in ('<<<<<<<', '>>>>>>>'))
            if filters:
                assert hit.evidence_scope_bound
                if 'evidence_layer' in filters:
                    assert hit.evidence_layer == filters['evidence_layer']
        if expected_path:
            assert any(h.file_path == expected_path for h in result.hits)
    row = {'label': label, 'status': 'pass', 'index': name, 'mode': mode,
           'expected_empty': empty, 'hits': len(result.hits),
           'filter_verification': tel.filter_verification,
           'paths': [h.file_path for h in result.hits]}
    results.append(row)
    save('results.json', results)
    print(json.dumps(row, ensure_ascii=False), flush=True)
