class _OkResp:
    """模拟 200 JSON 响应（可选 headers）。"""

    def __init__(self, payload, headers=None):
        self.status = 200
        self._raw = json.dumps(payload).encode("utf-8")
        self.headers = headers or {}

    def read(self):
        return self._raw

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        return False


class _RawResp:
    """模拟任意原始正文的 HTTP 响应（用于非 JSON 正文用例）。"""

    def __init__(self, body: bytes, status: int = 200, headers=None):
        self.status = status
        self._raw = body
        self.headers = headers or {}

    def read(self):
        return self._raw

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        return False


class _FakeWire:
    """按队列返回响应/抛 HTTPError，记录请求次数、睡眠时间与传入 timeout。"""

    def __init__(self, monkeypatch, events):
        self.events = list(events)
        self.calls = 0
        self.sleeps: list[float] = []
        self.timeouts: list[float] = []
        monkeypatch.setattr(
            "market_feature_store.hithink_client.load_api_key",
            lambda: "probe-key",
        )
        monkeypatch.setattr("urllib.request.urlopen", self._urlopen)
        monkeypatch.setattr(time, "sleep", self.sleeps.append)

    def _urlopen(self, req, timeout=None):
        self.calls += 1
        self.timeouts.append(timeout)
        event = self.events.pop(0)
        if isinstance(event, Exception):
            raise event
        return event


def _http429(body=b"<html><body>429 Too Many Requests</body></html>", headers=None):
    return urllib.error.HTTPError(
        "https://probe.invalid/api", 429, "Too Many", headers or {}, io.BytesIO(body)
    )

def test_retry_after_seconds_and_http_date_forms(monkeypatch):
    """Retry-After 秒数与 HTTP-date 两种合法形态都被采用（封顶 60s）。"""
    import market_feature_store.hithink_client as hc

    wire = _FakeWire(
        monkeypatch,
        [_http429(headers={"Retry-After": "2"}), _OkResp({"code": 0, "data": 1})],
    )
    hc.get_json("/api/probe", gap_seconds=0.0)
    assert wire.sleeps == [pytest.approx(2.0)]

    future = datetime.now(tz=SH).astimezone() + timedelta(seconds=5)
    wire = _FakeWire(
        monkeypatch,
        [
            _http429(headers={"Retry-After": format_datetime(future)}),
            _OkResp({"code": 0, "data": 1}),
        ],
    )
    hc.get_json("/api/probe", gap_seconds=0.0)
    assert wire.sleeps[0] == pytest.approx(5.0, abs=1.5)

    wire = _FakeWire(
        monkeypatch,
        [_http429(headers={"Retry-After": "86400"}), _OkResp({"code": 0, "data": 1})],
    )
    hc.get_json("/api/probe", gap_seconds=0.0)
    assert wire.sleeps[0] <= hc.MAX_RATE_LIMIT_SLEEP_SECONDS

def test_invalid_retry_after_falls_back_and_retry_count_bounded(monkeypatch):

    import market_feature_store.hithink_client as hc

    assert max(wire.sleeps) <= hc.MAX_RATE_LIMIT_SLEEP_SECONDS

def test_budget_is_admission_not_midread_interrupt_and_timeout_tightened(
    monkeypatch,
):
    """本轮新增：预算是发起前准入（wait 超 deadline 直接耗尽、不再睡眠），

    且准入后的在途读取只是把 timeout 收紧到剩余预算，而非强制中断。
    """
    import market_feature_store.hithink_client as hc

    # 预算 1s、Retry-After 5s：安排这次睡眠必然越过 deadline → 立即耗尽
    wire = _FakeWire(monkeypatch, [_http429(headers={"Retry-After": "5"})])
    with pytest.raises(hc.HithinkRateLimitError):
        hc.get_json("/api/probe", gap_seconds=0.0, rate_limit_budget_seconds=1.0)
    assert wire.calls == 1 and wire.sleeps == []

    # 预算 10s 时，退避后重发的请求 timeout 被收紧到 ≤ 剩余预算（首发不受限）
    wire = _FakeWire(monkeypatch, [_http429(), _OkResp({"code": 0, "data": 1})])
    hc.get_json("/api/probe", gap_seconds=0.0, rate_limit_budget_seconds=10.0)
    assert wire.timeouts[0] == 60.0
    assert wire.timeouts[1] <= 10.0 + 1e-6

def test_4001_keeps_baseline_retry_count_and_backoff(monkeypatch):

        monkeypatch,
        [

    assert wire.calls == 2  # baseline 同样是 retries=2 次尝试
    assert type(err.value) is hc.HithinkAPIError
    assert "code=4001" in str(err.value)

def _payload_for(kind, code, day):
    if kind == "anomaly":
        return {
            "code": 0,
            "data": {
                "item": [
                    {
                        "thscode": "600000.SH",
                        "analysis_content": "供应商文本",
                        "tag_name": "涨停",
                        "keyword_list": ["k"],
                    }
                ],
                "timestamp": _ms_of(day),
            },
        }
    if kind == "heat_trend":
        return {
            "code": 0,
            "data": {
                "item": [
                    {
                        "thscode": code,
                        "date": day.isoformat(),
                        "date_ms": _ms_of(day),
                        "rank": 3,
                    }
                ]
            },
        }
    return {
        "code": 0,
        "data": {
            "total": 2,
            "item": [
                {
                    "thscode": c,
                    "pe_ttm": 9.0,
                    "pe_mrq": None,
                    "pb_mrq": 1.0,
                    "ps_ttm": 2.0,
                    "pcf_ttm": None,
                }
                for c in ("600000.SH", "600519.SH")
            ],
        },
    }


def _run_sync(tmp_path, getter, **kwargs):

def _audit_rows(tmp_path):
    import duckdb

    con = duckdb.connect(str(tmp_path / "probe.duckdb"), read_only=True)
    rows = {
        r[0]: r[1:]
        for r in con.execute(
            "SELECT kind, status, error_type, payload_json, params_json "
            "FROM ops_hithink_research_request"
        ).fetchall()
    }
    con.close()
    return rows


def test_rate_limit_gap_leaves_failed_audit_and_continues(tmp_path):
    import market_feature_store.hithink_client as hc
    from market_feature_store.sync import sync_hithink_research as s

    today = date.today()
    calls: list[str] = []

    def getter(path, *, params):
        calls.append(path)
        if path == s.PATHS["valuation"]:
            raise hc.HithinkRateLimitError("budget gone")
        code = params.get("thscode") or "600000.SH"
        kind = next(k for k, v in s.PATHS.items() if v == path)
        return _payload_for(kind, code, today)

    result = _run_sync(tmp_path, getter, codes=["600000.SH", "600519.SH"])
    assert result["status"] == "partial"
    assert [m["kind"] for m in result["missing"]] == ["valuation"]
    assert result["missing"][0]["request_id"]
    # 失败之后仍尝试了其余请求（anomaly 在前，热度在后）
    heat_calls = [c for c in calls if c == s.PATHS["heat_trend"]]
    assert len(heat_calls) == 2
    assert calls.index(s.PATHS["heat_trend"]) > calls.index(s.PATHS["valuation"])

    rows = _audit_rows(tmp_path)
    assert rows["valuation"][0] == "failed"
    assert rows["valuation"][1] == "HithinkRateLimitError"
    assert rows["valuation"][2] is None  # 失败档不留上游正文
    assert "600519.SH" in rows["valuation"][3]  # 原始请求范围可溯
    assert rows["heat_trend"][0] == "ok"
    assert rows["anomaly"][0] == "ok"

def test_rate_limit_gap_then_ordinary_error_aborts_with_distinct_audit(tmp_path):
    """本轮新增（对应候选新增回归场景的独立复现）：限流耗尽可被容忍，

    但其后的普通错误必须熔断；两类失败的审计行完全分账。
    """
    import market_feature_store.hithink_client as hc
    from market_feature_store.sync import sync_hithink_research as s

    attempted: list[str] = []

    def getter(path, *, params):
        kind = next(k for k, v in s.PATHS.items() if v == path)
        attempted.append(kind)
        if kind == "anomaly":
            raise hc.HithinkRateLimitError("synthetic rate limit exhaustion")
        raise hc.HithinkAPIError("synthetic ordinary failure")

    with pytest.raises(s.HithinkCaptureError) as exc:
        _run_sync(tmp_path, getter, codes=["600000.SH"])
    assert exc.value.kind == "valuation"
    assert exc.value.rate_limited is False
    assert exc.value.request_id
    assert attempted == ["anomaly", "valuation"]  # 热度请求不再发起

    rows = _audit_rows(tmp_path)
    assert rows["anomaly"][:2] == ("failed", "HithinkRateLimitError")
    assert rows["valuation"][:2] == ("failed", "HithinkAPIError")
    assert rows["anomaly"][2] is None and rows["valuation"][2] is None  # 无上游正文

def test_write_path_guard_catches_hardlink_identity(tmp_path, monkeypatch):
    import market_feature_store.write_path as wp

    prod = tmp_path / "canonical-prod.duckdb"
    prod.write_bytes(b"probe")
    link = tmp_path / "disguised-clone.duckdb"
    os.link(prod, link)  # 同 inode 的硬链接：resolve 后路径不同
    monkeypatch.setattr(
        wp, "canonical_production_candidates", lambda: frozenset({prod})
    )
    assert wp.is_canonical_production(link) is True  # 硬链接被 samestat 识破
    other = tmp_path / "other.duckdb"
    other.write_bytes(b"x")
    assert wp.is_canonical_production(other) is False
    assert wp.is_canonical_production(tmp_path / "missing.duckdb") is False
    assert wp.production_write_blocked(False, link) is not None  # 非 direct 被拦
    assert wp.production_write_blocked(True, link) is None  # 显式 direct 放行
