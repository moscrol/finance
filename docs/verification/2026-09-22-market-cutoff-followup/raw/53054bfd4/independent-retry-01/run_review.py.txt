from __future__ import annotations

from datetime import datetime, timezone
import json
import os
from pathlib import Path
import signal

import sys
import time

    raise SystemExit("usage: run_review.py spec|quality")

def git(*args: str) -> str:
    return subprocess.check_output(
        ["git", "-C", str(TREE), *args],
        text=True,
        stderr=subprocess.STDOUT,
        env={**os.environ, "GIT_OPTIONAL_LOCKS": "0"},
    ).strip()


def identity() -> dict[str, str]:
    return {
        "revision": git("rev-parse", "HEAD"),
        "tree": git("rev-parse", "HEAD^{tree}"),
        "status": git("status", "--porcelain=v1", "--untracked-files=all"),
    }


focus = {
    "spec": (
        "重点审查规格与权限合同：是否有路径绕过明确 cutoff、把真实来源日期误报为无效、"
        "因日期不一致而拒绝已有事实，或扩大 local_only/material_only 能力。检查工具层、"
        "证据层和交付层的边界，并用源码证据或最小只读内存探针证明具体问题。"
    ),
    "quality": (
        "重点审查工程质量和回归风险：授权安全、证据/cache/trace 隔离、日期解析边界、"
        "JSON locator 与文件身份、NULL/0 和错误语义、工具 metadata、离线开关板同步及"
        "测试覆盖缺口。只报告具体可影响行为的缺陷，不报告格式偏好。"
    ),
}[ROLE]

prompt = f"""你是独立的 {ROLE} 审查者，不是实现作者。只审查精确候选 revision {SHA}（tree {TREE_SHA}），工作树为 {TREE}，基线为 {BASE}。这是新鲜审查会话；不要读取其他审查目录、作者 handoff、共享记忆或凭据来替代代码判断。

本任务要求：不同真实来源日期是 advisory，不是整体不可用/ready 失败；已有事实按真实来源日期交付，未知不猜、NULL 不补 0；比较仍须核对期间、单位和口径。明确用户历史信息截止必须由系统强制，覆盖工具读取以及 evidence、观察文本、gaps、trace、cache，不能只靠模型自觉。local_only 只能补已有授权的本地日快照，不得借综合 market_data 放开可能联网能力。material_only 必须在路径发现、预取和读取之前返回。历史日快照只能打开截止内的日文件，不能用较新文件、latest 或 meta 补历史；必须保留来源身份、质量、NULL/0 区别和原 JSON locator。封存 fixture 不得逃逸到生产路径。不要求完整自然语言解析，也不要求完整 publication-time/PIT 语义。

{focus}

1. findings 在前，按严重度列出文件/行号、触发条件、影响和最小复现；没有确证问题就明确写“本范围未发现问题”，不要为了凑数造问题。

(OUT / "prompt.txt").write_text(prompt, encoding="utf-8")

before = identity()
if before != {"revision": SHA, "tree": TREE_SHA, "status": ""}:
    raise RuntimeError(f"review identity is not clean or unexpected: {before}")
record: dict[str, object] = {
    "role": ROLE,
    "candidate_revision": SHA,
    "candidate_tree": TREE_SHA,
    "base": BASE,
    "tree_path": str(TREE),
    "before": before,
    "started_at": datetime.now(timezone.utc).isoformat(),
    "complete": False,
    "timeout_seconds": 1200,
    "automatic_retry": False,
    "fallback": False,
}
command = [
    "/Applications/ChatGPT.app/Contents/Resources/codex",
    "exec",
    "--ignore-user-config",
    "-m",
    "gpt-6-astra",
    "-s",
    "read-only",
    "--ephemeral",
    "--color",
    "never",
    "-C",
    str(TREE),
    "-o",
    str(OUT / "verdict.md"),
    prompt,
]
record["command"] = command[:-1] + ["<prompt embedded>"]
(OUT / "run.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
started = time.monotonic()
with (OUT / "log.txt").open("x", encoding="utf-8") as log:
    process = subprocess.Popen(
        command,
        cwd=TREE,
        stdout=log,
        stderr=subprocess.STDOUT,
        stdin=subprocess.DEVNULL,
        start_new_session=True,
    )
    record["pid"] = process.pid
    (OUT / "run.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    try:
        record["exit_code"] = process.wait(timeout=1200)
    except subprocess.TimeoutExpired:
        record["timeout"] = True
        os.killpg(process.pid, signal.SIGTERM)
        try:
            process.wait(timeout=5)
        except subprocess.TimeoutExpired:
            os.killpg(process.pid, signal.SIGKILL)
            process.wait()
        record["exit_code"] = 124

after = identity()
record.update(
    after=after,
    elapsed_seconds=round(time.monotonic() - started, 3),
    finished_at=datetime.now(timezone.utc).isoformat(),
    complete=True,
    identity_stable=before == after,
    verdict_file_present=(OUT / "verdict.md").is_file(),
)
record["review_status"] = (
    "PENDING_READING"
    if record["exit_code"] == 0 and record["verdict_file_present"] and record["identity_stable"]
    else "UNAVAILABLE"
)
(OUT / "run.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
print(json.dumps(record, ensure_ascii=False), flush=True)
raise SystemExit(int(record["exit_code"]))
