def main():
    files, _ = archive.sources()
    _, matches = archive.scan(files)
    for row in matches:
        row["classification"] = REVIEWED.get(row["match_sha256"], "unclassified")
    unknown = [row for row in matches if row["classification"] == "unclassified"]
    if unknown:
        print(json.dumps(unknown, indent=2))
        return 1
    path = Path(__file__).parent / "reviewed-matches.json"
    assert not path.exists(), "approval list is a one-shot record"
    path.write_text(json.dumps(matches, ensure_ascii=False, indent=2) + "\n")
    print(json.dumps({"locations": len(matches), "exact_reviewed_values": len({r["match_sha256"] for r in matches}), "unclassified": 0}))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
