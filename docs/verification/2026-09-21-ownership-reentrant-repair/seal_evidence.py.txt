"""Seal selected repair evidence without touching older archives or frozen trees.

Logs remain byte-for-byte originals (.txt is only an archive filename suffix).
Exclude Git objects, test temp trees, caches, databases and latest navigation.
Membership, original bytes and SHA256 are verified before manifest publication.
"""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import re

ROOT = Path(__file__).resolve().parent
REPO = Path.home() / "fwp-wt-ownership-closeout-0921"
DEST = REPO / "docs/verification/2026-09-21-ownership-reentrant-repair"
REV = "a092a021c3d7d978c4dfd9551e74aa984a523f18"
files = set()
for pattern in ("*.py", "*.json", "*.txt", "*.diff", "*.log", "README.md"):
    files.update(ROOT.glob(pattern))
for name in ("red-new", "green-new", "focused", "focused-final", "ruff-first", "ruff-final",
             "clean-source-full", "source-readback", "source-compatibility"):
    directory = ROOT / name
    assert directory.is_dir(), name
    files.update(p for p in directory.iterdir() if p.is_file())
    receipt_dir = directory / "receipts"
    if receipt_dir.is_dir():
        files.update(p for p in receipt_dir.rglob("*.json") if p.name != "latest.json")
for directory in (ROOT / "mutations", ROOT / "old-source-final-tests", ROOT / "exact-source-reproduction"):
    files.update(p for p in directory.rglob("*") if p.is_file() and not any(
        part in {".git", "basetemp", "__pycache__", ".pytest_cache", ".ruff_cache", "ruff-cache", "tmp", "home"}
        for part in p.relative_to(directory).parts
    ) and p.name != "latest.json")
files = sorted(p for p in files if p != ROOT / "archive-verification.json")
assert ROOT / "README.md" in files
DEST.mkdir(exist_ok=False)
entries = []
for source in files:
    relative = source.relative_to(ROOT)
    target_relative = Path(str(relative) + ".txt") if source.suffix in {".py", ".sh", ".log"} else relative
    blob = source.read_bytes()
    assert len(blob) <= 5 * 1024 * 1024, (source, len(blob))
    text = blob.decode("utf-8")
    assert not re.search(r"-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----|\b(?:sk-proj-|sk-live-)[A-Za-z0-9_-]{16,}", text), source
    assert source.suffix not in {".db", ".duckdb", ".sqlite", ".pdf", ".zip", ".pyc"}, source
    target = DEST / target_relative
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open("xb") as stream:
        stream.write(blob)
    entries.append({"path": str(target_relative), "source": str(source), "bytes": len(blob),
                    "sha256": hashlib.sha256(blob).hexdigest()})
assert {str(p.relative_to(DEST)) for p in DEST.rglob("*") if p.is_file()} == {row["path"] for row in entries}
for row in entries:
    blob = (DEST / row["path"]).read_bytes()
    assert blob == Path(row["source"]).read_bytes()
    assert len(blob) == row["bytes"] and hashlib.sha256(blob).hexdigest() == row["sha256"]
manifest = {"created_at": datetime.now(timezone.utc).isoformat(), "source_revision": REV,
            "scope": "#814 source repair only, not latest-main integration or completed independent review",
            "independent_review_status": "NOT_RERUN; previous v3 reviews remain incomplete",
            "entry_count_including_readme_excluding_manifest": len(entries),
            "selected_bytes": sum(row["bytes"] for row in entries),
            "exclusions": ["full-suite temporary fixtures", "Git objects", "caches", "DB binaries", "latest.json navigation"],
            "entries": entries}
(DEST / "manifest.json").write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + "\n")
result = {"time": datetime.now(timezone.utc).isoformat(), "archive": str(DEST),
          "files_excluding_manifest": len(entries), "bytes": manifest["selected_bytes"],
          "membership_sizes_hashes_original_bytes_ok": True,
          "manifest_sha256": hashlib.sha256((DEST / "manifest.json").read_bytes()).hexdigest()}
(ROOT / "archive-verification.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
print(json.dumps(result, ensure_ascii=False, indent=2))
