"""Probe 3 (independent, reviewer-authored): call-site forwarding capture + zero budget (C2/C5).

Patches llm_refine._open_deadline_http_response to capture (timeout, deadline,
is_cancelled) at each of the five HTTP call sites, using inspect-driven argument
binding so signatures cannot be assumed. Cases:
  P1-P5  : _post_chat, _post_chat_synthesis, _post_chat_message_stream,
           _post_chat_message, _post_chat_stream(_raw) each reach the transport
           with a non-None deadline (trigger reached is reported separately from
           downstream parsing success).
  Z1     : zero remaining budget (Deadline.from_timeout(0)) -> no HTTP request
           reaches the server and llm_refine.LLMDeadlineExceeded is raised (C5).
Also prints whether is_cancelled was forwarded on streaming sites.
"""
import inspect
import json
import sys
import threading
import time
import urllib.request
from contextlib import contextmanager
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

from intelligence.services import llm_refine as llm

PORT = 26003
RESULTS = []
CHAT = json.dumps({"choices": [{"message": {"content": "hello"}, "finish_reason": "stop"}],
                   "usage": {"prompt_tokens": 1, "completion_tokens": 1}}).encode()
SSE = b'data: {"choices":[{"delta":{"content":"a"}}]}\n\ndata: [DONE]\n\n'


class Handler(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"
    hits = []
    lock = threading.Lock()

    def log_message(self, *a):
        pass

    def do_POST(self):
        with Handler.lock:
            Handler.hits.append((self.path, time.monotonic()))
        self.send_response(200)
        self.send_header("Content-Length", str(len(CHAT)))
        self.send_header("Content-Type", "application/json")
        self.end_headers()
        self.wfile.write(CHAT)


class FakeResp:
    def __init__(self, stream):
        self.stream = stream
        self.status = 200

    def read(self):
        return CHAT

    def __iter__(self):
        if self.stream:
            for line in SSE.splitlines(keepends=True):
                yield line
        return


def record(name, ok, detail):
    RESULTS.append((name, ok, detail))
    print(f"{'PASS' if ok else 'FAIL'} {name}: {detail}")


def main():
    server = ThreadingHTTPServer(("127.0.0.1", PORT), Handler)
    threading.Thread(target=server.serve_forever, daemon=True).start()
    time.sleep(0.2)
    urllib.request.urlopen(f"http://127.0.0.1:{PORT}/v1/chat/completions",
                           data=b"{}", timeout=5).read()
    print("endpoint framing verified")

    provider = llm.LLMProvider(name="fake", api_key="k",
                               base_url=f"http://127.0.0.1:{PORT}/v1", model="m")
    captured = []

    @contextmanager
    def fake_open(req, timeout, *, deadline=None, is_cancelled=None):
        captured.append({"timeout": timeout, "deadline": deadline,
                         "is_cancelled": is_cancelled,
                         "url": req.full_url})
        yield FakeResp(stream=isinstance(is_cancelled, object) and callable(is_cancelled))

    messages = [{"role": "user", "content": "hi"}]

    def call_site(name, func, **extra):
        captured.clear()
        downstream = None
        reached = False
        try:
            sig = inspect.signature(func)
            kwargs = {}
            for pname, p in sig.parameters.items():
                if pname in extra:
                    kwargs[pname] = extra[pname]
                elif p.default is inspect.Parameter.empty and p.kind in (
                        p.POSITIONAL_OR_KEYWORD, p.KEYWORD_ONLY):
                    kwargs[pname] = {
                        "provider": provider, "messages": messages,
                        "timeout": 10, "temperature": 0.0, "on_delta": lambda s: None,
                        "on_connected": None, "is_cancelled": (lambda: False) if "is_cancelled" in sig.parameters else None,
                        "deadline": llm.Deadline.from_timeout(10),
                        "max_tokens": 128, "max_chars": 1000,
                    }.get(pname, None)
            if "deadline" in sig.parameters:
                kwargs["deadline"] = llm.Deadline.from_timeout(10)
            func(**kwargs)
        except BaseException as e:
            downstream = f"{type(e).__name__}: {e}"
        reached = len(captured) == 1
        forwarded = reached and captured[0]["deadline"] is not None
        cancel_fwd = reached and captured[0]["is_cancelled"] is not None
        record(f"P_{name}", forwarded,
               f"reached={reached} deadline_fwd={forwarded} cancel_fwd={cancel_fwd} "
               f"timeout={captured[0]['timeout'] if captured else None} downstream={downstream}")

    with llm.call_ledger_scope(max_calls=20), llm.provider_override(provider), \
            llm.http_transport_override(lambda *a, **k: None) if False else _null():
        pass

    # patch the wrapper used by all five sites
    orig = llm._open_deadline_http_response
    llm._open_deadline_http_response = fake_open
    try:
        with llm.call_ledger_scope(max_calls=20), llm.provider_override(provider):
            call_site("_post_chat", llm._post_chat)
            call_site("_post_chat_synthesis", llm._post_chat_synthesis,
                      max_tokens=128, max_chars=1000)
            call_site("_post_chat_message_stream", llm._post_chat_message_stream,
                      is_cancelled=lambda: False)
            call_site("_post_chat_message", llm._post_chat_message)
            call_site("_post_chat_stream", llm._post_chat_stream)
    finally:
        llm._open_deadline_http_response = orig

    # Z1: zero budget -> no HTTP request at all
    before = len(Handler.hits)
    exc = None
    try:
        with llm.call_ledger_scope(max_calls=5), llm.provider_override(provider):
            llm._post_chat(provider, messages, 10, deadline=llm.Deadline.from_timeout(0))
    except BaseException as e:
        exc = e
    served = len(Handler.hits) - before
    record("Z1_zero_budget_no_http",
           type(exc).__name__ == "LLMDeadlineExceeded" and served == 0,
           f"exc={type(exc).__name__} served={served}")

    server.shutdown()
    failed = [r for r in RESULTS if not r[1]]
    print(f"summary: {len(RESULTS) - len(failed)}/{len(RESULTS)} subcases passed")
    sys.exit(1 if failed else 0)


@contextmanager
def _null():
    yield


if __name__ == "__main__":
    main()
