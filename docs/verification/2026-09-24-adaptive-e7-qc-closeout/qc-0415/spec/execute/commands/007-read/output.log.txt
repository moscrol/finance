"""Probe 2: llm_refine five HTTP call paths, shared-deadline forwarding (C2/C4/C5).

Fake loopback server 127.0.0.1:26002; /stall trickles body over ~3s, /ok returns full JSON.
Positive success, 10s-slice vs 0.5s shared deadline per path, mid-stream cancellation,
zero call budget (no HTTP), expired root deadline (no HTTP).
Run: python llm_paths_probe.py
"""
import json, sys, time, threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.request import Request

sys.path.insert(0, "/Users/a77/.finance-runtime/reviews/pr868-merge-ready-20260924-0326/candidate")
from intelligence.services import llm_refine as llm

PORT = 26002
RESULTS = []
MESSAGES = [{"role": "user", "content": "offline fixture"}]


class H(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def log_message(self, *a):
        pass

    def do_POST(self):
        n = int(self.headers.get("Content-Length") or 0)
        self.rfile.read(n)
        body = b'{"choices": [{"message": {"content": "hello"}, "finish_reason": "stop"}], "usage": {}}'
        if self.path.strip("/").split("?")[0] == "stall":
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Transfer-Encoding", "chunked")
            self.end_headers()
            for i in range(0, len(body), 2):
                self.wfile.write(b"%x\r\n" % 2 + body[i:i + 2] + b"\r\n")
                self.wfile.flush()
                time.sleep(0.15)
            self.wfile.write(b"0\r\n\r\n")
            self.wfile.flush()
            return
        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)
        self.wfile.flush()


def record(name, ok, trigger, detail):
    RESULTS.append({"subcase": name, "pass": bool(ok), "trigger_reached": bool(trigger), **detail})


def main():
    srv = ThreadingHTTPServer(("127.0.0.1", PORT), H)
    threading.Thread(target=srv.serve_forever, daemon=True).start()
    time.sleep(0.2)
    provider = llm.LLMProvider(name="fake", base_url=f"http://127.0.0.1:{PORT}/stall",
                               model="fake-model", api_key="k")

    def run_path(name, fn):
        t0, exc = time.monotonic(), None
        try:
            with llm.call_ledger_scope(max_calls=10, reuse_existing=False):
                fn()
        except BaseException as e:
            exc = type(e).__name__
        wall = time.monotonic() - t0
        ok = exc == "LLMDeadlineExceeded" and wall < 2.0
        record(f"shared_deadline_{name}", ok, exc == "LLMDeadlineExceeded" and wall < 10.0,
               {"exception": exc, "wall": round(wall, 3)})

    D = lambda: llm.Deadline.from_timeout(0.5)
    okp = llm.LLMProvider(name="fake", base_url=f"http://127.0.0.1:{PORT}/ok",
                          model="fake-model", api_key="k")

    run_path("post_chat", lambda: llm._post_chat(provider, MESSAGES, 10, deadline=D()))
    run_path("post_chat_synthesis",
             lambda: llm._post_chat_synthesis(provider, MESSAGES, 10, 0.2, 64, 1000, deadline=D()))
    run_path("post_chat_message", lambda: llm._post_chat_message(provider, MESSAGES, 10, deadline=D()))
    run_path("post_chat_message_stream",
             lambda: llm._post_chat_message_stream(provider, MESSAGES, 10, 0.0, None, None, None,
                                                   lambda s: None, deadline=D()))
    run_path("post_chat_stream_raw",
             lambda: llm._post_chat_stream_raw(provider, MESSAGES, 10, 0, lambda s: None, None,
                                               None, D(), 128, 1000))

    # positive control: full success path
    t0, content, exc = time.monotonic(), None, None
    try:
        with llm.call_ledger_scope(max_calls=10, reuse_existing=False):
            content = llm._post_chat(okp, MESSAGES, 5)
    except BaseException as e:
        exc = type(e).__name__
    record("success_post_chat", exc is None and content == "hello" and time.monotonic() - t0 < 3,
           content == "hello", {"exception": exc, "content": content})

    # mid-stream cancellation on tools stream path
    state = {"timer": None, "cancelled_at": None, "flag": False}
    orig = llm._open_deadline_http_response
    from contextlib import contextmanager

    @contextmanager
    def arm(*args, **kwargs):
        with orig(*args, **kwargs) as response:
            def cancel():
                state["cancelled_at"] = time.monotonic()
                state["flag"] = True
            state["timer"] = threading.Timer(0.3, cancel)
            state["timer"].start()
            yield response

    t0, exc = time.monotonic(), None
    try:
        with llm.call_ledger_scope(max_calls=10, reuse_existing=False):
            import unittest.mock as mock
            with mock.patch.object(llm, "_open_deadline_http_response", arm):
                llm._post_chat_message_stream(provider, MESSAGES, 10, 0.0, None, None, None,
                                              lambda s: None, lambda: state["flag"],
                                              deadline=llm.Deadline.from_timeout(10))
    except BaseException as e:
        exc = type(e).__name__
    stopped = time.monotonic()
    if state["timer"]:
        state["timer"].cancel(); state["timer"].join()
    lat = stopped - state["cancelled_at"] if state["cancelled_at"] else None
    record("cancel_tools_stream",
           exc == "LLMStreamCancelled" and lat is not None and lat < 0.8,
           exc == "LLMStreamCancelled" and lat is not None,
           {"exception": exc, "cancel_to_stop": None if lat is None else round(lat, 3)})

    # C5a: zero call budget -> no HTTP, no reservation consumed
    srv.requests = 0
    t0, exc = time.monotonic(), None
    try:
        with llm.call_ledger_scope(max_calls=0, reuse_existing=False):
            llm._post_chat(okp, MESSAGES, 5)
    except BaseException as e:
        exc = type(e).__name__
    wall = time.monotonic() - t0
    record("zero_budget_no_http", exc == "LLMCallBudgetExceeded" and wall < 0.5,
           exc == "LLMCallBudgetExceeded" and wall < 0.5,
           {"exception": exc, "wall": round(wall, 3)})

    # C5b: expired root deadline -> fail before HTTP
    t0, exc = time.monotonic(), None
    try:
        with llm.call_ledger_scope(max_calls=10, reuse_existing=False):
            llm._post_chat(okp, MESSAGES, 5, deadline=llm.Deadline.from_timeout(0))
    except BaseException as e:
        exc = type(e).__name__
    wall = time.monotonic() - t0
    record("expired_root_no_http",
           exc == "LLMDeadlineExceeded" and wall < 1.0,
           exc == "LLMDeadlineExceeded" and wall < 1.0,
           {"exception": exc, "wall": round(wall, 3)})

    srv.shutdown()
    print(json.dumps(RESULTS, indent=1))
    bad = [r["subcase"] for r in RESULTS if not r["pass"]]
    print(f"SUBCASES total={len(RESULTS)} failed={len(bad)} {bad}")
    sys.exit(1 if bad else 0)


if __name__ == "__main__":
    main()
