"""Fresh host-only admission after a batch-parent metadata-only amendment."""
from concurrent.futures import ThreadPoolExecutor
import json
import os
from pathlib import Path
import shlex
import signal
import subprocess
import time
import xml.etree.ElementTree as ET

ROOT=Path('/Users/a77/.finance-runtime/reviews/pr868-glm-qc-20260924-0415')
OUT=Path(__file__).resolve().parent

def check(axis):
 folder=ROOT/axis
 config=json.loads((folder/'config.json').read_text())
 work=folder/'work'
 prefix='The exact second command (no env prefix, no copied files) is: '
 command=next(line[len(prefix):] for line in (folder/'prompt-execute.md').read_text().splitlines() if line.startswith(prefix))
 xml=work/'host-author.xml'
 args=[('--basetemp='+str(work/'host-author-tmp') if value.startswith('--basetemp=') else '--junitxml='+str(xml) if value.startswith('--junitxml=') else value) for value in shlex.split(command)]
 env={'HOME':str(work),'PATH':'/opt/homebrew/bin:/usr/bin:/bin:/usr/sbin:/sbin','LANG':'C.UTF-8','TMPDIR':str(work/'tmp'),'PYTHONDONTWRITEBYTECODE':'1','PYTHONPATH':config['tree'],'FWP_TEST_RECEIPT':'0','GIT_OPTIONAL_LOCKS':'0','FORESIGHT_USERS_DIR':str(work/'users')}
 started=time.monotonic()
 timed_out=False
 proc=subprocess.Popen(['/usr/bin/sandbox-exec','-f',str(folder/'tools.sb'),*args],cwd=config['tree'],env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,start_new_session=True)
 try:
  raw=proc.communicate(timeout=120)[0]
 except subprocess.TimeoutExpired:
  timed_out=True
  os.killpg(proc.pid,signal.SIGTERM)
  try: raw=proc.communicate(timeout=5)[0]
  except subprocess.TimeoutExpired:
   os.killpg(proc.pid,signal.SIGKILL)
   raw=proc.communicate()[0]
 with (OUT/(axis+'-author-admission-0415.log.txt')).open('xb') as log: log.write(raw)
 counts=[]
 if xml.is_file(): counts=[dict(s.attrib) for s in ET.parse(xml).getroot().iter('testsuite')]
 return {'axis':axis,'command':args,'exit_code':proc.returncode,'timed_out':timed_out,'seconds':round(time.monotonic()-started,3),'junit_suites':counts,'model_requests':0,'scope':'Host admission using original candidate tests and real sandbox with pipe capture; not reviewer execution.'}

with ThreadPoolExecutor(max_workers=2) as pool:
 rows=list(pool.map(check,('spec','quality')))
with (OUT/'author-admission-0415.json').open('x') as out: json.dump(rows,out,indent=2)
print(json.dumps(rows))
assert all(row['exit_code']==0 and not row['timed_out'] and len(row['junit_suites'])==1 and row['junit_suites'][0]['tests']=='16' and all(row['junit_suites'][0][field]=='0' for field in ('failures','errors','skipped')) for row in rows)
