assert json.loads((QC/'closure-current-turn.json').read_text())['do_not_resume']

DEST.mkdir()
manifest={}
patterns=rb'-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----|\b(?:ghp_|glpat-|sk-proj-)[A-Za-z0-9_-]{20,}|\b[a-f0-9]{32}\.[A-Za-z0-9]{16,}\b'
def add(source,relative):
 assert source.is_file() and not source.is_symlink(),source
 raw=source.read_bytes()
 assert not re.search(patterns,raw),source
 encoding='raw'
 data=raw
 if len(raw)>200000 or re.search(rb'[ \t]+\r?$',raw,re.M) or raw.endswith(b'\n\n'):
  encoding='gzip-base64-json' if len(raw)>200000 else 'base64-json'
  payload=gzip.compress(raw,mtime=0) if encoding.startswith('gzip') else raw
  data=(json.dumps({'encoding':encoding,'source_sha256':hashlib.sha256(raw).hexdigest(),'source_bytes':len(raw),'data':base64.b64encode(payload).decode('ascii')},indent=2)+'\n').encode()
  relative+='.encoded.json'
 elif source.suffix in ('.py','.mjs','.sh','.sb'):
  relative+='.txt'
 assert relative not in manifest,relative
 target=DEST/relative
 target.parent.mkdir(parents=True,exist_ok=True)
 target.write_bytes(data)
 decoded=data
 if encoding!='raw':
  decoded=base64.b64decode(json.loads(data)['data'],validate=True)
  if encoding.startswith('gzip'): decoded=gzip.decompress(decoded)
 assert decoded==raw
 manifest[relative]={'source':str(source),'encoding':encoding,'source_bytes':len(raw),'source_sha256':hashlib.sha256(raw).hexdigest(),'stored_bytes':len(data),'stored_sha256':hashlib.sha256(data).hexdigest()}
