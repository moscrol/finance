"""Host-only admission with pipe capture, matching the real tool boundary."""
from concurrent.futures import ThreadPoolExecutor
import json
from pathlib import Path
import shlex
import subprocess
import time

ROOT=Path('/Users/a77/.finance-runtime/reviews/pr868-glm-qc-20260924-0400')
OUT=Path(__file__).resolve().parent

def check(axis):
 folder=ROOT/axis
 config=json.loads((folder/'config.json').read_text())
 work=folder/'work'
 prefix='The exact second command (no env prefix, no copied files) is: '
 command=next(line[len(prefix):] for line in (folder/'prompt-execute.md').read_text().splitlines() if line.startswith(prefix))
 args=shlex.split(command)
 args=[('--basetemp='+str(work/'host-author-v2-tmp') if value.startswith('--basetemp=') else '--junitxml='+str(work/'host-author-v2.xml') if value.startswith('--junitxml=') else value) for value in args]
 env={'HOME':str(work),'PATH':'/opt/homebrew/bin:/usr/bin:/bin:/usr/sbin:/sbin','LANG':'C.UTF-8','TMPDIR':str(work/'tmp'),'PYTHONDONTWRITEBYTECODE':'1','PYTHONPATH':config['tree'],'FWP_TEST_RECEIPT':'0','GIT_OPTIONAL_LOCKS':'0','FORESIGHT_USERS_DIR':str(work/'users')}
 sandbox=['/usr/bin/sandbox-exec','-f',str(folder/'tools.sb')]
 control=sandbox+[config['python'],'-B','-c',"print('output-boundary-control')"]
 with (OUT/(axis+'-output-file-control.log.txt')).open('xb') as log:
  file_result=subprocess.run(control,cwd=config['tree'],env=env,stdout=log,stderr=subprocess.STDOUT,timeout=10)
 pipe_result=subprocess.run(control,cwd=config['tree'],env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,timeout=10)
 with (OUT/(axis+'-output-pipe-control.log.txt')).open('xb') as log: log.write(pipe_result.stdout)
 started=time.monotonic()
 result=subprocess.run(sandbox+args,cwd=config['tree'],env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,timeout=120)
 with (OUT/(axis+'-author-admission-v2.log.txt')).open('xb') as log: log.write(result.stdout)
 return {'axis':axis,'command':args,'exit_code':result.returncode,'seconds':round(time.monotonic()-started,3),'output_control':{'direct_file_exit':file_result.returncode,'pipe_exit':pipe_result.returncode,'pipe_stdout':pipe_result.stdout.decode()},'model_requests':0,'scope':'Host admission, original candidate test paths; pipe capture matches real tool. Separate basetemp/XML only; not reviewer execution.'}

with ThreadPoolExecutor(max_workers=2) as pool:
 rows=list(pool.map(check,('spec','quality')))
with (OUT/'author-admission-v2.json').open('x') as out: json.dump(rows,out,indent=2)
print(json.dumps(rows))
assert all(row['output_control']['direct_file_exit']==120 and row['output_control']['pipe_exit']==0 for row in rows)
assert all(row['exit_code']==0 for row in rows)
