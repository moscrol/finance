import fs from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { startFixture } from './fixture-upstream.mjs';
import { runReview } from './controller.mjs';

const root = path.dirname(fileURLToPath(import.meta.url));
const runRoot = path.join(root, `fixture-suite-${Date.now()}`);
await fs.mkdir(runRoot, {mode:0o700});
const checks = [];
for (const [name, axis, mode, options] of [
  ['spec', 'spec', 'normal', {finalizeAfter:1}],
  ['quality', 'quality', 'normal', {finalizeAfter:1}],
  ['no-retry', 'spec', 'error', {}],
  ['budget-loop', 'spec', 'loop', {maxRequests:2, finalizeAfter:1}],
  ['wall-stall', 'spec', 'stall', {deadlineMs:3500, finalizeMs:1500}],
]) {
  const fixture = await startFixture(mode);
  try {
    const dir = path.join(runRoot, name);
    await fs.cp(path.join(root, axis), dir, {recursive:true, errorOnExist:true});
    const config = JSON.parse(await fs.readFile(path.join(dir, 'config.json'), 'utf8'));
    Object.assign(config, {root:dir, fixture:true, deadlineMs:30000, finalizeMs:20000, maxRequests:24}, options);
    config.prompt = path.join(dir, 'prompt.md');
    config.sandbox = path.join(dir, 'tools.sb');
    const result = await runReview(config, fixture.url);
    const events = (await fs.readFile(path.join(dir,'events.jsonl'),'utf8')).trim().split('\n').map(JSON.parse);
    let ok;
    if (mode === 'normal') {
      const workers = events.filter(x => x.type === 'worker_result');
      ok = result.status === 'FINAL_RETURNED_REQUIRES_VALIDATION' && result.gateway.requestCount === 2 && fixture.seen[1].toolCount === 0 && fixture.seen[1].hasToolResults && workers.length === 4 && workers.filter(x=>x.ok).length===3 && workers.filter(x=>!x.ok).length===1;
    } else if (mode === 'error') {
      ok = result.status.startsWith('BLOCKED') && !result.reportPresent && result.gateway.requestCount===1 && fixture.seen.length===1;
    } else if (mode === 'loop') {
      ok = result.status.startsWith('BLOCKED') && !result.reportPresent && result.gateway.requestCount===2 && fixture.seen.length===2 && events.filter(x=>x.type==='worker_result').length===1;
    } else {
      ok = result.status.startsWith('BLOCKED') && !result.reportPresent && result.gateway.requestCount===1 && result.elapsedMs < 6000;
    }
    ok = Boolean(ok && result.after.status === '' && !events.some(x => ['auto_retry_start','compaction_start'].includes(x.type)));
    checks.push({name, ok, result, seen:fixture.seen});
  } finally { await fixture.close(); }
}
const summary = {status:checks.every(x=>x.ok)?'PASS':'FAIL', checks, real_model_requests:0, credentials_read:0};
await fs.writeFile(path.join(runRoot,'summary.json'), JSON.stringify(summary,null,2)+'\n',{flag:'wx',mode:0o600});
console.log(JSON.stringify({root:runRoot,status:summary.status, checks:checks.map(x=>({name:x.name,ok:x.ok,status:x.result.status,requests:x.result.gateway.requestCount}))}));
process.exitCode=summary.status==='PASS'?0:1;
