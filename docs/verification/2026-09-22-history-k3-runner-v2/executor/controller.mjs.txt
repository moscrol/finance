import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { spawn, spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import { createBoundedGateway } from './bounded-gateway.mjs';

const ROOT = path.dirname(fileURLToPath(import.meta.url));
const save = (p, v) => fs.writeFileSync(p, JSON.stringify(v, null, 2) + '\n', { flag: 'wx', mode: 0o600 });
const hash = p => crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
export const cleanEnv = home => ({ PATH: '/opt/homebrew/bin:/usr/bin:/bin', HOME: home, PI_OFFLINE: '1', PI_SKIP_VERSION_CHECK: '1', PYTHONDONTWRITEBYTECODE: '1', GIT_OPTIONAL_LOCKS: '0', GIT_CONFIG_NOSYSTEM: '1', GIT_CONFIG_GLOBAL: '/dev/null', LC_ALL: 'C.UTF-8' });
function identity(candidate) {
  const run = args => {
    const p = spawnSync('/usr/bin/git', ['-c', 'core.fsmonitor=false', ...args], { cwd: candidate, env: cleanEnv('/var/empty'), encoding: 'utf8', timeout: 15000 });
    if (p.status !== 0) throw new Error('Candidate identity command failed');
    return p.stdout.trim();
  };
  return { revision: run(['rev-parse', 'HEAD']), status: run(['status', '--porcelain', '--untracked-files=all']) };
}
export function credential(config, execute = spawnSync) {
  if (config.fixture) return 'offline-fake-access-token';
  const resolver = '/Users/a77/.claude/mirasim-kimi-token.sh';
  const expected = '602a59daf33af6b27c3c7504881001c6aafc4d92d96f40580069c11641a83e36';
  if (hash(resolver) !== expected) throw new Error('Resolver changed: stop before credential access');
  const script = fs.readFileSync(resolver, 'utf8');
  const body = script.split("python3 - <<'PY'\n")[1]?.split('\nPY')[0];
  if (!body) throw new Error('Resolver Python body missing');
  const p = execute(config.python, ['-I', '-B', '-c', body], { env: cleanEnv('/Users/a77'), encoding: 'utf8', timeout: 35000, maxBuffer: 65536 });
  save(path.join(config.root, 'credential.json'), { resolverSha256: expected, method: 'direct_python_same_body', exitCode: p.status, timedOut: p.error?.code === 'ETIMEDOUT', tokenPersisted: false, stderrSha256: crypto.createHash('sha256').update(p.stderr || '').digest('hex') });
  if (p.status !== 0 || !p.stdout.trim()) throw new Error('Credential bootstrap failed; raw secrets not logged');
  return p.stdout.trim();
}
export async function runReview(config, fixtureUpstream) {
  process.umask(0o077);
  const started = Date.now();
  const deadline = started + config.deadlineMs;
  const root = config.root;
  save(path.join(root, 'STARTED.json'), { started, fixture: config.fixture });
  const events = [];
  const log = event => { events.push(event); fs.appendFileSync(path.join(root, 'gateway-events.jsonl'), JSON.stringify(event) + '\n', { mode: 0o600 }); };
  let gateway, child, before, after, exitCode = null, reason = 'BLOCKED_BOOTSTRAP', timer, sampler;
  const disks = [];
  const sample = () => {
    const s = fs.statfsSync(root);
    const availableBytes = s.bavail * s.bsize;
    disks.push({ timestamp: Date.now(), availableBytes });
    return availableBytes >= 6 * 1024 ** 3;
  };
  const kill = () => { if (child?.pid) { try { process.kill(-child.pid, 'SIGKILL'); } catch (e) { if (e.code !== 'ESRCH') throw e; } } };
  const onSignal = () => { reason = 'BLOCKED_INTERRUPTED'; kill(); };
  process.once('SIGTERM', onSignal); process.once('SIGINT', onSignal);
  try {
    before = identity(config.candidate);
    if (before.revision !== config.revision || before.status) throw new Error('Candidate identity/clean guard failed');
    if (!sample()) throw new Error('Below 6GiB available disk stop line');
    const token = credential(config);
    const gatewayKey = crypto.randomBytes(32).toString('hex');
    gateway = await createBoundedGateway({
      upstream: config.fixture ? fixtureUpstream : 'http://127.0.0.1:18788', token, gatewayKey,
      maxRequests: config.maxRequests, deadlineMs: deadline - Date.now(), finalizeAfter: config.finalizeAfter,
      finalizeMs: Math.max(1, config.finalizeMs - (Date.now() - started)),
      onFinalize: () => save(path.join(root, 'finalizing.json'), { at: Date.now(), reason: 'request_or_time_reserve' }), log,
    });
    const stdout = fs.openSync(path.join(root, 'session.stdout.log'), 'wx', 0o600);
    const stderr = fs.openSync(path.join(root, 'session.stderr.log'), 'wx', 0o600);
    child = spawn('/opt/homebrew/bin/node', [path.join(ROOT, 'session.mjs')], { cwd: config.candidate, detached: true, env: cleanEnv(path.join(root, 'empty-home')), stdio: ['pipe', stdout, stderr] });
    fs.closeSync(stdout); fs.closeSync(stderr);
    child.stdin.on('error', () => {});
    child.stdin.end(JSON.stringify({ ...config, gateway: gateway.baseUrl, gatewayKey }));
    reason = 'BLOCKED_NO_FINAL';
    timer = setTimeout(() => { reason = 'BLOCKED_WALL_DEADLINE'; kill(); }, Math.max(1, deadline - Date.now()));
    sampler = setInterval(() => { if (!sample()) { reason = 'BLOCKED_DISK_STOP'; kill(); } }, 10000);
    exitCode = await new Promise((resolve, reject) => { child.once('error', reject); child.once('close', resolve); });
    if (exitCode === 0) reason = 'FINAL_RETURNED_REQUIRES_VALIDATION';
  } catch (e) {
    log({ type: 'controller_failure', error: String(e) });
  } finally {
    clearTimeout(timer); clearInterval(sampler);
    kill();
    if (gateway) await gateway.close();
    process.removeListener('SIGTERM', onSignal); process.removeListener('SIGINT', onSignal);
    try { after = identity(config.candidate); } catch { after = { error: 'identity_failed' }; }
    if (!before || after.revision !== before.revision || after.status !== '') reason = 'BLOCKED_IDENTITY';
    sample();
    const reportPath = path.join(root, `${config.axis.toUpperCase()}.md`);
    const result = { status: reason, exitCode, started, elapsedMs: Date.now() - started, before, after,
      gateway: gateway?.snapshot() || { requestCount: 0 }, fixture: config.fixture,
      tests_run: 0, retries: 0, fallback: false, reportPresent: fs.existsSync(reportPath),
      reportSha256: fs.existsSync(reportPath) ? hash(reportPath) : null, disk: disks,
      sandbox_layers_per_tool: 1, controller_trusted_unsandboxed: true, production_changes: false,
    };
    save(path.join(root, 'execution.json'), result);
    return result;
  }
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const axis = process.argv[2];
  if (!['spec', 'quality'].includes(axis) || process.argv[3] !== '--live-authorized') throw new Error('Explicit axis/live authorization required');
  const offline = JSON.parse(fs.readFileSync(path.join(ROOT, 'offline-acceptance.json')));
  if (offline.status !== 'PASS') throw new Error('Offline acceptance missing');
  const inputs = JSON.parse(fs.readFileSync(path.join(ROOT, 'frozen-inputs.json')));
  for (const [p, sha] of Object.entries(inputs)) if (hash(path.join(ROOT, p)) !== sha) throw new Error(`Frozen input changed: ${p}`);
  if (offline.frozenInputsSha256 !== hash(path.join(ROOT, 'frozen-inputs.json'))) throw new Error('Offline acceptance/input mismatch');
  const config = JSON.parse(fs.readFileSync(path.join(ROOT, axis, 'config.json')));
  if (config.fixture || config.maxRequests !== 24 || config.deadlineMs !== 600000) throw new Error('Live budget changed');
  const result = await runReview(config);
  console.log(JSON.stringify(result));
  process.exitCode = result.status === 'FINAL_RETURNED_REQUIRES_VALIDATION' ? 0 : 1;
}
