"""Seal selected per-candidate checks, without copying disposable pytest fixtures."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import shutil
import subprocess

ROOT = Path(__file__).resolve().parent
for pr in ('910', '911'):
    source = ROOT / pr
    summary = json.loads((source / 'summary.json').read_text())
    tree = Path(summary['tree'])
    assert summary['status'] == 'SCOPED_CHECKS_COMPLETE_NOT_RELEASE_READY'
    assert subprocess.check_output(['git', '-C', str(tree), 'rev-parse', 'HEAD'], text=True).strip() == summary['revision']
    assert subprocess.check_output(['git', '-C', str(tree), 'status', '--porcelain'], text=True).strip() == ''
    dest = tree / 'docs/verification' / f'2026-09-25-pr{pr}-main-forward'
    dest.mkdir(exist_ok=False)
    paths = sorted(p for p in source.iterdir() if p.is_file())
    paths += [ROOT / 'run_offline.py', ROOT / 'archive_offline.py', ROOT / 'retarget-receipt.json', ROOT / 'started.json']
    records = []
    for path in paths:
        name = path.name + '.txt' if path.suffix in {'.py', '.log'} else path.name
        target = dest / 'raw' / name
        target.parent.mkdir(exist_ok=True)
        shutil.copyfile(path, target)
        sha = hashlib.sha256(path.read_bytes()).hexdigest()
        assert hashlib.sha256(target.read_bytes()).hexdigest() == sha
        records.append({'path': target.relative_to(dest).as_posix(), 'source': str(path),
                        'sha256': sha, 'bytes': path.stat().st_size})
    manifest = {'schema_version': 1, 'pr': int(pr), 'candidate': summary['revision'], 'baseline': summary['baseline'],
                'scope': 'static gates and explicitly selected offline targets only',
                'full_scope_executed': False, 'independent_review_executed': False, 'model_requests': 0,
                'created_at': datetime.now(timezone.utc).isoformat(), 'files': records}
    (dest / 'manifest.json').write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + '\n')
    print(json.dumps({'pr': pr, 'archived_originals': len(records), 'archive': str(dest)}))
