// Subscription-only K3 auth with auditable dispatch, deliberately no usage cap.
// Native provider auth refreshes a short-lived credential in process memory;
// credential lifetime is not used as an artificial review deadline.
import childProcess from "node:child_process";
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { createHash } from "node:crypto";
import { createProvider } from "/opt/homebrew/lib/node_modules/@earendil-works/pi-coding-agent/node_modules/@earendil-works/pi-ai/dist/index.js";
import { openAICompletionsApi } from "/opt/homebrew/lib/node_modules/@earendil-works/pi-coding-agent/node_modules/@earendil-works/pi-ai/dist/api/openai-completions.lazy.js";

const ROOT = path.dirname(fileURLToPath(import.meta.url));
const RESOLVER = "/Users/a77/.claude/mirasim-kimi-token.sh";
const PROVIDER = "mirasim-kimi";
const MODEL = "kimi-k3";
const URL = "http://127.0.0.1:18788/v1";
const MIN_VALIDITY_SECONDS = 300;

export function bootstrap(pi) {
  const out = fs.realpathSync(process.env.REVIEW_OUTPUT_DIR || "/invalid-output");
  if (!out.startsWith(ROOT + path.sep)) throw new Error("output is outside review root");
  const recordPath = path.join(out, "credential.json");
  const record = { status: "RESERVED", started_at: new Date().toISOString(),
    provider: PROVIDER, model: MODEL, cached_only_in_process_memory: true,
    prices_verified: false, max_requests: null, review_deadline_seconds: null };
  fs.writeFileSync(recordPath, JSON.stringify(record, null, 2) + "\n", { flag: "wx", mode: 0o600 });
  const config = JSON.parse(fs.readFileSync(path.join(ROOT, "provider-template.json"), "utf8"));
  if (config.baseUrl !== URL || config.api !== "openai-completions" || config.models.length !== 1 ||
      config.models[0].id !== MODEL || config.models[0].baseUrl || config.apiKey || config.headers) {
    throw new Error("provider route mismatch");
  }
  let cached;
  let blocked = false;
  let generation = 0;
  let admitted = 0;
  const recordEvent = (entry) => fs.appendFileSync(path.join(out, "credential-events.jsonl"),
    JSON.stringify({ timestamp: new Date().toISOString(), ...entry }) + "\n", { mode: 0o600 });
  function resolveToken(signal) {
    signal?.throwIfAborted();
    if (blocked) throw new Error("credential resolution previously blocked; no retry or fallback");
    if (cached && Math.min(cached.exp, cached.plan_exp) > Date.now() / 1000 + MIN_VALIDITY_SECONDS) return cached.token;
    try {
      // Reservation precedes each initial/expiry refresh; not a model retry.
      recordEvent({ state: "REFRESH_RESERVED", generation: ++generation });
      const token = childProcess.execFileSync(RESOLVER, [], {
        encoding: "utf8", timeout: 35000, maxBuffer: 65536, stdio: ["ignore", "pipe", "pipe"],
        env: { HOME: "/Users/a77", PATH: "/opt/homebrew/bin:/usr/bin:/bin", LANG: "C.UTF-8" },
      }).trim();
      signal?.throwIfAborted();
      const parts = token.split(".");
      if (parts.length !== 3) throw new Error("invalid credential format");
      const claims = JSON.parse(Buffer.from(parts[1], "base64url").toString("utf8"));
      const until = Date.now() / 1000 + MIN_VALIDITY_SECONDS;
      if (claims.plan !== "plus" || !Number.isFinite(claims.exp) || claims.exp <= until ||
          !Number.isFinite(claims.plan_exp) || claims.plan_exp <= until) {
        throw new Error("existing Plus credential unavailable or expires too soon");
      }
      Object.assign(record, { status: "READY", generation, plan: claims.plan, expires_at: claims.exp,
        plan_expires_at: claims.plan_exp, resolver_sha256: createHash("sha256").update(fs.readFileSync(RESOLVER)).digest("hex") });
      recordEvent({ state: "READY", generation, expires_at: claims.exp, plan_expires_at: claims.plan_exp });
      fs.writeFileSync(recordPath, JSON.stringify(record, null, 2) + "\n");
      cached = { token, exp: claims.exp, plan_exp: claims.plan_exp };
      return token;
    } catch (error) {
      blocked = true;
      cached = undefined;
      Object.assign(record, { status: "BLOCKED", generation, error_type: error?.name || "Error",
        resolver_exit: Number.isInteger(error?.status) ? error.status : null,
        resolver_signal: error?.signal || null });
      // Never serialize resolver stdout/stderr or the exception message.
      fs.writeFileSync(recordPath, JSON.stringify(record, null, 2) + "\n");
      recordEvent({ state: "BLOCKED", generation });
      throw new Error("credential bootstrap blocked; details intentionally omit credentials");
    }
  }
  resolveToken();
  const native = createProvider({
    id: PROVIDER, name: "Existing Plus K3 review", baseUrl: URL,
    auth: { apiKey: {
      name: "Existing Plus credential (memory only)",
      async check() { return blocked ? undefined : { source: "existing Plus", type: "api_key" }; },
      async resolve({ signal }) { return { auth: { apiKey: resolveToken(signal) }, source: "existing Plus" }; },
    } },
    models: config.models.map((model) => ({ ...model, provider: PROVIDER, api: config.api, baseUrl: URL })),
    api: openAICompletionsApi(),
  });
  pi.registerProvider(native);
  pi.on("before_provider_request", (event, ctx) => {
    try {
      if (ctx.model?.provider !== PROVIDER || ctx.model?.id !== MODEL ||
          ctx.model?.baseUrl !== URL || ctx.model?.api !== config.api || event.payload?.model !== MODEL || blocked) {
        throw new Error("model request route denied");
      }
      fs.appendFileSync(path.join(out, "request-admissions.jsonl"), JSON.stringify({
        admission: admitted + 1, timestamp: new Date().toISOString(), provider: PROVIDER, model: MODEL,
        credential_generation: generation,
      }) + "\n", { mode: 0o600 });
      admitted += 1;
    } catch {
      process.stderr.write("Review route/audit blocked; no retry or fallback.\n");
      process.exit(75);
    }
  });
}

export default function (pi) {
  try { bootstrap(pi); }
  catch {
    process.stderr.write("Review credential bootstrap failed before model dispatch.\n");
    process.exit(70);
  }
}
