"""Audit a finished K3 v4 axis without treating report existence as signoff.

This finite operator recorder refuses running sessions and preserves every tool
result and write incarnation from the original event bytes. It checks the actual
shell wrapper records, not only tool timeout arguments. Semantic/source-defect
assessment is deliberately separate. No provider call or source mutation.
"""
from collections import Counter, defaultdict, deque
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import re
import subprocess
import sys

ROOT = Path(__file__).resolve().parent
REV = "6eb12c1b8a41071fd4af8bee343950fe0b85b221"
GIT_TREE = "e99dad14cb946a112d92537289854d914e558936"


def sha(data):
    return hashlib.sha256(data).hexdigest()


def put(path, value):
    with path.open("x") as stream:
        stream.write(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def git(tree, *args):
    return subprocess.run(["git", "-C", str(tree), *args], capture_output=True,
                          text=True, check=True, timeout=30).stdout.rstrip("\n")


def audit(axis):
    assert axis in {"spec", "quality"}
    out = ROOT / f"{axis}-k3"
    execution = json.loads((out / "execution.json").read_text())
    assert execution["complete"], "review still running; do not snapshot as final"
    assert execution["revision"] == REV
    destination = ROOT / "operator-qc" / f"{axis}-mechanics"
    assert not destination.exists(), "refuse overwrite of an operator audit"
    before, after = execution["identity_before"], execution["identity_after"]
    identity_checks = {}
    for key in ("revision", "git_tree", "status", "candidate_files", "author", "inputs_sha256"):
        identity_checks[f"start_end_{key}_equal"] = before[key] == after[key]
    identity_checks["expected_revision"] = after["revision"] == REV
    identity_checks["expected_git_tree"] = after["git_tree"] == GIT_TREE
    identity_checks["clean_start_end"] = before["status"] == after["status"] == ""
    identity_checks["unlimited_authorization"] = all(execution[key] is None for key in (
        "timeout_seconds", "max_provider_request_admissions", "forced_exploration_cutoff"))
    input_mismatches = [name for name, expected in after["inputs_sha256"].items()
                        if sha((ROOT / name).read_bytes()) != expected]
    output_mismatches = [name for name, expected in execution["sha256"].items()
                         if sha((out / name).read_bytes()) != expected]
    tree = Path(after["checkout"])
    current = {"revision": git(tree, "rev-parse", "HEAD"),
               "git_tree": git(tree, "rev-parse", "HEAD^{tree}"),
               "status": git(tree, "status", "--porcelain=v1", "--untracked-files=all"),
               "candidate_files": {p: sha((tree / p).read_bytes()) for p in after["candidate_files"]}}
    identity_checks["current_same_identity"] = all(current[key] == after[key] for key in current)
    destination.mkdir(parents=True)
    recovered = destination / "trace"
    recovered.mkdir()
    trace, by_id, event_counts, anomalies = [], {}, Counter(), []
    assistant_count, model_ids, model_errors, compactions = 0, set(), [], []
    writes = defaultdict(list)
    usage = dict.fromkeys(("input", "output", "cacheRead", "cacheWrite", "totalTokens"), 0)
    with (out / "events.jsonl").open("rb") as stream:
        for number, line in enumerate(stream, 1):
            if not line.strip():
                continue
            try:
                event = json.loads(line)
            except (ValueError, UnicodeError):
                anomalies.append({"invalid_event_line": number, "line_sha256": sha(line)})
                continue
            kind = event.get("type", "unknown")
            event_counts[kind] += 1
            if "compact" in kind or "retry" in kind:
                compactions.append(event)
            if kind == "message_end" and event.get("message", {}).get("role") == "assistant":
                message = event["message"]
                assistant_count += 1
                model_ids.add(str(message.get("provider")) + "/" + str(message.get("model")))
                if message.get("stopReason") in {"error", "aborted"}:
                    model_errors.append(message.get("errorMessage", message.get("stopReason")))
                for key in usage:
                    usage[key] += message.get("usage", {}).get(key, 0) or 0
            if kind == "tool_execution_start":
                tool, args = event["toolName"], event.get("args", {})
                item = {"index": len(trace) + 1, "id": event["toolCallId"], "tool": tool,
                        "event_line": number, "args": {k: v for k, v in args.items() if k != "content"}}
                if item["id"] in by_id:
                    anomalies.append({"duplicate_tool_id": item["id"]})
                if tool == "write":
                    target = Path(args["path"])
                    normalized = target if target.is_absolute() else tree / target
                    item["write_within_output_lexically"] = normalized.is_relative_to(out)
                    data = args.get("content", "").encode()
                    saved = recovered / f"{item['index']:04d}-write-{target.name}.txt"
                    saved.write_bytes(data)
                    item.update(content_sha256=sha(data), recovered_write=str(saved.relative_to(ROOT)))
                    writes[str(target)].append(item["index"])
                trace.append(item)
                by_id[item["id"]] = item
            elif kind == "tool_execution_end":
                item = by_id.get(event.get("toolCallId"))
                if not item:
                    anomalies.append({"unmatched_tool_end": event.get("toolCallId")})
                    continue
                if "end_event_line" in item:
                    anomalies.append({"duplicate_tool_end": item["id"]})
                item["end_event_line"] = number
                # Original event line is retained byte-for-byte, not reserialized.
                saved = recovered / f"{item['index']:04d}-result-event.jsonl"
                saved.write_bytes(line)
                text = "\n".join(c.get("text", "") for c in event.get("result", {}).get("content", []) if isinstance(c, dict))
                (recovered / f"{item['index']:04d}-result.txt").write_text(text)
                item.update(result_event=str(saved.relative_to(ROOT)), result_event_sha256=sha(line),
                            result_summary=text[:320], is_error=event.get("isError", False))
    for item in trace:
        if "end_event_line" not in item:
            anomalies.append({"incomplete_tool": item["index"]})
    shell_by_command = defaultdict(deque)
    for item in trace:
        if item["tool"] == "bash":
            shell_by_command[item["args"]["command"]].append(item)
    shells = []
    for path in sorted((out / "shell-records").glob("*.json")):
        row = json.loads(path.read_text())
        log = path.with_name(path.stem + ".combined.log")
        item = shell_by_command[row["command"]].popleft() if shell_by_command[row["command"]] else None
        check = {"record": str(path.relative_to(ROOT)), "log": str(log.relative_to(ROOT)),
                 "complete": row.get("complete"), "exit_code": row.get("exit_code"), "error": row.get("error"),
                 "effective_timeout": row.get("timeout"), "timeout_valid": isinstance(row.get("timeout"), (int, float)) and 0 < row["timeout"] <= 120,
                 "log_hash_matches": log.is_file() and sha(log.read_bytes()) == row.get("log_sha256"),
                 "trace_index": item["index"] if item else None}
        if item:
            item["shell_record"] = check["record"]
        else:
            anomalies.append({"unmatched_shell_record": str(path)})
        shells.append(check)
    unmatched_shell_calls = [item["index"] for queue in shell_by_command.values() for item in queue]
    # Suspicious strings are review leads, not an assertion that a violation occurred.
    leads = []
    patterns = {
        "outside_tmp": r"(?:^|[\s\"'=])/tmp/",
        "credential_or_global": r"(?:Keychain|security find|mirasim-kimi-token|/\.ssh/|/\.pi/|/agent-memory/|/\.claude/)",
        "network_or_install": r"(?:\bcurl\b|\bwget\b|\bfetch\b|https?://|\bpip install\b|\bpnpm install\b)",
        "git_mutation_or_deletion": r"(?:\bgit\b[^\n]*(?:commit|reset|checkout|switch|merge|rebase|push)|\brm\s)",
    }
    for item in trace:
        if item["tool"] == "bash":
            hits = [name for name, pattern in patterns.items() if re.search(pattern, item["args"]["command"])]
            if hits:
                leads.append({"trace_index": item["index"], "matches": hits})
    admissions = [json.loads(line) for line in (out / "request-admissions.jsonl").read_text().splitlines()]
    report = out / "REPORT.md"
    verdict_path = out / "verdict.json"
    verdict = json.loads(verdict_path.read_text()) if verdict_path.is_file() else {}
    completion_path = out / "completion.json"
    completion = json.loads(completion_path.read_text()) if completion_path.is_file() else {}
    result = {"axis": axis, "revision": REV, "time": datetime.now(timezone.utc).isoformat(),
              "semantic_qc": "PENDING", "exit_code": execution["exit_code"], "stop_reason": execution["stop_reason"],
              "elapsed_seconds": execution["elapsed_seconds"], "identity_checks": identity_checks,
              "input_hash_mismatches": input_mismatches, "output_hash_mismatches": output_mismatches,
              "current_identity": current, "event_counts": dict(event_counts),
              "model_identities": sorted(model_ids), "model_errors": model_errors, "assistant_messages": assistant_count,
              "usage_message_end_only_not_invoice": usage, "compaction_and_retry_events": compactions,
              "admissions": len(admissions), "admission_sequence_valid": [r["admission"] for r in admissions] == list(range(1, len(admissions) + 1)),
              "admission_routes_valid": all(r.get("provider") == "mirasim-kimi" and r.get("model") == "kimi-k3" for r in admissions),
              "tool_calls": len(trace), "tool_failures": [t["index"] for t in trace if t.get("is_error")],
              "trace_anomalies": anomalies, "shell_records": shells, "shell_calls_without_execution_record": unmatched_shell_calls,
              "shell_command_leads_for_manual_review": leads,
              "repeated_write_targets": {p: ids for p, ids in writes.items() if len(ids) > 1},
              "report_present": report.is_file(), "verdict_present": verdict_path.is_file(),
              "final_report_identity_matches": verdict.get("revision") == REV and verdict.get("axis") == axis and verdict.get("complete") is True,
              "completion_identity_matches": completion.get("revision") == REV and completion.get("axis") == axis,
              "model_verdict": verdict.get("verdict"), "reported_check_count_not_scenarios": len(verdict.get("checks", [])),
              "limitations": ["no OS sandbox", "direct-path and shell command review does not prove all filesystem accesses", "clean start/end does not rule out transient modifications", "shell combined bytes do not identify every nested process exit"],
              "sha256": {p.name: sha(p.read_bytes()) for p in (out / "execution.json", out / "events.jsonl", report, verdict_path, completion_path) if p.is_file()}}
    put(destination / "tool-trace.json", trace)
    put(destination / "mechanical-audit.json", result)
    print(json.dumps({k: result[k] for k in ("axis", "exit_code", "admissions", "tool_calls", "model_verdict", "final_report_identity_matches", "identity_checks", "trace_anomalies")}, indent=2))


if __name__ == "__main__":
    audit(sys.argv[1])
