"""One material and one DuckDB question through the actual conversation HTTP door.

No retries of a failed question. Each first run is retained; checks prove wiring,
not quality improvement or a performance trend. Credentials never enter receipts.
"""
from __future__ import annotations
import argparse
import collections
import json
import time
import urllib.error
import urllib.request
from pathlib import Path

from intelligence.eval.variance_baseline import extract_from_run_dir, score_distribution

MATERIAL = "甲公司2026年上半年营业收入180亿元，同比增长20%；归母净利润9亿元，同比增长15%。公司拟投资50亿元建设先进封装产线，预计2027年投产。"
QUESTIONS = {
    "material": "只依据以下材料回答，不要查询外部数据。\n\n「" + MATERIAL + "」\n\n1. 归母净利率是多少？\n2. 拟投资金额相当于上半年营业收入的多少倍？\n3. 材料没有给出哪些判断产能扩张合理性所需的信息？",
    "grounded": "请复盘长电科技（600584）最近一个交易日的表现：涨跌幅、成交额、所属板块的表现，并明确说明数据日期。",
}

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--port', type=int, required=True)
    ap.add_argument('--user', required=True)
    ap.add_argument('--users-dir', type=Path, required=True)
    ap.add_argument('--output', type=Path, required=True)
    ap.add_argument('--revision', required=True)
    ap.add_argument('--known-data-lag-baseline', type=Path)
    args=ap.parse_args()
    args.output.mkdir(parents=True,exist_ok=False)
    base=f'http://127.0.0.1:{args.port}'
    def http(method, path, body=None):
        data=json.dumps(body,ensure_ascii=False).encode() if body is not None else None
        req=urllib.request.Request(base+path,data=data,method=method,headers={'Content-Type':'application/json'} if data else {})
        with urllib.request.urlopen(req,timeout=60) as r: return json.load(r)
    until=time.monotonic()+420
    while True:
        try:
            ready=http('GET','/api/readiness')
        except urllib.error.HTTPError as exc:
            if exc.code != 503:
                raise
            ready=json.loads(exc.read())
        except (OSError,ValueError):
            ready={}
        checks=ready.get('checks') or {}
        if checks and all(v is True for v in checks.values()):
            break
        if args.known_data_lag_baseline and checks:
            baseline=json.loads(args.known_data_lag_baseline.read_text())
            failed=[k for k,v in checks.items() if v is not True]
            if failed == ['market_data_consistency'] and ready.get('market_database') == baseline.get('market_database'):
                assert baseline['checks']['market_data_consistency'] is False
                print('readiness remains RED: same pre-existing market data lag as production; proceeding only with scoped model wiring probe',flush=True)
                break
        if time.monotonic()>until: raise RuntimeError('canary readiness timeout')
        time.sleep(5)
    health=http('GET','/api/health')
    (args.output/'pre-health.json').write_text(json.dumps(health,indent=2)+'\n')
    (args.output/'readiness.json').write_text(json.dumps(ready,indent=2)+'\n')
    runtime=health['runtime']
    assert runtime['source_revision']==args.revision and runtime['source_dirty'] is False and runtime['code_matches_repo'] is True
    assert runtime['agent_runtime']['model']=='kimi-k3'
    results=[]
    for name, question in QUESTIONS.items():
        t=time.monotonic()
        conv=http('POST','/api/conversations',{'title':f'K3 / 无判官验收 {name}', 'user':args.user})
        cid=conv['conversation_id']
        run=http('POST',f'/api/conversations/{cid}/messages',{'content':question,'skill_mode':'auto','perspective_mode':'neutral','user':args.user})
        rid=run['run_id']
        print(name,cid,rid,flush=True)
        (args.output/f'{name}-submitted.json').write_text(json.dumps({'conversation_id':cid,'run_id':rid,'question':question},ensure_ascii=False,indent=2)+'\n')
        until=time.monotonic()+930
        status={}
        while time.monotonic()<until:
            try: status=http('GET',f'/api/runs/{rid}?user={args.user}')
            except urllib.error.HTTPError as e:
                if e.code!=404: raise
            if status.get('status') in ('completed','failed','cancelled'): break
            time.sleep(5)
        rd=args.users_dir/args.user/'runs'/rid
        until=time.monotonic()+90
        while not (rd/'report.json').exists() and time.monotonic()<until: time.sleep(2)
        def read(n):
            return json.loads((rd/n).read_text()) if (rd/n).is_file() else {}
        ep, report, rawrun=read('continuous-episode.json'), read('report.json'), read('run.json')
        answer=(rd/'answer.md').read_text() if (rd/'answer.md').is_file() else ''
        outcome=ep.get('outcome',{})
        semantic=ep.get('semantic_verifier',{})
        models=[e.get('payload',{}).get('served_model') for e in outcome.get('events',[]) if e.get('kind')=='model_turn']
        calibers=collections.Counter()
        dates=collections.Counter()
        def walk(x):
            if isinstance(x,dict):
                if isinstance(x.get('caliber'),str): calibers[x['caliber']]+=1
                for key in ('data_date','as_of_date','trade_date','snapshot_date'):
                    if isinstance(x.get(key),str): dates[x[key]]+=1
                for v in x.values(): walk(v)
            elif isinstance(x,list):
                for v in x: walk(v)
        walk(ep)
        repeat=extract_from_run_dir(rd)
        score=score_distribution({'questions':[{'question_id':name,'repeats':[repeat]}]})
        judge_calls=ep.get('metrics',{}).get('judge_usage',{}).get('calls')
        receipt={
            'question':question,'run_id':rid,'conversation_id':cid,'run_dir':str(rd),'elapsed_s':round(time.monotonic()-t,2),
            'run_status':status.get('status'),'outcome_status':outcome.get('status'),'stop_reason':outcome.get('stop_reason'),
            'model_turn_served_models':models,'report_llm':report.get('llm'), 'judge_mode':semantic.get('judge_mode'),
            'judge_calls':judge_calls,'material_review_calls':len(semantic.get('material_review_calls') or []),
            'readiness_status':ready.get('status'),'readiness_checks':ready.get('checks'),
            'gate_receipt':report.get('gate_receipt'),'independent_n':score['independent_n'],'no_judge_rate':score['no_judge_rate'],
            'calibers':dict(calibers),'data_dates':dict(dates),'degrades':rawrun.get('degrades'),'answer':answer,
        }
        checks={
            'run_completed':status.get('status')=='completed',
            'episode_completed':outcome.get('status')=='completed',
            'actual_k3':bool(models) and all(m=='kimi-k3' for m in models),
            'deterministic':semantic.get('judge_mode')=='deterministic',
            'zero_judge_calls':judge_calls==0,
            'no_material_model_review':not semantic.get('material_review_calls'),
            'public_not_independent':report.get('gate_receipt',{}).get('correlated_judge','missing') is None,
            'eval_no_independent':score['independent_n']==0 and score['no_judge_rate']==1.0,
            'nonempty_answer':bool(answer.strip()),
        }
        if name=='material':
            checks.update({'q1_5_percent':'5%' in answer or '5.0%' in answer or '5.00%' in answer,'q2_ratio':any(v in answer for v in ('0.28','0.278','0.2778','27.78%','27.8%'))})
        else:
            checks['stock_fact_used']='fact_stock_daily' in calibers
            checks['no_no_data_disclaimer']='本轮没有连接本地市场数据' not in answer
        receipt['checks']=checks
        receipt['ok']=all(checks.values())
        (args.output/f'{name}-receipt.json').write_text(json.dumps(receipt,ensure_ascii=False,indent=2)+'\n')
        results.append({'case':name,'run_id':rid,'checks':checks,'ok':receipt['ok']})
        print(json.dumps(results[-1],ensure_ascii=False),flush=True)
    (args.output/'result.json').write_text(json.dumps(results,ensure_ascii=False,indent=2)+'\n')
    return 0 if all(r['ok'] for r in results) else 1

if __name__=='__main__': raise SystemExit(main())
