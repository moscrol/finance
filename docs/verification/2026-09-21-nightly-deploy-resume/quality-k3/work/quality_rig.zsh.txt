state="$LAUNCHCTL_STATE_DIR"; mkdir -p "$state"
printf '%s\n' "$*" >> "$LAUNCHCTL_CALL_LOG"
cmd="$1"
case "$cmd" in
  bootout)
    label="${2##*/}"
    if [ -n "${LAUNCHCTL_FAIL_BOOTOUT:-}" ] && [ "${LAUNCHCTL_FAIL_BOOTOUT:-}" = "$label" ]; then echo "Boot-out failed: 5" >&2; exit 5; fi
    rm -f "$state/$label"; exit 0 ;;
  bootstrap)
    plist="$3"
    label=$(/usr/libexec/PlistBuddy -c 'Print :Label' "$plist" 2>/dev/null) || label=$(basename "$plist" .plist)
    if [ -n "${LAUNCHCTL_FAIL_BOOTSTRAP:-}" ] && [ "${LAUNCHCTL_FAIL_BOOTSTRAP:-}" = "$label" ]; then echo "Bootstrap failed: 5: I/O error" >&2; exit 5; fi
    if [ -f "$state/$label" ]; then echo "Bootstrap failed: 17: service already loaded" >&2; exit 17; fi
    printf '%s\n' "$plist" > "$state/$label"; exit 0 ;;
  kickstart) exit 0 ;;
  *) exit 0 ;;
esac
EOF
chmod +x "$BIN/launchctl"
