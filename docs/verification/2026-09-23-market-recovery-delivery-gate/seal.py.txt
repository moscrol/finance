"""Seal independent deliveries without rewriting reviewer artifacts or old failures."""
from collections import Counter
import json
from pathlib import Path
import shutil

import run

DEST = Path('/Users/a77/fwp-wt-market-recovery-qc-fix-0923/docs/verification/2026-09-23-market-recovery-delivery-gate')
DEST.mkdir(exist_ok=False)
selected = []
stages = ['F2-explore', 'F2-execute', 'F2R-explore', 'F2R-execute', 'F2R-report',
          'F1-explore', 'F1-execute', 'F1-report', 'F3-explore', 'F3-execute', 'F3-report',
          'author-scoped-gate', 'supplementary-checks']
summary = {'status': 'HOLD', 'revision': run.REV, 'base': run.BASE,
           'tree': run.git('rev-parse', 'HEAD^{tree}'), 'raw_root': str(run.ROOT),
           'final_identity': run.identity(), 'stage_records': {}, 'model_requests': 0,
           'http_status_counts': {}, 'claims': {}, 'final_independent_probes': 0,
           'final_author_tests': 0, 'initial_probe_failures': 7,
           'initial_author_collection_errors': 1, 'production_write': False,
           'merge_push_deployment': False, 'full_repo_receipt': False,
           'execution_owner': 'deterministic controller; probe and report artifacts authored by isolated K3 sessions',
           'qualifications': [
               'F2 original fixture failures and initial author collection error are retained, not product defects.',
               'F2R snapshots compare logical row values, not physical database bytes; no other-date seed exists, so cross-date preservation is not independently established.',
               'F3 unchanged assertions compare only stock_ts_code and close, not full rows or bytes. The missing-policy case does not assert the other-day row. Broader reviewer wording is not adopted.',
               'Deliberate positive controls validate red-test detection, not mutation sensitivity of product guards.',
               'Reviewer verdicts are claim-scoped, not production, full-repo, arbitrary concurrency, or business-contract acceptance.',
               'run-v2.py is archived and hash-sealed at closeout; original stage input watch covered run.py but not this wrapper.',
           ]}
statuses = Counter()
provider_errors = Counter()
for stage in stages:
    folder = run.ROOT / stage
    record = json.loads((folder / 'execution.json').read_text())
    if stage.endswith(('explore', 'report', 'execute')):
        assert record['complete'], stage
        assert record['before'] == record['after'], stage
    status = record.get('status', record.get('exit_code', 'executed'))
    for error in record.get('model_errors', []):
        if str(error).startswith('504 '):
            provider_errors['504'] += 1
            status = 'BLOCKED_PROVIDER_504'
    summary['stage_records'][stage] = status
    for path in sorted(folder.rglob('*')):
        if not path.is_file() or any(part in {'pi-config', 'tmp', 'users'} for part in path.relative_to(folder).parts):
            continue
        if path.name in {'events.stdout', 'packet.md', 'credential.json'}:
            continue
        selected.append(path)
    admissions = folder / 'request-admissions.jsonl'
    if admissions.exists():
        rows = [json.loads(line) for line in admissions.read_text().splitlines()]
        assert [r['admission'] for r in rows] == list(range(1, len(rows) + 1))
        assert len(rows) <= 4
        assert all(r['model'] == 'kimi-k3' and r['stream'] and not r['has_temperature'] and r['tools'] == 1 for r in rows)
        summary['model_requests'] += len(rows)
    http = folder / 'http-responses.jsonl'
    if http.exists():
        statuses.update(str(json.loads(line)['status']) for line in http.read_text().splitlines())
summary['http_status_counts'] = dict(statuses)
summary['model_reported_provider_errors'] = dict(provider_errors)
for claim, name in [('F2', 'F2R'), ('F1', 'F1'), ('F3', 'F3')]:
    execute = json.loads((run.ROOT / f'{name}-execute/execution.json').read_text())
    report_path = run.ROOT / f'{name}-report/delivery/report.json'
    report = json.loads(report_path.read_text()) if report_path.exists() else None
    if report:
        assert report['revision'] == run.REV
    summary['claims'][claim] = {'verdict': report['verdict'] if report else summary['stage_records'][f'{name}-report'],
                               'probe': execute['probe']['counts'], 'author': execute['author']['counts'],
                               'positive_control_detected': execute['positive_control_detected'],
                               'controller_adoption': 'narrowed_to_actual_probe_assertions' if report else 'no_formal_report'}
    summary['final_independent_probes'] += execute['probe']['counts']['passed']
    summary['final_author_tests'] += execute['author']['counts']['passed']
for path in sorted(run.ROOT.iterdir()):
    if path.is_file() and (path.suffix in {'.py', '.mjs', '.json', '.log'}):
        selected.append(path)
for path in selected:
    relative = path.relative_to(run.ROOT)
    target = DEST / relative
    # Frozen code is evidence, not a new pytest target or lint-modified author fixture.
    if path.suffix in {'.py', '.mjs'}:
        target = target.with_name(target.name + '.txt')
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(path, target)
raw_manifest = {str(p.relative_to(run.ROOT)): run.sha(p) for p in sorted(run.ROOT.rglob('*'))
                if p.is_file() and 'candidate' not in p.relative_to(run.ROOT).parts
                and 'pi-config' not in p.relative_to(run.ROOT).parts
                and '__pycache__' not in p.relative_to(run.ROOT).parts
                and 'tmp' not in p.relative_to(run.ROOT).parts
                and 'users' not in p.relative_to(run.ROOT).parts}
run.save(DEST / 'raw-manifest.json', raw_manifest)
run.save(DEST / 'summary.json', summary)
manifest = {str(p.relative_to(DEST)): run.sha(p) for p in sorted(DEST.rglob('*')) if p.is_file()}
run.save(DEST / 'manifest.json', manifest)
print(json.dumps(summary, indent=2))
