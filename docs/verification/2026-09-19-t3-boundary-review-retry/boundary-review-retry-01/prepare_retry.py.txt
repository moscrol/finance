No business/request/claim/check-manifest changes. Only the operational deadline
is predeclared as 1200 seconds for this one call; no subsequent retries.
"""

import hashlib
import json
from pathlib import Path
import shutil
import subprocess

BASE = Path('/Users/a77/.finance-runtime/convergence-20260919/retention-repair')
REPO = Path('/Users/a77/fwp-q-research-data-readiness')
PRIOR = BASE / 'qc-repair-d46c2c3b'
STATE = BASE / 'qc-retry-d46c2c3b-01'

EXPECTED_REQUEST = '5fa3e7b7215bc83243e5f4a3e429cbca7770c791579cdce5d27d7a9cd13169d6'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def replace_once(text, old, new):
    assert text.count(old) == 1, old
    return text.replace(old, new, 1)


assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=REPO, text=True).strip() == TIP
assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=REPO, text=True).strip()
assert not subprocess.check_output(['git', 'diff', REV + '..' + TIP, '--', 'intelligence', 'market_feature_store'], cwd=REPO)
assert sha(PRIOR / 'requests/ARL-0005.json') == EXPECTED_REQUEST
# Verify all old review originals against the committed archive, before copying.
archive = REPO / 'docs/verification/2026-09-19-t3-ratio-boundaries'
manifest = json.loads((archive / 'manifest.json').read_text())
for row in manifest['files']:
    stored = archive / row['archive']
    assert len(stored.read_bytes()) == row['bytes'] and sha(stored) == row['sha256']
    source = Path(row['source'])
    assert source.read_bytes() == stored.read_bytes(), source
STATE.mkdir(exist_ok=False)
rows = []


def copy(source, relative):
    dest = STATE / relative
    dest.parent.mkdir(exist_ok=True, parents=True)
    assert not dest.exists()
    shutil.copyfile(source, dest)
    assert source.read_bytes() == dest.read_bytes()
    rows.append({'source': str(source), 'copy': relative, 'bytes': dest.stat().st_size, 'sha256': sha(dest)})


for folder in ('requests', 'claims', 'verdicts'):
    (STATE / folder).mkdir()
    for source in sorted((PRIOR / folder).glob('ARL-*.json')):
        copy(source, folder + '/' + source.name)

        copy(source, 'prior-invalid/' + source.name)

    copy(PRIOR / 'runs/ARL-0005' / name, 'prior-timeout/' + name)
copy(PRIOR / 'dispatch-result.json', 'prior-timeout/dispatch-result.json')

copy(PRIOR / 'cli-auth-settings.json', 'cli-auth-settings.json')
(STATE / 'inherited-chain.json').write_text(json.dumps(rows, ensure_ascii=False, indent=2) + '\n')

    'max_cli_calls_this_window': 1, 'deadline_seconds': 1200, 'max_budget_usd': 6,

}

source = PRIOR / 'qc_repair_once.py'

    'One user-authorized bounded continuation for the SAME immutable ARL-0005 request and d46 revision, in a NEW exclusive root.
