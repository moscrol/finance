"""Archive the stopped round with tracked suffixes and the existing Git checker format."""
import hashlib
import json
from pathlib import Path
import shutil

ROOT = Path(__file__).resolve().parent
TREE = Path('/Users/a77/fwp-wt-runtime-postmerge-qc-0923')
DEST = TREE / 'docs/verification/2026-09-24-pr884-closeout-05'
paths = sorted(path for path in ROOT.iterdir() if path.is_file()
               and path.suffix in {'.py', '.json', '.md', '.txt', '.jsonl'})
assert all('credential' not in path.name and 'auth' not in path.name for path in paths)
assert json.loads((ROOT / 'verification.json').read_text())['accepted'] is False
DEST.mkdir(exist_ok=False)
items = []
for source in paths:
    name = source.name + '.txt' if source.suffix == '.py' else source.name
    target = DEST / name
    raw = source.read_bytes()
    shutil.copyfile(source, target)
    assert target.read_bytes() == raw
    items.append({'source': source.name, 'archived_path': name, 'bytes': len(raw),
                  'sha256': hashlib.sha256(raw).hexdigest()})
with (DEST / 'SOURCE-MAP.json').open('x') as output:
    json.dump({'scope': 'selected round-05 records; no credentials, user data or dependency trees',
               'candidate_revision': '5344d8b0f87a9dcf14b08d87a19f63c7c3a25a1a',
               'files': len(items), 'bytes': sum(item['bytes'] for item in items), 'artifacts': items}, output, indent=2)
    output.write('\n')
files = sorted(path for path in DEST.iterdir() if path.is_file())
with (DEST / 'sha256-manifest.txt').open('x') as manifest:
    for path in files:
        manifest.write(hashlib.sha256(path.read_bytes()).hexdigest() + '  ' + path.relative_to(TREE).as_posix() + '\n')
print(json.dumps({'source_files': len(items), 'manifest_members': len(files),
                  'source_bytes': sum(item['bytes'] for item in items), 'archive': str(DEST)}, indent=2))
