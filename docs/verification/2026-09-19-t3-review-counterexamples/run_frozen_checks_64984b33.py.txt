import json
import os

import re

ROOT = Path('/Users/a77/fwp-q-research-data-readiness')
REV = '64984b33e6f38eccc50c004b7b79120f1f45a775'
OUT = Path(__file__).parent / ('candidate-' + REV[:8])

PY = Path('/Users/a77/finance-workspace-private/.venv-workbench/bin/python')
ENV = {k: os.environ[k] for k in ('HOME', 'PATH', 'LANG', 'TMPDIR') if k in os.environ}
ENV['FORESIGHT_LLM_KEYCHAIN'] = '0'

def git(root, *args):
    return subprocess.check_output(['git', *args], cwd=root, text=True).strip()


def state(root=ROOT):
    return {'revision': git(root, 'rev-parse', 'HEAD'), 'status': git(root, 'status', '--porcelain')}


assert state() == {'revision': REV, 'status': ''}
result = {'before': state(), 'checks': [], 'live_model_calls': 0, 'finished': False}


def save():
    (OUT / 'result.json').write_text(json.dumps(result, ensure_ascii=False, indent=2))


def check(name, args, *, cwd=ROOT, timeout=1200, extra=None):
    started = time.monotonic()
    with (OUT / (name + '.log')).open('w') as log:
        try:
            rc = subprocess.run([str(a) for a in args], cwd=cwd, env=ENV | (extra or {}), stdout=log,
                                stderr=subprocess.STDOUT, timeout=timeout).returncode
        except subprocess.TimeoutExpired:
            rc = 124
    row = {'name': name, 'command': [str(a) for a in args], 'cwd': str(cwd), 'exit_code': rc,
           'seconds': round(time.monotonic() - started, 2), 'revision': REV,
           'finished_at': datetime.now(timezone.utc).isoformat()}
    result['checks'].append(row)
    save()
    print(json.dumps(row, ensure_ascii=False), flush=True)
    return row

layout = OUT / 'registry-pinned'

    'knowledge-base-private': (Path('/Users/a77/.finance-runtime/kb-v4-merge-20260919/closeout/knowledge-base-private'), '1254224be89e2c4974350b7f3e985dbedb5dc043'),
