#!/Users/a77/finance-workspace-private/.venv-workbench/bin/python
"""One-shot independent T3 repair review; uses the existing review worker protocol."""
from __future__ import annotations

import ast
from datetime import datetime, timezone
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import subprocess
import sys

PY = '/Users/a77/finance-workspace-private/.venv-workbench/bin/python'
STATE = Path(os.environ['STATE_ROOT'])
ROOT = Path(os.environ['REVIEW_WORKTREE'])
REQUEST = Path(os.environ['REQUEST_FILE'])
CLAIM = Path(os.environ['CLAIM_FILE'])
RUN = STATE / 'runs' / os.environ['REVIEW_ID']
request = json.loads(REQUEST.read_text())
claim = json.loads(CLAIM.read_text())
revision = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip()
assert revision == request['commit'] == os.environ['COMMIT']
assert hashlib.sha256(REQUEST.read_bytes()).hexdigest() == claim['request_sha256']
os.umask(0o022)
ENV = {key: os.environ[key] for key in ('HOME', 'PATH', 'LANG', 'TMPDIR') if key in os.environ}
ENV['FORESIGHT_LLM_KEYCHAIN'] = '0'
checks = {}


def run(name, command, *, cwd=ROOT):
    result = subprocess.run(command, cwd=cwd, env=ENV, capture_output=True, text=True, timeout=600)
    raw = result.stdout + result.stderr
    (RUN / (name + '.log')).write_text(raw)
    checks[name] = {'exit_code': result.returncode, 'command': command, 'cwd': str(cwd),
                    'output_tail': raw[-5000:]}


tests = sorted({test for paths in request['artifact_tests'].values() for test in paths})
run('artifact-tests', [PY, '-m', 'pytest', '-q', *tests])
run('git-diff-check', ['git', 'diff', '--check', request['parent_commit'] + '..' + revision])
clean_root = Path(os.environ['CLEAN_REVIEW_TARGET'])
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=clean_root, text=True).strip() == revision
assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=clean_root, text=True).strip()
run('original-fact-retention-probe', [PY, 'scripts/review_probes/check_delivery_fact_retention.py',
    '--code-root', str(clean_root), '--expect-revision', revision], cwd=clean_root)
run('research-delivery-mutations', [PY, 'scripts/review_probes/run_extraction_mutations.py',
    '--suite', 'research-delivery', '--revision', revision, '--output', str(RUN / 'mutations')])
changed = subprocess.check_output(['git', 'diff', '--name-only', request['parent_commit'] + '..' + revision], cwd=ROOT, text=True).splitlines()
sys.path.insert(0, str(ROOT))
from scripts.agent_review.contract import forbidden_artifact_paths
preflight = {'revision': revision, 'request_sha256': claim['request_sha256'],
             'tests': tests, 'checks': checks, 'changed_paths': changed,
             'forbidden_paths': forbidden_artifact_paths(changed),
             'created_at': datetime.now(timezone.utc).isoformat()}
(RUN / 'preflight.json').write_text(json.dumps(preflight, ensure_ascii=False, indent=2))


def excerpt(path, names=None):
    text = (ROOT / path).read_text()
    if names is None:
        return text
    lines = text.splitlines()
    pieces = []
    for node in ast.walk(ast.parse(text)):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name in names:
            pieces.append('\n'.join(f'{i+1}: {lines[i]}' for i in range(node.lineno-1, node.end_lineno)))
    return '\n\n'.join(pieces)


diff = subprocess.check_output(['git', 'diff', '--unified=5', request['parent_commit'] + '..' + revision], cwd=ROOT, text=True)
context = {}
for path, names in (
    ('intelligence/services/research_delivery_checks.py', None),
    ('intelligence/services/episode_semantic_verifier.py', {'_recheck_research_delivery', 'recheck_material_public_delivery'}),
    ('intelligence/tests/test_research_delivery_retention.py', None),
    ('intelligence/tests/test_research_delivery_repair.py', None),
):
    context[path] = excerpt(path, names)

spec = importlib.util.spec_from_file_location('existing_schema', '/Users/a77/.finance-runtime/agent-review-loop/claude-oneshot-reviewer.py')
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
schema = module.verdict_schema(request, claim)
prompt = f'''You are claude:independent-reviewer, an external code falsification reviewer, NOT the producer.
Review exact commit {revision}, parent {request['parent_commit']}. You have no tools and cannot edit anything.
An isolated worker verified request identity and independently ran the mechanical checks below.
The request is a finite research-delivery repair, not a review-loop implementation or a release to production.
Inspect the COMPLETE changed diff and relevant unchanged call path context, not just test results.

Problem: rejecting a false 'empty search -> no company disclosures' inference used to delete an independently sourced fact and citation if connected with a comma. This patch separates sentence-wide detection context from the rejected assertion spans; ratio prose similarly marks only bad numbers/units as unverified.
Try to falsify: independent fact/citation retention with different punctuation and conjunctions; false inferences still rejected; honest qualifiers/conditional hypotheses remain; raw Markdown coordinate mapping; multi-findings merging; same-turn repair unchanged/timeout/budget exhausted/final projection; no tests or gates weakened merely to pass. Describe concrete input/path for any issue and whether introduced or pre-existing. Do not assume finite Chinese regex is a general grammar parser. A semantic counterexample may block even if the tests pass.

Scope honesty: this is static code review plus independently executed scripted checks, NOT real model-answer acceptance. The judge=llm tests use a deterministic success stub. No model-backed finance conversation is in these receipts. Neither a PASS nor these receipts grants deployment or broader T3 natural-quality acceptance.
Return one schema-2 verdict using the supplied schema. Include every exact string in request.required_checks and every unique artifact test path plus git_diff_check, artifact_hygiene, canonical_safety. The first two required check names with scripts refer to independently executed probes; other semantic check names must be judged from code, not claimed to be commands. PASS only if all required checks are justified and no high severity issue exists. Ordinary defects => CHANGES_REQUIRED; do not use BLOCKED for ordinary defects. Reviewed time can use preflight.created_at (timezone-aware). Do not invent test counts. Findings can include lower-severity limitations without hiding them.

REQUEST:\n{json.dumps(request, ensure_ascii=False, indent=2)}
CLAIM:\n{json.dumps(claim, ensure_ascii=False, indent=2)}
INDEPENDENT PREFLIGHT:\n{json.dumps(preflight, ensure_ascii=False, indent=2)}
FULL DIFF:\n{diff}
SOURCE CONTEXT:\n{json.dumps(context, ensure_ascii=False, indent=2)}
'''
(RUN / 'review-prompt.txt').write_text(prompt)
command = [str(Path.home() / '.local/bin/claude'), '-p', prompt, '--model', 'sonnet',
           '--max-budget-usd', '6', '--output-format', 'json', '--json-schema',
           json.dumps(schema), '--permission-mode', 'dontAsk', '--tools', '',
           '--setting-sources', 'user', '--strict-mcp-config', '--mcp-config', '{"mcpServers":{}}',
           '--no-session-persistence']
result = subprocess.run(command, cwd=ROOT, env=ENV, capture_output=True, text=True, timeout=600)
(RUN / 'reviewer-envelope.json').write_text(result.stdout)
(RUN / 'reviewer-stderr.log').write_text(result.stderr)
if result.returncode:
    print(json.dumps({'status': 'REVIEWER_INACTIVE', 'cli_exit': result.returncode}))
    raise SystemExit(result.returncode)
envelope = json.loads(result.stdout)
verdict = envelope.get('structured_output')
if not isinstance(verdict, dict):
    raw = envelope.get('result')
    try:
        verdict = json.loads(raw) if isinstance(raw, str) else None
    except ValueError:
        verdict = None
if not isinstance(verdict, dict):
    print(json.dumps({'status': 'NO_STRUCTURED_VERDICT', 'subtype': envelope.get('subtype')}))
    raise SystemExit(4)
Path(os.environ['VERDICT_FILE']).write_text(json.dumps(verdict, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({'status': verdict.get('status'), 'cost_usd': envelope.get('total_cost_usd'),
                  'model_usage': envelope.get('modelUsage')}, ensure_ascii=False))
