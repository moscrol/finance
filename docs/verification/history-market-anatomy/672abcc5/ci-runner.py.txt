"""Frozen four-leaf engineering run; not real-model/business acceptance."""
from concurrent.futures import ThreadPoolExecutor
import hashlib
import json
import os
from pathlib import Path
import subprocess

os.umask(0o022)
root = Path(__file__).parent
repo = Path.home() / "fwp-wt-history-market-anatomy"
python = Path.home() / "finance-workspace-private/.venv-workbench/bin/python"
revision = "672abcc504b3466a2ddd24dfa450c55703265c70"
env = {k: os.environ[k] for k in ("HOME", "TMPDIR") if k in os.environ}
env.update(PATH="/opt/homebrew/bin:/usr/bin:/bin:/usr/sbin:/sbin", WORKBENCH_PYTHON=str(python),
           WORKBENCH_E2E_PORT="18891", RE06_E2E_PORT="18894", RE06_E2E_URL="http://127.0.0.1:18894",
           FORESIGHT_EPISODE_STORE=str(root / "e2e-episodes"), FORESIGHT_LLM_KEYCHAIN="0")

def git(*args):
    return subprocess.check_output(["git", *args], cwd=repo, text=True).strip()

assert git("rev-parse", "HEAD") == revision
assert not git("status", "--porcelain")
base_before = git("rev-parse", "gitea/main")
commands = {
    "python": [[str(python), "-m", "ruff", "check", "."], [str(python), "-m", "pytest", "-q"]],
    "frontend": [["pnpm", action] for action in ("lint", "typecheck", "test", "build")],
    "e2e": [["pnpm", "test:e2e"]],
    "registry": [[str(python), "scripts/build_registry.py", *args] for args in [
        ("check-parseability",), ("check",), ("backfill-tables", "--check"), ("generate-views", "--check")
    ]] + [[str(python), "scripts/audit_ledger_spec_crosswalk.py"]],
}

def run(leaf):
    results = []
    with (root / f"{leaf}.txt").open("x") as log:
        for cmd in commands[leaf]:
            log.write("$ " + " ".join(cmd) + "\n"); log.flush()
            result = subprocess.run(cmd, cwd=repo / "intelligence/webapp" if leaf in ("frontend", "e2e") else repo,
                                    env=env, stdout=log, stderr=subprocess.STDOUT)
            results.append({"command": cmd, "exit_code": result.returncode})
    receipt = {"commands": results, "exit_code": int(any(r["exit_code"] for r in results)),
               "sha256": hashlib.sha256((root / f"{leaf}.txt").read_bytes()).hexdigest()}
    with (root / f"{leaf}-receipt.json").open("x") as f:
        json.dump(receipt, f, indent=2)
    print(leaf, receipt["exit_code"], flush=True)
    return leaf, receipt

with ThreadPoolExecutor(max_workers=4) as pool:
    leaves = dict(pool.map(run, commands))
report = {"tested_revision": revision, "scope": "author-side equivalent CI, not independent QC or live-model acceptance",
          "leaves": leaves, "revision_after": git("rev-parse", "HEAD"),
          "dirty_after": git("status", "--porcelain").splitlines(),
          "gitea_main_before": base_before, "gitea_main_after": git("rev-parse", "gitea/main")}
with (root / "receipt.json").open("x") as f:
    json.dump(report, f, indent=2)
raise SystemExit(int(any(r["exit_code"] for r in leaves.values()) or report["dirty_after"] or report["revision_after"] != revision))
