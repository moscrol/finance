"""Offline consumer snapshot for the unchanged four questions; not a model retest."""
from dataclasses import replace
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
from tempfile import TemporaryDirectory

ROOT = Path(__file__).parent
REPO = Path.home() / "fwp-wt-history-market-anatomy"
REVISION = "672abcc504b3466a2ddd24dfa450c55703265c70"
assert subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=REPO, text=True).strip() == REVISION
assert not subprocess.check_output(["git", "status", "--porcelain"], cwd=REPO, text=True).strip()
sys.path.insert(0, str(REPO))
os.environ["PYTEST_CURRENT_TEST"] = "offline-history-expression-snapshot"
os.environ["ENTITY_ANCHOR_SECURITIES_DB"] = "0"
os.environ["FORESIGHT_STRICT_DERIVATION"] = "1"
from intelligence.services.conversation_store import ConversationStore
from intelligence.services.episode_verifier import verify_episode_outcome
from intelligence.services.research_harness import FinanceResearchHarness
from intelligence.services.research_tool_registry import ResearchToolRegistry
from intelligence.services import track_contract as track, ranking_contract as ranking
from tests.test_history_expression_contract_boundary import _context, _unfulfilled, _with_track_contract_gaps
from tests.test_history_live_seams import FIRST, FOLLOWUPS, _decide

questions_path = REPO / "docs/verification/history-market-anatomy/fbd8f2a6/questions.json"
assert json.loads(questions_path.read_text())["questions"] == [FIRST, *FOLLOWUPS]
turns = []
headings = ("【跟踪表达契约】", "【排序与情景表达契约】", "【情景树表达契约】")
with TemporaryDirectory(prefix="history-expression-snapshot-") as temp:
    os.environ["FORESIGHT_USERS_DIR"] = str(Path(temp) / "users")
    os.environ["FORESIGHT_EPISODE_STORE"] = str(Path(temp) / "episodes")
    os.environ["FINANCE_WS"] = str(Path(temp) / "finance")
    store = ConversationStore("history-snapshot", root=Path(temp) / "conversations")
    conversation = store.create_conversation()
    previous = None
    for n, question in enumerate((FIRST, *FOLLOWUPS), 1):
        decision = _decide(store, conversation, question, previous, n - 1)
        frame = decision.task_frame
        context = _context(frame)
        system, user = FinanceResearchHarness().assemble_prompt(frame, context, ResearchToolRegistry(()))
        request, _ = json.JSONDecoder().raw_decode(user)
        outcome = _unfulfilled(frame, context)
        structural = verify_episode_outcome(context.contract, outcome)
        actual = _with_track_contract_gaps(structural, context)
        without_scope = _with_track_contract_gaps(structural, replace(context, history_intent=None))
        assert actual == structural and actual.missing_outputs
        assert not any(h in system + user for h in headings)
        kw = {"query": question, "question_type": frame.question_type, "history_intent": frame.history_intent}
        turns.append({
            "turn": n, "question": question,
            "task_frame_hash": frame.task_frame_hash,
            "history_intent": frame.history_intent.to_dict(),
            "cutoff": context.information_cutoff.as_of_date.isoformat(),
            "data_scope": frame.material_contract.data_scope,
            "question_type_rules": request["question_type_rules"],
            "user_prompt_sha256": hashlib.sha256(user.encode()).hexdigest(),
            "system_prompt_sha256": hashlib.sha256(system.encode()).hexdigest(),
            "forward_contract_headings": [],
            "structural_missing_outputs": list(structural.missing_outputs),
            "after_expression_check": list(actual.missing_outputs),
            "counterfactual_without_history_intent": list(without_scope.missing_outputs),
            "track_contract": track.contract_receipt(outcome.draft, **kw),
            "ranking_contract": ranking.ranking_receipt(outcome.draft, **kw),
        })
        previous = decision.turn_intent
report = {
    "schema": "history-expression-offline-consumers-v1", "revision": REVISION,
    "questions_sha256": hashlib.sha256(questions_path.read_bytes()).hexdigest(),
    "scope": "real controller/user-chain and Harness assembly; empty registry and scripted unfulfilled draft, no model or market DB",
    "not_proven": ["same-window rankings", "startup feature/control groups", "live answer quality", "autonomous saved-artifact reads"],
    "turns": turns,
}
with (ROOT / "candidate-contracts.json").open("x") as f:
    json.dump(report, f, ensure_ascii=False, indent=2)
print("Four unchanged questions: real offline prompt/check consumers exclude forward slots and retain evidence gaps.")
