"""Read frozen live trace; extract actual conflicting instructions, not new model evidence."""
import hashlib
import json
from pathlib import Path
import subprocess

root = Path(__file__).parent
repo = Path.home() / "fwp-wt-history-market-anatomy"
source = Path.home() / ".finance-runtime/history-anatomy-live-20260918-fbd8f2a6"
revision = "f9afb4a0029224ed13dd3b3bb16d0a76e7c0f2e8"
assert subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=repo, text=True).strip() == revision
assert not subprocess.check_output(["git", "status", "--porcelain"], cwd=repo, text=True).strip()
records = []
for artifact in sorted((source / "users/history-anatomy-acceptance-v2/runs").glob("*/continuous-episode.json")):
    run_id = artifact.parent.name
    events_path, = (source / "episodes").glob(run_id + "_*/events.jsonl")
    raw_events, raw_artifact = events_path.read_bytes(), artifact.read_bytes()
    events = [json.loads(line) for line in raw_events.splitlines()]
    prompt_line, prompt = next((n, event["payload"]) for n, event in enumerate(events, 1) if event.get("kind") == "prompt_assembled")
    # Episode input is a leading JSON object followed by domain context blocks.
    request, _ = json.JSONDecoder().raw_decode(prompt["user"])
    for kind in ("user", "system"):
        assert hashlib.sha256(prompt[kind].encode()).hexdigest() == prompt[kind + "_sha256"]
    saved = json.loads(raw_artifact)
    rules = request["question_type_rules"]
    record = {
        "run_id": run_id,
        "events": {"path": str(events_path), "bytes": len(raw_events), "sha256": hashlib.sha256(raw_events).hexdigest(), "prompt_line": prompt_line},
        "artifact": {"path": str(artifact), "bytes": len(raw_artifact), "sha256": hashlib.sha256(raw_artifact).hexdigest()},
        "user_prompt_sha256": prompt["user_sha256"], "system_prompt_sha256": prompt["system_sha256"],
        "question": request["task_frame"]["raw_question"],
        "history_intent": request["task_frame"]["history_intent"],
        "forward_contract_headings": [heading for heading in ("【跟踪表达契约】", "【排序与情景表达契约】", "【情景树表达契约】") if heading in rules],
        "rules_sha256": hashlib.sha256(rules.encode()).hexdigest(),
        "track_contract": saved["track_contract"], "ranking_intent": saved["ranking_contract"]["ranking_intent"],
        "ranking_missing_outputs": saved["ranking_contract"]["missing_outputs"],
        "structural_missing_outputs": saved["structural_verifier"]["missing_outputs"],
    }
    if record["forward_contract_headings"]:
        record["question_type_rules"] = rules
    assert events_path.read_bytes() == raw_events and artifact.read_bytes() == raw_artifact
    records.append(record)
assert len(records) == 4
assert records[1]["forward_contract_headings"] == ["【跟踪表达契约】", "【排序与情景表达契约】"]
assert "track_next_watch" in records[1]["structural_missing_outputs"]
assert "ranking_flip_conditions" in records[1]["structural_missing_outputs"]
report = {
    "schema": "history-forward-template-conflict-v1", "inspected_at_revision": revision,
    "scope": "read-only extraction of old fbd8 prompts and receipts; no model, DB or new conversation",
    "finding": "The actual turn-2 prompt required forward track and company priority templates, and structural checks pursued their missing sections.",
    "limits": "Proves a conflicting instruction/check path, not a unique causal explanation of the entire failed answer or the four-turn group; does not resolve mismatched rank windows or startup comparison.",
    "source_unchanged": True, "turns": records,
}
with (root / "saved-template-conflict.json").open("x") as f:
    json.dump(report, f, ensure_ascii=False, indent=2)
print("Four saved prompt identities verified; turn-2 forward instruction/check conflict extracted; sources unchanged.")
