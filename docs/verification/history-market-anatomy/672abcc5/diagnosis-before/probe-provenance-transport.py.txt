"""Diagnostic only: retain collection/order up to the failed test; log mocked IO identity.

In-memory instrumentation of its transport fixture, not a source edit or acceptance rerun.
No request bodies, headers, credentials or raw query strings are recorded.
"""
import hashlib
import inspect
import json
import os
from pathlib import Path
import subprocess
import sys
import textwrap
import threading
import traceback
from urllib.parse import urlsplit

import pytest

os.umask(0o022)
ROOT = Path(__file__).parent
REPO = Path.home() / "fwp-wt-history-market-anatomy"
REVISION = "f9afb4a0029224ed13dd3b3bb16d0a76e7c0f2e8"
TARGET = "intelligence/tests/test_ask_call_provenance.py::test_receipt_covers_calls_and_reuses_outer_budget"
assert subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=REPO, text=True).strip() == REVISION
assert not subprocess.check_output(["git", "status", "--porcelain"], cwd=REPO, text=True).strip()
os.chdir(REPO)
sys.path.insert(0, str(REPO))
log = (ROOT / "provenance-transport-events.jsonl").open("x")
lock = threading.Lock()


def capture(request, count):
    url = urlsplit(str(getattr(request, "full_url", request)))
    event = {
        "ordinal": count,
        "request_type": type(request).__name__,
        "endpoint": {"scheme": url.scheme, "host": url.hostname, "path": url.path},
        "thread": threading.current_thread().name,
        "stack": [{"file": f.filename, "line": f.lineno, "function": f.name}
                  for f in traceback.extract_stack(limit=18)],
    }
    with lock:
        log.write(json.dumps(event, ensure_ascii=False) + "\n")
        log.flush()


class Probe:
    def pytest_collection_modifyitems(self, session, config, items):
        index = next(i for i, item in enumerate(items) if item.nodeid == TARGET)
        removed = items[index + 1:]
        items[:] = items[:index + 1]
        config.hook.pytest_deselected(items=removed)
        target = items[-1]
        source = textwrap.dedent(inspect.getsource(target.obj))
        old = "        requests.append(request)"
        assert source.count(old) == 1
        patched = source.replace(old, old + "\n        _probe_capture(request, len(requests))")
        target.obj.__globals__["_probe_capture"] = capture
        namespace = {}
        exec(compile(patched, "<in-memory-transport-probe>", "exec"), target.obj.__globals__, namespace)
        target.obj = namespace[target.obj.__name__]
        with (ROOT / "provenance-probe-selection.json").open("x") as out:
            json.dump({"revision": REVISION, "instrumented_source_sha256": hashlib.sha256(source.encode()).hexdigest(),
                       "target": TARGET, "selected": [item.nodeid for item in items],
                       "deselected": len(removed)}, out, indent=2)


code = pytest.main(["-q", "--tb=short"], plugins=[Probe()])
log.close()
assert not subprocess.check_output(["git", "status", "--porcelain"], cwd=REPO, text=True).strip()
raise SystemExit(code)
