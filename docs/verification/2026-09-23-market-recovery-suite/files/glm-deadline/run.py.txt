def save(path, obj):
    path.write_text(json.dumps(obj, indent=2, ensure_ascii=False) + '\n')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git(*args):
    return subprocess.check_output(['git', '-C', str(TREE), *args], text=True).strip()


def identity():
    return {'revision': git('rev-parse', 'HEAD'), 'status': git('status', '--porcelain', '--untracked-files=all'),
            'main': git('rev-parse', 'gitea/main')}

        chunks.append(f'\nSOURCE {name} sha256={sha(TREE / name)}\n' + '\n'.join(f'{i}: {line}' for i, line in enumerate(lines, 1)))

def make_out(name):
    out = ROOT / name
    out.mkdir(exist_ok=False)
    (out / 'tmp').mkdir()
    (out / 'users').mkdir()
    return out


def bounded(command, out, env, timeout, stem, sandbox=False):
    if sandbox:
        command = ['/usr/bin/sandbox-exec', '-f', str(out / 'tools.sb'), *command]
    started = time.monotonic()
    with (out / f'{stem}.stdout').open('xb') as stdout, (out / f'{stem}.stderr').open('xb') as stderr:
        proc = subprocess.Popen(command, cwd=TREE, env=env, stdin=subprocess.DEVNULL,
                                stdout=stdout, stderr=stderr, start_new_session=True)
        timeout_hit = False
        try:
            proc.wait(timeout=timeout)
        except subprocess.TimeoutExpired:
            timeout_hit = True
            os.killpg(proc.pid, signal.SIGTERM)
            try:
                proc.wait(timeout=5)
            except subprocess.TimeoutExpired:
                os.killpg(proc.pid, signal.SIGKILL)
                proc.wait()
    return {'exit_code': proc.returncode, 'deadline_hit': timeout_hit,
            'elapsed_seconds': round(time.monotonic() - started, 3), 'command': command}

                'Prefer one adversarial case plus a valid control over broad coverage with no artifact. '
                'State untested subclaims honestly. You cannot execute yet; next host phase executes your immutable probe. '
                'Do not request other reads or tools; they are unavailable. Do not declare PASS before execution.')
    else:

                'Report JSON must include verdict, claim, revision, verified_subclaims, not_verified_subclaims, '

                'positive_control (must classify intentional red assertion as probe_bug), limitations. '
                'Use PASS_WITH_LIMITS when only part of a claim was tested. Test collection/import errors are '
                'BLOCKED_PROBE, not product bugs; unexecuted claims are not_verified. Do not claim full repo or production acceptance. '
                'Only PASS/PASS_WITH_LIMITS/CHANGES_REQUIRED/BLOCKED_* verdicts are valid. All three artifacts are mandatory.')
    (out / 'packet.md').write_text(packet)

def counts(path):
    if not path.exists():
        return None
    root = ET.parse(path).getroot()
    suites = [root] if root.tag == 'testsuite' else list(root.iter('testsuite'))
    values = {key: sum(int(s.get(key, '0')) for s in suites) for key in ['tests', 'failures', 'errors', 'skipped']}
    values['passed'] = values['tests'] - values['failures'] - values['errors'] - values['skipped']
    return values


def execute(claim):
    before = identity()
    assert before['revision'] == REV and before['status'] == '', before
    delivered = ROOT / f'{claim}-explore/delivery'
    assert (delivered / 'manifest.json').is_file()
    manifest = json.loads((delivered / 'manifest.json').read_text())
    assert all(sha(delivered / name) == digest for name, digest in manifest.items())
    text = (delivered / 'probe.py').read_text()
    tree = ast.parse(text)
    assert any(isinstance(n, (ast.FunctionDef, ast.ClassDef)) and n.name.startswith(('test_', 'Test')) for n in tree.body)
    out = make_out(f'{claim}-execute')
    shutil.copy2(delivered / 'probe.py', out / 'probe.py')
    (out / 'positive_control.py').write_text('def test_intentional_red_probe_bug():\n    assert 1 == 2, "intentional positive control: probe_bug"\n')
    sandbox_profile(out)
    env = env_for(out)
    env['PYTEST_DISABLE_PLUGIN_AUTOLOAD'] = '1'
    env['PYTEST_ADDOPTS'] = ''
    env['REVIEW_CANDIDATE'] = str(TREE)
    inputs = {p.name: sha(p) for p in [out / 'probe.py', out / 'positive_control.py', out / 'tools.sb']}
    base_command = [PY, '-B', '-m', 'pytest', '-q', '-o', 'addopts=', '-p', 'no:cacheprovider', '--confcutdir', str(out)]
    record = {'claim': claim, 'before': before, 'python': PY, 'inputs_sha256': inputs,
              'independent_probe_provenance': str(delivered), 'production_access': 'OS sandbox denies', 'complete': False}
    save(out / 'execution.json', record)
    module = 'intelligence.services.finance_query'
    preflight = bounded([PY, '-B', '-c', f'import {module} as m; from pathlib import Path; assert Path(m.__file__).is_relative_to({str(TREE)!r}); print(m.__file__)'], out, env, 90, 'preflight', True)
    assert preflight['exit_code'] == 0, preflight
    for name, paths in [('probe', [str(out / 'probe.py')]), ('positive', [str(out / 'positive_control.py')]), ('author', AUTHOR_TESTS[claim])]:
        result = bounded([*base_command, *paths, f'--junitxml={out / (name + ".xml")}'], out, env, 180, name, True)
        result['counts'] = counts(out / f'{name}.xml')
        record[name] = result
        save(out / 'execution.json', record)
    record.update(complete=True, after=identity(), inputs_unchanged=inputs == {name: sha(out / name) for name in inputs})
    record['positive_control_detected'] = record['positive']['exit_code'] == 1 and record['positive']['counts'] == {'tests': 1, 'failures': 1, 'errors': 0, 'skipped': 0, 'passed': 0}
    save(out / 'execution.json', record)
    print(json.dumps({k: record[k] for k in ['claim', 'positive_control_detected']}), flush=True)
