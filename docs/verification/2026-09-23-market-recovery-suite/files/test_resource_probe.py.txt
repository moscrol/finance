import unittest
from resource_probe import pytest_pids


class ResourceProbeTests(unittest.TestCase):
    def test_macos_long_interpreter_path(self):
        self.assertEqual(pytest_pids(b' 71388 /opt/homebrew/Cellar/python@3.12/3.12.13_2/Frameworks/Python.framework/Versions/3.12/Resources/Python.app/Contents/MacOS/Python -m pytest -q\n'), [71388])

    def test_versioned_interpreter_and_flags(self):
        self.assertEqual(pytest_pids(b'12 /repo/.venv/bin/python3.12 -B -m pytest -q\n'), [12])

    def test_non_utf8_neighbor_does_not_erase_runner(self):
        self.assertEqual(pytest_pids(b'2 /bin/tool \xff\xfe\n3 /venv/bin/python -m pytest -q\n'), [3])

    def test_shell_and_grep_are_not_python(self):
        self.assertEqual(pytest_pids(b'4 /bin/bash -c /venv/bin/python -m pytest -q\n5 /bin/rg -m pytest\n'), [])

    def test_inline_python_payload_is_not_a_test_runner(self):
        self.assertEqual(pytest_pids(b'6 /venv/bin/python -c print(" -m pytest ")\n'), [])

    def test_non_pytest_module(self):
        self.assertEqual(pytest_pids(b'7 /venv/bin/python -m uvicorn main:app\n'), [])

    def test_multiple_runners(self):
        self.assertEqual(pytest_pids(b'8 /venv/bin/python -m pytest\n9 /venv/bin/python -m pytest -q\n'), [8, 9])


if __name__ == '__main__':
    unittest.main()
