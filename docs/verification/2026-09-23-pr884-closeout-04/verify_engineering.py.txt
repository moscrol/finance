from pathlib import Path
import datetime

import json

ROOT = Path(__file__).resolve().parent
TREE = ROOT / "finance-workspace-private"
REV = "db2605d1b1a6a929d9bac468c84ef6aeaddce9ff"
BASE = "b59d6eed0356ae093b52bd291ab328628de8790e"

proof = {}


def load(path):
    data = path.read_bytes()
    proof[str(path.relative_to(ROOT))] = hashlib.sha256(data).hexdigest()
    return json.loads(data)


def log(path, expected):
    raw = path.read_bytes()
    digest = hashlib.sha256(raw).hexdigest()
    assert digest == expected, path
    proof[str(path.relative_to(ROOT))] = digest
    return raw.decode()


records = {}
expected_checks = {
    "python": ["sandbox-preflight", "toolchain", "main-gate"],
    "frontend": ["sandbox-preflight", "toolchain", "dependency-install", "frontend-gate"],
    "registry": ["sandbox-preflight", "toolchain", "registry-0", "registry-1", "registry-2", "registry-3", "ledger-crosswalk"],
}
for leaf in ("python", "frontend", "registry"):
    directory = "python-native" if leaf == "python" else leaf
    record = load(ROOT / directory / "receipt.json")
    assert record["complete"] is True and record["accepted"] is True, record
    assert record["identity_stable"] is True
    assert record["before"] == record["after"] == {"revision": REV, "status": ""}
    assert record["revision"] == REV
    driver = "run.py"
    assert record["driver_sha256"] == hashlib.sha256((ROOT / driver).read_bytes()).hexdigest()
    if leaf == "python":
        assert record["outer_sandbox_on_main_gate"] is False
    assert [row["name"] for row in record["checks"]] == expected_checks[leaf]
    for row in record["checks"]:
        assert row["exit_code"] == 0 and not row.get("timed_out") and not row.get("resource_stop")
        assert not row.get("remaining_after_cleanup")
        text = log(ROOT / directory / (row["name"] + ".log.txt"), row["log_sha256"])
        if row["name"] == "sandbox-preflight":
            preflight = json.loads(text)
            assert preflight["accepted"] is True and preflight["profile"] == record["profile"]
        if row["name"] == "toolchain":
            toolchain = json.loads(text)
            assert toolchain["node"] == "v22.23.2" and toolchain["python"] == "3.12.13"
    records[leaf] = record

def inventory(payload):
    cases = list(ET.fromstring(payload).iter("testcase"))
    ids = {case.attrib["classname"] + "::" + case.attrib["name"] for case in cases}
    assert len(ids) == len(cases)
    return ids

counts = {}
for row in front["checks"]:
    assert row["exit_code"] == 0
    text = log(ROOT / "frontend/gate" / row["log"], row["log_sha256"])
    clean = re.sub(r"\x1b\[[0-9;]*m", "", text)
    if row["command"][-1] == "test":
        matches = re.findall(r"(?m)^\s*Tests\s+(\d+) passed", clean)
        assert len(matches) == 1
        counts["unit_passed"] = int(matches[0])
    if row["command"][-1] == "test:e2e":
        for label in ("passed", "skipped"):
            matches = re.findall(r"(?m)^\s*(\d+) " + label + r"(?:\s|$)", clean)
            assert len(matches) == 1
            counts["e2e_" + label] = int(matches[0])

assert counts == {"unit_passed": 120, "e2e_passed": 34, "e2e_skipped": 2}, counts
