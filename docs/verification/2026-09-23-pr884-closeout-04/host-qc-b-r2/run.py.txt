"""Run the host C5 probes under the offline sandbox with isolated temporary state."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

root = Path(__file__).resolve().parent
tree = root.parent / 'finance-workspace-private'
if sys.argv[1:] == ['inner']:
    import intelligence.userspace as userspace
    userspace.USERS_DIR = root / 'users'
    import pytest
    raise SystemExit(pytest.main(['-q', '-p', 'no:cacheprovider', '--tb=short', '--rootdir=' + str(root), '--basetemp=' + str(root / 'basetemp'), '--junitxml=' + str(root / 'junit.xml'), str(root / 'test_c5.py')]))
for name in ('users', 'tmp'):
    (root / name).mkdir(exist_ok=False)
profile = f'(version 1) (allow default) (deny network*) (deny file-write*) (allow file-write* (subpath {json.dumps(str(root))}) (literal "/dev/null")) (deny process-exec (literal "/usr/bin/security"))'
env = {'HOME': str(root), 'PATH': '/opt/homebrew/bin:/usr/bin:/bin', 'LANG': 'C.UTF-8',
       'TMPDIR': str(root / 'tmp'), 'PYTHONPATH': str(tree), 'PYTHONDONTWRITEBYTECODE': '1',
       'FORESIGHT_USERS_DIR': str(root / 'users'), 'FWP_TEST_RECEIPT': '0', 'FORESIGHT_LLM_KEYCHAIN': '0'}
command = ['/usr/bin/sandbox-exec', '-p', profile, sys.executable, '-B', str(Path(__file__)), 'inner']
with (root / 'output.log.txt').open('xb') as log:
    result = subprocess.run(command, cwd=tree, env=env, stdin=subprocess.DEVNULL, stdout=log, stderr=log, timeout=60)
(root / 'receipt.json').write_text(json.dumps({'role': 'host, not independent reviewer', 'command': command, 'exit_code': result.returncode, 'source_sha256': hashlib.sha256((root / 'test_c5.py').read_bytes()).hexdigest()}, indent=2) + '\n')
raise SystemExit(result.returncode)
