original = source.read_text().replace(str(ROOT/'spec'), str(OUT))

allow = '\n(allow file-read-metadata (literal "'+str(ROOT)+'") (literal "'+str(ROOT/'candidate')+'"))\n'

control = WORK/'test_sandbox_controls.py'
control.write_text('''import errno

env = {'HOME': str(WORK), 'PATH': '/opt/homebrew/bin:/usr/bin:/bin:/usr/sbin:/sbin',
       'LANG': 'C.UTF-8', 'TMPDIR': str(WORK/'tmp'), 'PYTHONDONTWRITEBYTECODE': '1',

       'FORESIGHT_USERS_DIR': str(WORK/'users')}
records = []

def run(label, profile, argv, expected):
    command = ['/usr/bin/sandbox-exec', '-f', str(OUT/profile)] + argv
    result = subprocess.run(command, cwd=TREE, env=env, capture_output=True, timeout=90)
    output = result.stdout+result.stderr
    (OUT/(label+'.log.txt')).write_bytes(output)
    row = {'label': label, 'command': command, 'returncode': result.returncode,
           'expected_returncode': expected, 'matched': result.returncode == expected,
           'output_sha256': hashlib.sha256(output).hexdigest(), 'output_bytes': len(output)}
    records.append(row)
    (OUT/'progress.json').write_text(json.dumps(records, indent=2)+'\n')

for profile, expected in [('before.sb', 2), ('after.sb', 0)]:
    label = profile.removesuffix('.sb')
    run(label+'-pytest', profile, [PYTHON, '-m', 'pytest', str(control), '-q', '-p', 'no:cacheprovider',

run('positive-control', 'after.sb', [PYTHON, '-B', str(positive)], 1)
