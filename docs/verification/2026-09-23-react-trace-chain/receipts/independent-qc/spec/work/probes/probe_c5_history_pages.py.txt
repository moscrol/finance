"""Independent spec probe (C5): absolute sample identity & paging hints in _result.

Pure-function level: exercise intelligence/services/historical_research/episode.py
_result/_model_projection directly with synthetic payloads. Expectations are derived
from the claimed contract, not from author test output:
  - sample ids are absolute result-row ids (enumerate start=payload offset), stable
    and non-overlapping across pages of the same immutable result;
  - next_offset = offset+returned_count iff returned_count>0 and page_end<total_matched,
    else None (null in the compact observation json);
  - the page-navigation prose advertises next_offset paging and cross-page id stability;
  - the caller's payload dict is not mutated.
"""

import sys

CANDIDATE = "/Users/a77/.finance-runtime/reviews/pr832-glm-qc-20260923-2255/candidate/finance-workspace-private"
if CANDIDATE not in sys.path:
    sys.path.insert(0, CANDIDATE)

from intelligence.services.historical_research.episode import _result  # noqa: E402

REF = "run-history-review/history-query-" + "b" * 64 + ".json"


def _payload(rows, *, total=None, returned=None, **extra):
    payload = {
        "query_id": "q-probe-c5",
        "operation": "inspect_history",
        "status": "research_only",
        "total_matched": len(rows) if total is None else total,
        "returned_count": len(rows) if returned is None else returned,
        "truncated": False,
        "preview": rows,
    }
    payload.update(extra)
    return payload


def _rows(n, start_day=3):
    return [
        {"entity_code": "A.FP", "trade_date": f"2026-08-{start_day + i:02d}"}
        for i in range(n)
    ]


def _sample_rows(result):
    rows = {}
    for item in result.evidence:
        detail = __import__("json").loads(item.detail)
        if "entity_code" in detail:
            rows[detail["sample"]] = detail["trade_date"]
    return rows


def _scope(result):
    import json

    for item in result.evidence:
        detail = json.loads(item.detail)
        if "total_matched" in detail:
            return detail
    raise AssertionError("scope block with total_matched not projected")


def test_pages_share_absolute_sample_ids_across_offsets():
    rows = _rows(5)
    first = _result(_payload(rows[:2], total=5), result_ref=REF, tool="read_history_result", offset=0)
    second = _result(_payload(rows[2:4], total=5), result_ref=REF, tool="read_history_result", offset=2)
    overlap = _result(_payload(rows[1:4], total=5), result_ref=REF, tool="read_history_result", offset=1)
    assert _sample_rows(first) == {0: "2026-08-03", 1: "2026-08-04"}
    assert _sample_rows(second) == {2: "2026-08-05", 3: "2026-08-06"}
    # Row id 1 must keep its identity and its trade_date on every page that shows it.
    assert _sample_rows(overlap)[1] == "2026-08-04" == _sample_rows(first)[1]


def test_next_offset_null_exactly_when_no_more_rows():
    rows = _rows(5)
    middle = _result(_payload(rows[:2], total=5), result_ref=REF, offset=1)
    last = _result(_payload(rows[3:5], total=5), result_ref=REF, offset=3)
    empty = _result(_payload([], total=5, returned=0), result_ref=REF, offset=4)
    zero_total = _result(_payload([], total=0, returned=0), result_ref=REF)
    assert _scope(middle)["next_offset"] == 3
    assert _scope(last)["next_offset"] is None  # page_end == total_matched
    assert _scope(empty)["next_offset"] is None  # empty page never advertises a next
    assert _scope(zero_total)["next_offset"] is None
    assert _scope(middle)["offset"] == 1 and _scope(zero_total)["offset"] == 0


def test_observation_advertises_paging_hints_and_absolute_ids():
    rows = _rows(2)
    page = _result(_payload(rows[:1], total=2), result_ref=REF, tool="read_history_result", offset=1)
    assert '"offset":1' in page.observation
    assert '"next_offset":null' in page.observation
    assert "跨页不重置" in page.observation and "按next_offset分页" in page.observation
    assert "null表示已到末页" in page.observation
    first = _result(_payload(rows[:1], total=2), result_ref=REF)
    assert '"next_offset":1' in first.observation and '"offset":0' in first.observation


def test_explicit_offset_wins_over_stale_payload_offset_and_payload_is_copied():
    rows = _rows(3)
    payload = _payload(rows[1:3], total=3, offset=99)  # stale caller-side offset
    snapshot = dict(payload)
    result = _result(payload, result_ref=REF, offset=1)
    assert set(_sample_rows(result)) == {1, 2}  # projection used the kwarg, not 99
    assert _scope(result)["offset"] == 1
    assert payload == snapshot  # no mutation of the caller's mapping


def test_truncated_flag_is_reported_as_given_and_scope_exposes_it():
    rows = _rows(4)
    page = _result(_payload(rows[1:2], total=4, truncated=True), result_ref=REF, offset=1)
    assert _scope(page)["truncated"] is True
    assert '"truncated":true' in page.observation
