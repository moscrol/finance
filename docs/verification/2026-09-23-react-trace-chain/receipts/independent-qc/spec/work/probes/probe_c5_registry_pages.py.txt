"""Independent spec probe (C5): registry-level paging + failure must not record success.

Uses the author helper _registry (real duckdb session, offline tmp_path) but with
self-derived scenarios distinct from tests/test_history_model_projection.py:
  - a final page is still flagged truncated (it is part, not the whole result) and
    next_offset is null;
  - a read past the end yields an empty page, never a next_offset;
  - sample ids stay absolute across an initial query and overlapping read pages;
  - a projection failure raised INSIDE the real _result (deeper seam than the
    author's _result monkeypatch) must not append a successful history_results
    entry nor mutate the stored artifact.
"""

import sys
from copy import deepcopy

CANDIDATE = "/Users/a77/.finance-runtime/reviews/pr832-glm-qc-20260923-2255/candidate/finance-workspace-private"
if CANDIDATE not in sys.path:
    sys.path.insert(0, CANDIDATE)

import pytest  # noqa: E402

from intelligence.services.historical_research import episode  # noqa: E402
from intelligence.tests.test_historical_research_episode import _registry  # noqa: E402


def _samples(result):
    rows = {}
    for item in result.evidence:
        detail = __import__("json").loads(item.detail)
        if "entity_code" in detail:
            rows[detail["sample"]] = detail["trade_date"]
    return rows


def _query(registry, context, **extra):
    arguments = {
        "operation": "inspect_history",
        "start": "2026-08-03",
        "end": "2026-08-04",
        "entity_codes": ["A.FP"],
    }
    arguments.update(extra)
    return registry.execute("history_query", arguments, context=context, step_id="q")


def test_final_page_is_truncated_with_null_next_and_absolute_ids(tmp_path):
    registry, context, _session = _registry(tmp_path)
    first = _query(registry, context, preview_limit=1)
    ref = first.telemetry["result_ref"]
    assert first.telemetry["next_offset"] == 1 and first.telemetry["offset"] == 0
    assert set(_samples(first)) == {0}
    last = registry.execute(
        "read_history_result", {"result_ref": ref, "offset": 1, "limit": 1},
        context=context, step_id="last",
    )
    rows = _samples(last)
    assert rows == {1: "2026-08-04"}
    scope = next(
        __import__("json").loads(item.detail)
        for item in last.evidence
        if "total_matched" in item.detail
    )
    # The page holds 1 of 2 rows: still truncated even though it is the final page.
    assert scope["truncated"] is True and scope["next_offset"] is None
    assert scope["offset"] == 1 and scope["total_matched"] == 2
    overlap = registry.execute(
        "read_history_result", {"result_ref": ref, "offset": 0, "limit": 2},
        context=context, step_id="overlap",
    )
    assert _samples(overlap) == {0: "2026-08-03", 1: "2026-08-04"}


def test_read_past_the_end_returns_empty_page_without_next_offset(tmp_path):
    registry, context, _session = _registry(tmp_path)
    first = _query(registry, context)
    ref = first.telemetry["result_ref"]
    beyond = registry.execute(
        "read_history_result", {"result_ref": ref, "offset": 5, "limit": 1},
        context=context, step_id="beyond",
    )
    assert beyond.trace.status == "success"
    assert _samples(beyond) == {}
    assert beyond.telemetry["next_offset"] is None and beyond.telemetry["offset"] == 5
    assert context.history_results[-1]["returned_count"] == 0


def test_projection_failure_inside_real_result_records_no_success(tmp_path, monkeypatch):
    registry, context, session = _registry(tmp_path)
    first = _query(registry, context, preview_limit=1)
    ref = first.telemetry["result_ref"]
    original = deepcopy(session.read(ref))
    before = deepcopy(context.history_results)

    real_projection = episode._model_projection

    def fail(payload, result_ref):
        assert result_ref == ref  # the real _result forwards ref to the projection
        raise ValueError("probe: projection failed")

    monkeypatch.setattr(episode, "_model_projection", fail)
    with pytest.raises(ValueError, match="probe: projection failed"):
        registry.execute(
            "read_history_result", {"result_ref": ref, "offset": 1, "limit": 1},
            context=context, step_id="failed-page",
        )
    assert context.history_results == before  # no success entry registered
    assert session.read(ref) == original  # stored artifact untouched
    monkeypatch.setattr(episode, "_model_projection", real_projection)
    retry = registry.execute(
        "read_history_result", {"result_ref": ref, "offset": 1, "limit": 1},
        context=context, step_id="retry",
    )
    assert _samples(retry) == {1: "2026-08-04"}  # seam restored: same ref still pages
