import errno
from pathlib import Path
import socket
import sqlite3


def test_file_backed_sqlite(tmp_path):
    with sqlite3.connect(tmp_path / "control.sqlite3") as connection:
        connection.execute("PRAGMA journal_mode=WAL")
        connection.execute("CREATE TABLE control (value INTEGER)")
        connection.execute("INSERT INTO control VALUES (81)")
        assert connection.execute("SELECT value FROM control").fetchone() == (81,)


def test_production_and_escape_still_denied():
    for target in FORBIDDEN_READS:
        try:
            Path(target).read_bytes()
        except PermissionError:
            pass
        else:
            raise AssertionError("forbidden read allowed: " + target)
    for target in FORBIDDEN_WRITES:
        try:
            Path(target).write_text("denied control")
        except PermissionError:
            pass
        else:
            raise AssertionError("forbidden write allowed: " + target)
    for host, port in [("127.0.0.1", 8792), ("127.0.0.1", 19899), ("192.0.2.1", 443)]:
        with socket.socket() as client:
            client.settimeout(0.2)
            try:
                client.connect((host, port))
            except OSError as error:
                assert error.errno in (errno.EPERM, errno.EACCES), error
            else:
                raise AssertionError("network allowed")
