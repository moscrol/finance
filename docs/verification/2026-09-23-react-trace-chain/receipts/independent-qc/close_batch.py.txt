import hashlib
import json
from pathlib import Path
import re
import shutil

ROOT = Path(__file__).resolve().parent
DOCS = Path('/Users/a77/fwp-wt-react-trace-chain-docs-0923')
DEST = DOCS/'docs/verification/2026-09-23-react-trace-chain/receipts/independent-qc'

def load(path):
    return json.loads(path.read_text())

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

stages = {}
requests = 0
for stage in ['gateway', 'explore', 'execute']:
    controller = load(ROOT/'spec'/f'{stage}.controller.json')
    counts = load(ROOT/'spec'/f'{stage}-shim/counts.json')
    assert counts['active'] == 0 and counts['shutdown_complete']
    assert counts['requests'] == counts['dispatched'] == counts['completed']
    assert counts['failed'] == counts['cancelled'] == 0 and controller['request_counts_agree']
    requests += counts['requests']
    stages[stage] = controller
assert requests == 49
assert stages['execute']['status'] == 'BLOCKED_STAGE_OR_ACCOUNTING'
assert not (ROOT/'spec/report').exists() and not list((ROOT/'quality').glob('*-shim'))
checked_artifacts = 0
for stage in ['explore', 'execute']:
    execution = load(ROOT/'spec'/stage/'execution.json')
    assert execution['candidate_before'] == execution['candidate_after'] == {'revision':REV, 'status':''}
    assert execution['author_before'] == execution['author_after'] == {'revision':REV, 'status':''}
    assert execution['inputs_unchanged'] and not execution['timed_out'] and not execution['errors']
    for relative, expected in execution['artifact_sha256'].items():
        assert digest(ROOT/'spec'/stage/relative) == expected, (stage, relative)
        checked_artifacts += 1
report = (ROOT/'spec/execute/REPORT.md').read_text()
blocks = re.findall(r'```json\s*(.*?)```', report, flags=re.S)
assert len(blocks) == 1
try:
    json.loads(blocks[0])
except json.JSONDecodeError as error:
    report_error = {'message':error.msg, 'line':error.lineno, 'column':error.colno, 'position':error.pos}
else:
    raise AssertionError('Expected the preserved invalid JSON final report')
identities = {}

    sha = subprocess.check_output(['git','-C',str(tree),'rev-parse','HEAD'],text=True).strip()
    status = subprocess.check_output(['git','-C',str(tree),'status','--porcelain','--untracked-files=all'],text=True)
    assert sha == REV and not status
    identities[name] = {'tree':str(tree), 'revision':sha, 'dirty':False}
remote = subprocess.check_output(['git','-C',str(DOCS),'ls-remote','gitea','refs/heads/main','refs/heads/fix/react-trace-qc-0921'],text=True)
refs = {row.split()[1]:row.split()[0] for row in remote.splitlines()}
assert refs['refs/heads/fix/react-trace-qc-0921'] == REV
controls = load(ROOT/'offline-repair-v2/receipt.json')
replay = load(ROOT/'offline-repair-v2/registry-replay.json')
assert controls['status'] == 'PASS' and replay['exit_code'] == 0 and replay['probe_bytes_unchanged']

selected = set()

             'check_offline_permissions.py','check_offline_permissions_v2.py','check_registry_replay.py',
             'glm_review_shim.py','glm_credential.py']:
    selected.add(ROOT/name)
for axis in ['spec','quality']:
    folder = ROOT/axis
    for path in folder.iterdir():
        if path.is_file() and path.name != 'credential.json':
            selected.add(path)
    selected.update((folder/'inputs').glob('*'))
    selected.add(folder/'pi-config/settings.json')

entries = []
for source in sorted(selected):
    assert source.is_file() and not source.is_symlink()
    relative = source.relative_to(ROOT)
    if relative.parent == Path('spec/work/tmp'):
        relative = Path('spec/work/diagnostic-probes')/relative.name
    if source.suffix in {'.py','.mjs','.sb','.log'}:
        relative = Path(str(relative)+'.txt')
    target = DEST/relative
    target.parent.mkdir(parents=True,exist_ok=True)
    shutil.copyfile(source,target)
    assert target.read_bytes() == source.read_bytes()
    entries.append({'path':str(relative), 'source':str(source), 'bytes':target.stat().st_size, 'sha256':digest(target)})
