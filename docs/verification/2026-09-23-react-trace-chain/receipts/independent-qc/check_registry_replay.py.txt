import json

ROOT = Path(__file__).resolve().parent
OUT = ROOT/'offline-repair-v2'
WORK = OUT/'work'
TREE = ROOT/'candidate/finance-workspace-private'

original = ROOT/'spec/work/probes/probe_c5_registry_pages.py'

           '-q', '-p', 'no:cacheprovider', '--rootdir='+str(TREE), '--basetemp='+str(WORK/'tmp/registry-replay')]

           'old_failures_preserved':True, 'independent_qc_still_blocked':True}
