import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import { pathToFileURL, fileURLToPath } from 'node:url';
const batch = path.dirname(fileURLToPath(import.meta.url));
const seed = path.join(batch,'spec');
const out = path.join(batch,'read-delivery-control');
fs.mkdirSync(out);
const source = fs.readFileSync(path.join(seed,'review.mjs'),'utf8');
const marker = '      return content(dir, record);\n    }) });\n  }\n';
assert.equal(source.split(marker).length,2);
const fixed = source.replace(marker, '      if (kind === \'read\') return { content: [{ type: \'text\', text: fs.readFileSync(path.join(dir,\'output.log\'),\'utf8\') }], details: record };\n'+marker);
const results = [];
for(const [name,code] of [['before',source],['after',fixed]]) {
  const root = path.join(out,name);
  fs.mkdirSync(root);
  for(const folder of ['work/tmp','work/users','work/probes','receipt/commands']) fs.mkdirSync(path.join(root,folder),{recursive:true});
  for(const file of ['config.json','provider-template.json','file_tool.py','junit_summary.py']) fs.copyFileSync(path.join(seed,file),path.join(root,file));
  const rules = fs.readFileSync(path.join(seed,'tools.sb'),'utf8').replaceAll(seed,root)+'\n(allow file-read-metadata (literal "'+out+'"))\n';
  fs.writeFileSync(path.join(root,'tools.sb'),rules);
  fs.writeFileSync(path.join(root,'review.mjs'),code);
  const {install} = await import(pathToFileURL(path.join(root,'review.mjs')));
  const tools = {};
  install({registerProvider(){},on(){},registerTool(t){tools[t.name]=t;}},{out:path.join(root,'receipt'),token:'OFFLINE_FAKE_CREDENTIAL',stage:'preflight'});
  const fixture = path.join(root,'work/tmp/lines.txt');
  const lines = Array.from({length:400},(_,i)=>'LINE_'+String(i+1).padStart(4,'0'));
  fs.writeFileSync(fixture,lines.join('\n')+'\n');
  const result = await tools.read.execute('read400',{path:fixture,offset:1,limit:400});
  const text = result.content.map(c=>c.text).join('\n');
  const firstVisible = lines.find(line=>text.includes(line));
  const allVisible = lines.every(line=>text.includes(line));
  fs.writeFileSync(path.join(root,'returned-content.txt'),text);
  assert.equal(allVisible,name==='after');
  assert.equal(text.includes('LINE_0400'),true);
  const row = {case:name,requested_lines:400,returned_fixture_lines:lines.filter(line=>text.includes(line)).length,first_visible:firstVisible,last_visible:'LINE_0400',all_visible:allVisible,target_exit_code:result.details.target_exit_code};
  results.push(row);
}
const receipt = {status:'PASS',model_requests:0,scope:'Host read-tool delivery diagnostic only; sealed model-batch tooling unchanged.',cause:'Common tail truncation hid the beginning of read responses after the file helper had already applied its own read bounds.',fix:'Read returns the already-bounded file helper response without a second tail truncation; command logs still use bounded tail.',results};
fs.writeFileSync(path.join(out,'receipt.json'),JSON.stringify(receipt,null,2)+'\n');
console.log(JSON.stringify(receipt));
