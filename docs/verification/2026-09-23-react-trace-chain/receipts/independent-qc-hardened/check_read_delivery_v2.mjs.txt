import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {pathToFileURL,fileURLToPath} from 'node:url';
const batch=path.dirname(fileURLToPath(import.meta.url)), seed=path.join(batch,'spec'), out=path.join(batch,'read-delivery-control-v2');
fs.mkdirSync(out);
const source=fs.readFileSync(path.join(seed,'review.mjs'),'utf8');
const marker='      return content(dir, record);\n    }) });\n  }\n';
assert.equal(source.split(marker).length,2);
const fixed=source.replace(marker,"      if (kind === 'read') return { content: [{ type: 'text', text: fs.readFileSync(path.join(dir,'output.log'),'utf8') }], details: record };\n"+marker);
const helper=fs.readFileSync(path.join(seed,'file_tool.py'),'utf8');
const old="    text = ''.join(lines[start:end])\n    data = text.encode()\n    print(data[:48000].decode('utf-8', errors='ignore'), end='')\n    if end < len(lines) or len(data) > 48000:\n        print(f'\\n[truncated; total lines={len(lines)}, selected={start + 1}:{end}; request a later offset]')";
assert.equal(helper.split(old).length,2);
const repaired=helper.replace(old,"    selected = []\n    size = 0\n    for line in lines[start:end]:\n        width = len(line.encode())\n        if size + width > 48000:\n            break\n        selected.append(line)\n        size += width\n    if start < end and not selected:\n        raise ValueError('single line exceeds read byte limit; use a targeted inspect query')\n    end = start + len(selected)\n    print(''.join(selected), end='')\n    if end < len(lines):\n        print(f'\\n[truncated; total lines={len(lines)}, selected={start + 1}:{end}; next_offset={end + 1}]')");
const results=[];
for(const [name,code,py] of [['before',source,helper],['after',fixed,repaired]]) {
  const root=path.join(out,name); fs.mkdirSync(root);
  for(const dir of ['work/tmp','work/users','work/probes','receipt/commands']) fs.mkdirSync(path.join(root,dir),{recursive:true});
  for(const file of ['config.json','provider-template.json','junit_summary.py']) fs.copyFileSync(path.join(seed,file),path.join(root,file));
  fs.writeFileSync(path.join(root,'tools.sb'),fs.readFileSync(path.join(seed,'tools.sb'),'utf8').replaceAll(seed,root)+'\n(allow file-read-metadata (literal "'+out+'"))\n');
  fs.writeFileSync(path.join(root,'review.mjs'),code); fs.writeFileSync(path.join(root,'file_tool.py'),py);
  const {install}=await import(pathToFileURL(path.join(root,'review.mjs')));
  const tools={}; install({registerProvider(){},on(){},registerTool(t){tools[t.name]=t;}},{out:path.join(root,'receipt'),token:'OFFLINE_FAKE_CREDENTIAL',stage:'preflight'});
  for(const heavy of [false,true]) {
    const file=path.join(root,'work/tmp/'+(heavy?'heavy':'short')+'.txt');
    const lines=Array.from({length:400},(_,i)=>'ROW_'+String(i+1).padStart(4,'0')+(heavy?' '+String.fromCodePoint(0x6d4b).repeat(60):''));
    fs.writeFileSync(file,lines.join('\n')+'\n');
    const received=[]; let offset=1,calls=0;
    do {
      const result=await tools.read.execute('read-'+calls,{path:file,offset,limit:400});
      const text=result.content.map(c=>c.text).join('\n');
      fs.writeFileSync(path.join(root,`${heavy?'heavy':'short'}-${calls}.returned.txt`),text);
      received.push(...[...text.matchAll(/^ROW_(\d{4})/gm)].map(m=>Number(m[1])));
      const next=text.match(/next_offset=(\d+)/);
      offset=next?Number(next[1]):null; calls++;
      assert.ok(calls<=3);
    } while(offset && name==='after');
    const complete=JSON.stringify(received)===JSON.stringify(Array.from({length:400},(_,i)=>i+1));
    assert.equal(complete,name==='after');
    results.push({case:name,heavy_utf8:heavy,calls,received_lines:received.length,first:received[0],last:received.at(-1),complete});
  }
  if(name==='after') {
    const file=path.join(root,'work/tmp/oversize.txt'); fs.writeFileSync(file,'x'.repeat(48001));
    await assert.rejects(()=>tools.read.execute('oversize',{path:file}),/single line exceeds read byte limit/);
  }
}
const receipt={status:'PASS',model_requests:0,scope:'Post-stop isolated delivery fix only; no model signature, no changes to sealed batch.',results,oversize_line_fails_explicitly:true};
fs.writeFileSync(path.join(out,'receipt.json'),JSON.stringify(receipt,null,2)+'\n'); console.log(JSON.stringify(receipt));
