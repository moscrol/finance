def identity():
    def git(*args):
        return subprocess.check_output(["git", "-C", str(TREE), *args], text=True).strip()
    return {"revision": git("rev-parse", "HEAD"), "status": git("status", "--porcelain"),
            "tree_object": git("rev-parse", "HEAD^{tree}")}


def child(case_name, mutate):
    sys.path.insert(0, str(TREE))
    os.chdir(TREE)
    os.umask(0o022)
    module_name, function_name, old, new, target = CASES[case_name]
    if mutate:
        module = importlib.import_module(module_name)
        source = Path(module.__file__).read_text()
        node = next(n for n in ast.parse(source).body if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)) and n.name == function_name)
        original = ast.get_source_segment(source, node)
        assert original.count(old) == (2 if case_name == "mismatch_as_failure" else 1)
        replacement = original.replace(old, new)
        assert replacement != original
        # Compile only the selected function, preserving module types/registrations.
        exec(compile("from __future__ import annotations\n" + replacement, str(module.__file__), "exec"), module.__dict__)
    import pytest
    tag = "mutant" if mutate else "baseline"
    output = OUT / case_name
    return pytest.main(["-q", "-p", "no:cacheprovider", "-o", "tmp_path_retention_policy=failed",
                       "-o", "tmp_path_retention_count=1", f"--basetemp={output / (tag + '-tmp')}",
                       f"--junitxml={output / (tag + '.xml')}", target])


def main():
    if len(sys.argv) == 4 and sys.argv[1] == "child":
        return child(sys.argv[2], sys.argv[3] == "mutant")
    OUT.mkdir(exist_ok=False)
    before = identity()
    assert before["revision"] == SHA and not before["status"], before
    results = []
    env = {"HOME": str(Path.home()), "PATH": "/opt/homebrew/bin:/usr/bin:/bin:/usr/sbin:/sbin",
           "LANG": "C.UTF-8", "PYTHONDONTWRITEBYTECODE": "1", "FWP_TEST_RECEIPT": "0"}
    for name in CASES:
        folder = OUT / name
        folder.mkdir()
        row = {"case": name, "target": CASES[name][-1]}
        for tag in ("baseline", "mutant"):
            with (folder / (tag + ".log")).open("xb") as log:
                result = subprocess.run([sys.executable, __file__, "child", name, tag], cwd=TREE,
                                        env=env, stdout=log, stderr=subprocess.STDOUT, timeout=120)
            root = ET.parse(folder / (tag + ".xml")).getroot()
            failures = root.findall(".//failure")
            errors = root.findall(".//error")
            row[tag] = {"exit_code": result.returncode, "failures": len(failures), "errors": len(errors),
                        "testcases": len(root.findall(".//testcase")),
                        "assertion_failure": any("AssertionError" in (f.text or "") or "assert " in (f.text or "") for f in failures)}
        row["caught"] = (row["baseline"]["exit_code"] == 0 and row["baseline"]["testcases"] > 0
                         and row["mutant"]["exit_code"] == 1 and row["mutant"]["errors"] == 0
                         and row["mutant"]["failures"] > 0 and row["mutant"]["assertion_failure"])
        results.append(row)
        (OUT / "progress.json").write_text(json.dumps(results, indent=2) + "\n")
        print(json.dumps(row), flush=True)
    after = identity()
    report = {"revision": SHA, "before": before, "after": after, "source_unchanged": before == after,
              "cases": results, "pass": all(r["caught"] for r in results) and before == after}
    (OUT / "result.json").write_text(json.dumps(report, indent=2) + "\n")
    return 0 if report["pass"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
