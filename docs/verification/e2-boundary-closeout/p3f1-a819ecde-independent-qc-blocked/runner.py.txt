Not an OS sandbox. Application tests must use the reviewer's audited launcher.

import subprocess
from pathlib import Path

prefix = Path('/tmp/e2-p3f1-qc-a819ecde')
cmd = [
    '/opt/homebrew/bin/pi', '--provider', 'dcprwo', '--model', 'gpt-6-astra',
    '--thinking', 'low', '--mode', 'json', '--print', '--no-session',
    '--no-extensions', '--extension',
    '/Users/a77/.pi/agent/extensions/retryable-stream-errors.ts',
    '--no-skills', '--no-prompt-templates', '--tools', 'read,bash,write,edit',
    '--offline', '@' + str(prefix) + '-prompt.txt',
]
with open(str(prefix) + '.jsonl', 'x') as out, open(str(prefix) + '.err', 'x') as err:
    try:
        result = subprocess.run(cmd, cwd=prefix, stdout=out, stderr=err, timeout=1200)
        code = result.returncode
    except subprocess.TimeoutExpired:
        code = 124
Path(str(prefix) + '.exit').write_text(str(code) + '\n')
