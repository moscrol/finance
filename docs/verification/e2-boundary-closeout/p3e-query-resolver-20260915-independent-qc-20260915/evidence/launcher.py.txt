"""Author-only P3e pytest launcher. Not an OS sandbox or independent verdict.
Count before denying Python audit IO; tests additionally guard native DuckDB IO.
"""
import json
import os
from pathlib import Path
import sys
import traceback

ROOT = Path(os.environ['E2_TEST_ROOT']).resolve()
EVIDENCE = Path(os.environ['E2_TEST_EVIDENCE']).resolve()
NAME = os.environ['E2_TEST_NAME']
DATA = (Path('/tmp/e2-p3e-qc-0b83e14f-data') / NAME).resolve()
EVIDENCE.mkdir(parents=True, exist_ok=True)
for key in list(os.environ):
    if any(x in key for x in ('API_KEY', 'TOKEN', 'SECRET')) or key.startswith(('FORESIGHT_', 'SUBCONSCIOUS_')):
        os.environ.pop(key, None)
for key, value in {
    'HOME': DATA / 'home', 'FINANCE_WS': DATA / 'finance',
    'FORESIGHT_USERS_DIR': DATA / 'users', 'FORESIGHT_EPISODE_STORE': DATA / 'episodes',
    'MARKET_FEATURE_STORE_DB': DATA / 'absent.duckdb',
    'KB_VAULT': DATA / 'kb', 'KNOWLEDGE_WIKI': DATA / 'kb',
    'ENTITY_ANCHOR_SECURITIES_DB': '0',
    'PYTEST_DISABLE_PLUGIN_AUTOLOAD': '1', 'PYTHONDONTWRITEBYTECODE': '1',
}.items():
    os.environ[key] = str(value)
(DATA / 'home').mkdir(parents=True, exist_ok=True)
sys.dont_write_bytecode = True
sys.path.insert(0, str(ROOT))
violations = []


def deny(event, details):
    violations.append({'event': event, 'details': str(details), 'stack': traceback.format_stack(limit=14)})
    raise PermissionError('Author test forbidden IO: ' + event)


def audit(event, args):
    if event in {'socket.connect', 'socket.connect_ex', 'socket.getaddrinfo', 'socket.sendto', 'socket.bind', 'socket.gethostbyname'}:
        deny(event, args)
    if event in {'subprocess.Popen', 'os.system', 'os.posix_spawn', 'os.exec', 'os.fork'}:
        deny(event, args[:2])
    if event in {'open', 'os.listdir', 'os.scandir'} and args and isinstance(args[0], (str, bytes, os.PathLike)):
        p = Path(os.fsdecode(args[0])).resolve()
        protected_repo = any(p.is_relative_to(ROOT / sub) for sub in ('intelligence/users', 'db', '.agent-memory'))
        protected_home = p.is_relative_to('/Users/a77') and not (
            p.is_relative_to(ROOT) or p.is_relative_to('/Users/a77/finance-workspace-private/.venv-workbench')
        )
        if protected_repo or protected_home or p.name.startswith('.env') or p.name in {'mcp_config.json', 'feishu_config.json'}:
            deny(event, str(p))


sys.addaudithook(audit)
code = 2
try:
    import duckdb
    original_connect = duckdb.connect
    def guarded_connect(database=':memory:', *args, **kwargs):
        if str(database) != ':memory:' and not Path(database).resolve().is_relative_to(DATA):
            deny('duckdb.connect', str(database))
        return original_connect(database, *args, **kwargs)
    duckdb.connect = guarded_connect
    import pytest
    code = int(pytest.main(['-p', 'no:cacheprovider', '--confcutdir=' + str(ROOT / 'intelligence/tests'), *sys.argv[1:]]))
finally:
    (EVIDENCE / (NAME + '-io.json')).write_text(json.dumps({'executor': 'independent pi reviewer', 'forbidden_attempts': len(violations), 'events': violations, 'pytest_exit': code}, ensure_ascii=False, indent=2))
    (EVIDENCE / (NAME + '-pytest.exit')).write_text(str(code) + '\n')
assert not violations, f'Forbidden attempts: {len(violations)}'
sys.exit(code)
