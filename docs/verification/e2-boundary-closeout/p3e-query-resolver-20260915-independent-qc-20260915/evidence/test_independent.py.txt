"""Independent probes; does not import author tests."""
import json
from collections import Counter
from pathlib import Path
import duckdb
import pytest
from intelligence.adapters.knowledge import KnowledgeAdapter, clear_relation_cache
from intelligence.services import query_resolution as qr, entity_anchor

@pytest.mark.parametrize('prefix,restricted', [
    ('不读取材料外数据。', True),
    ('仅根据以下材料回答。', True),
    ('不要联网。', False),
    ('可以查真实数据。', False),
    ('', False),
])
def test_independent_scope(tmp_path, monkeypatch, prefix, restricted):
    wiki = tmp_path / 'wiki'
    rel = wiki / 'relations'
    rel.mkdir(parents=True)
    (rel / 'entity_exposures.json').write_text(json.dumps({'entities': {'星舟设备': {'codes': ['300019.SZ'], 'concepts': {'合成专题': {}}}}}))
    (rel / 'aliases.json').write_text('{"aliases": {}}')
    db = tmp_path / 'stock.duckdb'
    with duckdb.connect(str(db)) as con:
        con.execute('create table fact_stock_daily(stock_name varchar, stock_ts_code varchar)')
        con.execute("insert into fact_stock_daily values ('星舟设备', '300019.SZ')")
    monkeypatch.setenv('ENTITY_ANCHOR_SECURITIES_DB', str(db))
    adapter = KnowledgeAdapter(wiki_root=wiki)
    clear_relation_cache(); entity_anchor._clear_entity_lexicon_cache(); qr._THEME_CACHE.clear()
    counts = Counter()
    op, stat, connect = Path.open, Path.stat, duckdb.connect
    def track(p, label):
        if p.is_relative_to(wiki) or p == db:
            counts[label] += 1
    def opening(p, *a, **kw):
        track(p, 'open'); return op(p, *a, **kw)
    def stating(p, *a, **kw):
        track(p, 'stat'); return stat(p, *a, **kw)
    def connecting(database, *a, **kw):
        counts['connect'] += 1
        return connect(database, *a, **kw)
    monkeypatch.setattr(Path, 'open', opening)
    monkeypatch.setattr(Path, 'stat', stating)
    monkeypatch.setattr(duckdb, 'connect', connecting)
    resolver = qr.QueryResolver(adapter)
    query = prefix + '\n\n「星舟设备收入80，成本60。」\n\n1. 星舟设备的利润率如何计算？'
    for warm in (False, True):
        counts.clear()
        result = resolver.resolve(query)
        if restricted:
            assert not counts, dict(counts)
            assert result.envelope.task_frame.material_contract.data_scope == 'material_only'
            assert result.anchor is None and not result.candidates and not result.comparison_entities
        else:
            assert counts['stat'] > 0
            if not warm:
                assert counts['open'] > 0 and counts['connect'] > 0
            assert result.anchor.entity == '星舟设备'
    print(prefix, 'restricted=', restricted, 'last_counts=', dict(counts))
