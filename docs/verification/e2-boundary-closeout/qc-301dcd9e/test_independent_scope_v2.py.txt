def restrict(context, caps=()):
    c = context.contract
    return replace(context, contract=replace(c, allowed_capabilities=tuple(caps),
        evidence_plan=EvidencePlan(requirements=()), required_outputs=tuple(
        replace(o, evidence_types=tuple(t for t in o.evidence_types if t in caps)) for o in c.required_outputs)))

class Sink:
    def __init__(self): self.events=[]; self.fail=False
    def emit(self, kind, payload):
        self.events.append((kind, dict(payload)))
        if self.fail: raise ValueError('intentional sink failure')

def spec(count, effect='local_read', capability='finance_query', name='same', runner=None):
    def parse(raw):
        count['parser'] += 1
        return raw.get('query', ''), raw.get('query', '')
    def run(q, ctx):
        count['runner'] += 1
        if effect != 'local_read':
            count['forbidden_runner'] += 1
            raise AssertionError('denied implementation ran')
        return ([AgentEvidence(tool=name,title='temporary',detail='fixture '+str(q),source='local',source_date='2026-07-24')],
            'temporary observation', ProviderTrace(provider='fixture', capability=capability, status='success'))
    return ToolSpec(name=name, capability=capability, description='fixture',cost='local',freshness='stable',
        runner=runner or run,parse_arguments=parse,io_effect=effect)

def session(context, registry, sink=None):
    s = EpisodeScope(context.contract.task_id, 'temporary-user', context, registry,event_sink=sink)
    return s, EpisodeToolBatchSession(scope=s)

def call(cid='x', query='new'):
    return (ModelToolCall(cid,'same',{'query':query}),)

@pytest.mark.parametrize('variant',['unknown','external_or_mixed','different_capability','deleted','contract_removed'])
@pytest.mark.parametrize('entry',['menu_first','execute_first'])
def test_replacement_matrix(variant,entry):
    _, ctx = fixture()
    good, bad, dispatch = Counter(), Counter(), []
    reg = ResearchToolRegistry((spec(good),))
    sink=Sink(); old,batch=session(ctx,reg,sink)
    old.record_invocation('history'); old.record_derive_mismatch('before')
    sink.fail=True; old.emit('old_fail',{}); sink.fail=False
    if variant in ('unknown','external_or_mixed'): replacement=ResearchToolRegistry((spec(bad,variant),))
    elif variant=='different_capability': replacement=ResearchToolRegistry((spec(bad,capability='web_search'),))
    elif variant=='deleted': replacement=reg.without('same')
    else: replacement=reg; ctx=restrict(ctx)
    batch.on_dispatch=dispatch.append
    if entry=='menu_first':
        menu=batch.menu(registry=replacement,context=ctx)
        assert not menu.visible
        assert tool_definitions_for_menu(ToolMenu(('same',),(),20),registry=replacement,context=ctx)==[]
    result=batch.execute(call(),registry=replacement,context=ctx,remaining_slots=2)
    current=batch._scope
    assert result.executed_count==0 and result.items[0].error=='unknown_or_unauthorized_tool'
    assert not current.authorize('same').allowed
    assert good==bad==Counter() and dispatch==[]
    denials=[p for k,p in sink.events if k==TOOL_ERROR]
    assert len(denials)==1 and denials[0]['reason']
    assert denials[0]['reason']==current.authorize('same').reason
    assert current.invoked_tools is old.invoked_tools and current._derive_mismatches is old._derive_mismatches
    assert current._event_sink_failures is old._event_sink_failures
    assert current.event_sink is old.event_sink and current.dump()['event_sink_failures']==1
    assert current.user_id==old.user_id and old.authorize('same').allowed
    with pytest.raises(FrozenInstanceError): old.registry=replacement
    COUNTS['replacement_denials']+=1

@pytest.mark.parametrize('scope',['full','local_only'])
def test_two_concurrent_batches_bind_under_lock(scope):
    _,ctx=fixture(scope); entered=Event(); release=Event(); bad=Counter(); good=Counter()
    base=spec(good)
    def slow(q, tc):
        entered.set(); assert release.wait(3)
        return base.runner(q,tc)
    reg=ResearchToolRegistry((replace(base,runner=slow),)); sink=Sink(); old,batch=session(ctx,reg,sink)
    result=[]; errors=[]; dispatch=[]; batch.on_dispatch=dispatch.append
    def first():
        try: result.append(batch.execute(call('old'),registry=reg,context=ctx,remaining_slots=1))
        except Exception as e: errors.append(e)
    def second():
        try: result.append(batch.execute(call('new'),registry=ResearchToolRegistry((spec(bad,'unknown'),)),context=restrict(ctx),remaining_slots=1))
        except Exception as e: errors.append(e)
    a=Thread(target=first); b=Thread(target=second); a.start(); assert entered.wait(2); b.start()
    time.sleep(.03); assert not result; release.set(); a.join(3); b.join(3)
    assert not errors and not a.is_alive() and not b.is_alive()
    assert [r.executed_count for r in result]==[1,0] and good['runner']==1 and not bad
    assert len(dispatch)==1 and batch._scope.context.contract.allowed_capabilities==()
    denial=[p for k,p in sink.events if k==TOOL_ERROR]; assert denial and denial[-1]['reason']
    assert old.authorize('same').allowed
    COUNTS['concurrent_cases']+=1

def test_late_result_captures_old_scope_not_new_context(monkeypatch):
    # QC v2: a single in-flight raw callback, not a second ToolRunnerAdapter
    # invocation after the deadline. Product timeout, wait(3), and all original
    # scope/result assertions are unchanged. Additional ordering asserts below.
    from pathlib import Path
    from intelligence.services.episode_scope import TOOL_PRE_EXECUTE
    _,ctx=fixture(); entered=Event(); release=Event(); count=Counter(); deliveries=[]
    phases=[]; trace=[]; start=time.monotonic()
    def record(phase):
        phases.append(phase)
        trace.append({'phase':phase, 'elapsed_ms':round((time.monotonic()-start)*1000,3)})
    orig_emit=EpisodeScope.emit
    def observed(self,kind,payload):
        deliveries.append((self,kind,dict(payload)))
        if kind == TOOL_RESULT: record('old_result_emitted')
        orig_emit(self,kind,payload)
    monkeypatch.setattr(EpisodeScope,'emit',observed)
    def slow(q,tc):
        assert not tc.deadline.expired
        record('raw_runner_entered')
        entered.set(); assert release.wait(3)
        assert tc.deadline.expired
        count['runner'] += 1
        record('raw_runner_returning')
        return ([AgentEvidence(tool='same',title='temporary',detail='fixture '+str(q),source='local',source_date='2026-07-24')],
            'temporary observation', ProviderTrace(provider='fixture', capability='finance_query', status='success'))
    reg=ResearchToolRegistry((spec(count,runner=slow),)); old,batch=session(ctx,reg,Sink())
    short=replace(ctx,deadline=ResearchDeadline.from_timeout(.08))
    try:
        with ThreadPoolExecutor(max_workers=1) as executor:
            batch._executor=executor
            try:
                timed=batch.execute(call('late'),registry=reg,context=short,remaining_slots=1)
                assert entered.is_set() and timed.items[0].status=='timeout'
                record('batch_timeout_returned')
                assert timed.items[0].error == 'tool_timeout' and timed.executed_count == 1
                assert count['runner'] == 0 and not release.is_set()
                assert not [s for s,k,p in deliveries if k == TOOL_RESULT]
                execution_scope=batch._scope
                fresh=restrict(replace(ctx,deadline=ResearchDeadline.from_timeout(5)))
                batch.bind_scope(registry=reg.without('same'),context=fresh)
                current=batch._scope
                assert not current.authorize('same').allowed
                record('new_scope_bound')
            finally:
                release.set()
        assert count['runner']==1
        results=[s for s,k,p in deliveries if k==TOOL_RESULT]
        assert results==[execution_scope] and current is not execution_scope
        assert execution_scope.context is short and current.context is fresh
        assert current.invoked_tools is old.invoked_tools
        assert timed.items[0].observation is None and not current.allowed_tools()
        assert [s for s,k,p in deliveries if k == TOOL_PRE_EXECUTE] == [execution_scope]
        assert not [s for s,k,p in deliveries if k == TOOL_ERROR]
        result_payload = next(p for s,k,p in deliveries if k == TOOL_RESULT)
        assert result_payload['tool_call_id'] == 'late' and result_payload['evidence_count'] == 1
        assert len(result_payload['evidence_hashes']) == 1
        assert phases == ['raw_runner_entered', 'batch_timeout_returned', 'new_scope_bound',
            'raw_runner_returning', 'old_result_emitted']
        COUNTS['late_result_old_scope']+=1
    finally:
        path=Path('/tmp/e2-qc-301dcd9e-evidence')/('corrected-late-trace-'+uuid4().hex+'.json')
        path.write_text(json.dumps(trace,ensure_ascii=False,indent=2))
        print('QC_CORRECTED_LATE_TRACE', json.dumps(trace,ensure_ascii=False))

@pytest.mark.parametrize('variant',['unknown','external_or_mixed','different_capability','deleted','contract_removed','stricter_registry'])
def test_repair_variants_current_scope_denial_and_history(variant):
    frame,ctx=fixture(); count=Counter(); reg=ResearchToolRegistry((spec(count),)); states=[]; live=[]
    class Model:
        repair=False
        menus=[]
        def complete(self,*,messages,tools,timeout):
            self.menus.append(tools)
            if self.repair:
                self.repair=False
                assert 'same' not in {x['function']['name'] for x in tools}
                return ModelTurn('',call('bad-repair'),'offline-script','')
            return final(ctx)
    model=Model(); ep=ContinuousAgentEpisode(model,event_sink=live.append)
    previous=ep.run(task_frame=frame,context=ctx,registry=reg,_continuation_sink=states)
    state=states[0]; old=state.episode_scope
    old.record_invocation('historic'); old.record_derive_mismatch('historic-mismatch')
    history=(state.accumulator,state.messages,state.evidence_ledger,old.event_sink)
    if variant in ('unknown','external_or_mixed'): state.registry=ResearchToolRegistry((spec(count,variant),))
    elif variant=='different_capability': state.registry=ResearchToolRegistry((spec(count,capability='web_search'),))
    elif variant=='deleted': state.registry=reg.without('same')
    elif variant=='contract_removed': state.context=restrict(ctx)
    else: state.registry=reg.with_read_scope('material_only')
    dispatch=[]; original=state.tool_session.on_dispatch
    def record(intent): dispatch.append(intent); original(intent)
    state.tool_session.on_dispatch=record; model.repair=True
    outcome=ep.resume(state,previous,goal(ctx))
    assert not count and not dispatch
    assert not state.episode_scope.authorize('same').allowed
    assert state.episode_scope.registry is state.registry
    assert (state.accumulator,state.messages,state.evidence_ledger,state.episode_scope.event_sink)==history
    assert state.episode_scope._derive_mismatches is old._derive_mismatches
    assert state.episode_scope.invoked_tools is old.invoked_tools and old.authorize('same').allowed
    assert state.ledger.derive_mismatch_sink.__self__ is state.episode_scope
    deny=[e.payload for e in live if e.kind==TOOL_ERROR and e.payload.get('tool_call_id')=='bad-repair']
    assert len(deny)==1 and deny[0]['reason']==state.episode_scope.authorize('same').reason
    assert not [e for e in outcome.events if e.kind=='tool_request' and 'replay' in e.payload]
    COUNTS['repair_denials']+=1


def test_repair_actual_downgrade_hook_rebinds_contract_before_menu():
    frame,ctx=fixture(); count=Counter(); reg=ResearchToolRegistry((spec(count),)); states=[]
    class NarrowHarness(FinanceResearchHarness):
        def downgrade_unreachable(self,g,*,contract):
            narrowed=restrict(replace(ctx,contract=contract)).contract
            return RepairDowngrade((),narrowed,g)
    class Model:
        repair=False
        def complete(self,*,messages,tools,timeout):
            if self.repair: assert tools==[]
            return final(ctx)
    model=Model(); ep=ContinuousAgentEpisode(model,harness=NarrowHarness())
    previous=ep.run(task_frame=frame,context=ctx,registry=reg,_continuation_sink=states)
    state=states[0]; model.repair=True; ep.resume(state,previous,goal(ctx))
    assert state.context.contract is state.episode_scope.context.contract
    assert state.episode_scope.context.contract.allowed_capabilities==()
    assert not state.episode_scope.authorize('same').allowed and not count
    COUNTS['repair_hook_contract_narrow']+=1
