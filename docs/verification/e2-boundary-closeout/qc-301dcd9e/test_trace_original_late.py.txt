"""Trace original failing QC probe without changing its source or assertions."""
import json
import time
from pathlib import Path
import test_independent_scope as original
from intelligence.services import agent_research
from intelligence.services.episode_scope import EpisodeScope
from intelligence.runtime.episode_tool_batch import EpisodeToolBatchSession

def test_original_late_with_execution_trace(monkeypatch):
    trace = []
    start = time.monotonic()
    def record(event, **fields):
        trace.append(dict(event=event, elapsed_ms=round((time.monotonic()-start)*1000, 3), **fields))
    run = agent_research._run_tool
    def traced_run(runner, query, context, **kwargs):
        record('adapter_enter', runner=getattr(runner, '__qualname__', str(runner)), expired=context.deadline.expired)
        try:
            result = run(runner, query, context, **kwargs)
        except BaseException as exc:
            record('adapter_raise', error=type(exc).__name__, reason=str(exc))
            raise
        record('adapter_return')
        return result
    emit = EpisodeScope.emit
    def traced_emit(self, kind, payload):
        record('emit', scope=id(self), kind=kind, payload=dict(payload))
        return emit(self, kind, payload)
    execute = EpisodeToolBatchSession.execute
    def traced_execute(self, *args, **kwargs):
        record('batch_execute_enter')
        result = execute(self, *args, **kwargs)
        record('batch_execute_return', scope=id(self._scope), status=result.items[0].status, error=result.items[0].error)
        return result
    bind = EpisodeToolBatchSession.bind_scope
    def traced_bind(self, **kwargs):
        result = bind(self, **kwargs)
        record('bind_return', scope=id(result), allowed=list(result.allowed_tools()))
        return result
    monkeypatch.setattr(agent_research, '_run_tool', traced_run)
    monkeypatch.setattr(EpisodeScope, 'emit', traced_emit)
    monkeypatch.setattr(EpisodeToolBatchSession, 'execute', traced_execute)
    monkeypatch.setattr(EpisodeToolBatchSession, 'bind_scope', traced_bind)
    try:
        original.test_late_result_captures_old_scope_not_new_context(monkeypatch)
    finally:
        text = json.dumps(trace, ensure_ascii=False, indent=2)
        Path('/tmp/e2-qc-301dcd9e-evidence/original-late-trace.json').write_text(text)
        print('QC_ORIGINAL_LATE_TRACE', text)
