"""Independent offline launcher; app and repository tests are not edited."""
import os, sys, json
from pathlib import Path
ROOT = Path('/tmp/e2-qc-301dcd9e')
DATA = Path('/tmp/e2-qc-301dcd9e-data')
EVIDENCE = Path('/tmp/e2-qc-301dcd9e-evidence')
for key in list(os.environ):
    if any(s in key.upper() for s in ('API_KEY','TOKEN','SECRET')) or key.startswith(('FORESIGHT_', 'SUBCONSCIOUS_')):
        os.environ.pop(key, None)
for key, value in {
    'HOME': str(DATA/'home'), 'FINANCE_WS': str(DATA/'finance'),
    'FINANCE_ROOT': str(DATA/'finance'), 'KB_VAULT': str(DATA/'wiki'),
    'KNOWLEDGE_WIKI': str(DATA/'wiki'), 'MARKET_FEATURE_STORE_DB': str(DATA/'absent.duckdb'),
    'FWP_TEST_RECEIPT': '0', 'PYTHONDONTWRITEBYTECODE': '1',
    'ENTITY_ANCHOR_SECURITIES_DB': '0', 'FORESIGHT_STRICT_DERIVATION': '1',
    'PYTEST_DISABLE_PLUGIN_AUTOLOAD': '1',
}.items(): os.environ[key] = value
for part in ('home','finance','wiki','users'): (DATA/part).mkdir(exist_ok=True, parents=True)
sys.dont_write_bytecode = True
sys.path.insert(0, str(ROOT))
sys.path.insert(0, str(EVIDENCE))
attempts = []
allowed_children = []
def audit(event, args):
    if event in ('socket.connect','socket.getaddrinfo','socket.gethostbyname','socket.sendto'):
        attempts.append({'event': event, 'detail': str(args)[:200]})
        raise RuntimeError('QC offline denied '+event)
    if event == 'subprocess.Popen':
        exe, argv, *_ = args
        # Only audit infrastructure and explicitly offline Python test children.
        if Path(str(exe)).name in ('git', 'uname'):
            allowed_children.append(str(argv)); return
        attempts.append({'event':event, 'detail':str(argv)[:200]})
        raise RuntimeError('QC subprocess denied: '+str(argv)[:100])
    if event == 'open' and isinstance(args[0], (str, bytes)):
        name = os.fsdecode(args[0])
        if name.startswith('/Users/a77/') and '/.venv-workbench/' not in name:
            attempts.append({'event': 'host_open', 'detail': name})
            raise RuntimeError('QC production read denied: '+name)
sys.addaudithook(audit)
import intelligence.userspace as userspace
userspace.USERS_DIR = DATA/'users'
import pytest
class Audit:
    @pytest.fixture(autouse=True)
    def count_original_fail_seams(self, request, monkeypatch):
        # Author test helpers throw, but may be swallowed. Count before throwing.
        module = request.module
        seen = []
        original = getattr(module, 'fail_external', None)
        if original is not None:
            def counted(*a, **kw):
                seen.append('fail_external')
                return original(*a, **kw)
            monkeypatch.setattr(module, 'fail_external', counted)
        yield
        assert not seen, ('forbidden seam attempts', request.node.nodeid, seen)
name = os.environ.get('QC_LOG_NAME', 'offline')
try:
    rc = pytest.main(['-p','no:cacheprovider', *sys.argv[1:]], plugins=[Audit()])
finally:
    (EVIDENCE/(name+'-io.json')).write_text(json.dumps({'forbidden_attempts':attempts,'allowed_infrastructure_children':allowed_children}, ensure_ascii=False, indent=2))
    print('QC_IO_ATTEMPTS', json.dumps(attempts, ensure_ascii=False))
sys.exit(rc if not attempts else 90)
