"""One bounded offline diagnosis, not a resumed review or natural-acceptance batch."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

ROOT = Path(__file__).resolve().parent
TREE = Path('/Users/a77/finance-worktrees/adaptive-research-loop')
PYTHON = Path('/Users/a77/finance-workspace-private/.venv-workbench/bin/python')
KB = Path('/Users/a77/knowledge-base-private')
MODEL = Path('/Users/a77/.cache/huggingface/hub/models--BAAI--bge-m3/snapshots/5617a9f61b028005a4858fdac845db406aefb181')
PATHS = ['intelligence/services/llm_http_transport.py', 'scripts/review_probes/diagnose_llm_timeout.py', 'scripts/review_probes/prepare_adaptive_l6.py', 'scripts/review_probes/retrieval_readiness_worker.py']


def dump(name, value):
    with (ROOT / name).open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')


def identity():
    return {
        'revision': subprocess.check_output(['git', '-C', str(TREE), 'rev-parse', 'HEAD'], text=True).strip(),
        'status': subprocess.check_output(['git', '-C', str(TREE), 'status', '--porcelain'], text=True),
        'source_sha256': {name: hashlib.sha256((TREE / name).read_bytes()).hexdigest() for name in PATHS},
        'loadavg': os.getloadavg(),
    }


def main():
    before = identity()
    dump('protocol.json', {
        'scope': 'offline diagnosis only; dirty source hashes bound; no acceptance claim',
        'source_before': before, 'pid': os.getpid(),
        'strict': {'invocations': 1, 'timeout_seconds': 0.8, 'tolerance_seconds': 0.2, 'controller_limit_seconds': 45},
        'retrieval': {'invocations': 1, 'limit_seconds': 120, 'offline': True},
        'real_model_requests': 0, 'retries': 0, 'natural_questions': 0,
        'independent_diagnostics': True,
    })
    command = [str(PYTHON), '-m', 'scripts.review_probes.diagnose_llm_timeout', '--assert-deadline', '--output', str(ROOT / 'strict.json')]
    started = time.monotonic()
    with (ROOT / 'strict.log').open('xb') as log:
        try:
            result = subprocess.run(command, cwd=TREE, stdout=log, stderr=subprocess.STDOUT, timeout=45)
            code = result.returncode
        except subprocess.TimeoutExpired:
            code = 124
    dump('strict-controller.json', {'command': command, 'exit_code': code, 'elapsed_seconds': time.monotonic() - started})
    sys.path.insert(0, str(TREE))
    from scripts.review_probes.prepare_adaptive_l6 import retrieval_probe
    retrieval = retrieval_probe(KB / '.rag_venv/bin/python', KB, MODEL, ROOT)
    after = identity()
    dump('closure.json', {'strict_exit_code': code, 'retrieval_ready': retrieval['ready'], 'source_after': after,
                         'source_hashes_unchanged': before['source_sha256'] == after['source_sha256'],
                         'real_model_requests': 0, 'natural_questions': 0, 'retries': 0,
                         'acceptance': 'NOT_EXERCISED'})


if __name__ == '__main__':
    main()
