RESULTS = []
FAILED = []


def record(name, ok, detail):
    RESULTS.append((name, ok, detail))
    print(("PASS " if ok else "FAIL ") + name + " :: " + detail)
    if not ok:
        FAILED.append(name)


def pick_port():
    for p in range(26001, 26009):
        try:
            srv = ThreadingHTTPServer(("127.0.0.1", p), Handler)
            return srv
        except OSError:
            continue
    raise RuntimeError("no port in 26001-26008 free")


class Handler(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"
    mode = "ok"

    def log_message(self, *a):
        pass

    def do_POST(self):
        mode = self.server.serve_mode
        self.rfile.read(int(self.headers.get("Content-Length", 0)))
        if mode == "delay_headers":
            time.sleep(1.5)
            self.send_response(200)
            self.send_header("Content-Length", "2")
            self.end_headers()
            self.wfile.write(b"{}")
            return
        body = b'{"ok": true}'
        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        if mode == "ok":
            self.wfile.write(body)
        elif mode == "trickle":
            for _ in range(10):
                time.sleep(0.3)
                self.wfile.write(b"x" * 10)
                self.wfile.flush()
        elif mode == "delay_body":
            time.sleep(2.9)
            self.wfile.write(body)


def request(srv, body=b"{}"):
    import urllib.request
    return urllib.request.Request(
        "http://127.0.0.1:%d/v1/chat/completions" % srv.server_address[1],
        data=body, headers={"Content-Type": "application/json"}, method="POST",
    )


def run_subcases():
    # A: 正向成功（SSE/普通 JSON 全量读取）
    srv = pick_port(); srv.serve_mode = "ok"; threading.Thread(target=srv.serveforever, daemon=True).start()
    try:
        resp = T.urlopen(request(srv), 5.0)
        data = resp.read()
        record("A_success_read", data == b'{"ok": true}' and resp.status == 200,
               "status=%d body=%r" % (resp.status, data[:20]))
        resp.close()
    except Exception as e:
        record("A_success_read", False, "unexpected %r" % e)
    srv.shutdown()

    # B: trickle body 超过绝对 deadline（覆盖 C1 流式读取包含在 deadline 内；C3 判官晚到载荷模式）
    srv = pick_port(); srv.serve_mode = "trickle"; threading.Thread(target=srv.serveforever, daemon=True).start()
    t0 = time.monotonic(); reached = "none"
    try:
        resp = T.urlopen(request(srv), 1.2)
        chunks = 0
        for _ in resp:
            chunks += 1
            reached = "streaming"
        record("B_trickle_deadline", False, "completed without deadline trigger")
    except T.HTTPDeadlineExceeded:
        elapsed = time.monotonic() - t0
        record("B_trickle_deadline", reached == "streaming" and elapsed < 2.0,
               "trigger=%s elapsed=%.2f (deadline=1.2)" % (reached, elapsed))
    except Exception as e:
        record("B_trickle_deadline", False, "wrong exc %r reached=%s" % (e, reached))
    srv.shutdown()

    # C: 流中取消（is_cancelled），需证明已进入流式阶段后取消
    srv = pick_port(); srv.serve_mode = "trickle"; threading.Thread(target=srv.serveforever, daemon=True).start()
    box = {"cancelled": False, "chunks": 0}
    def is_cancelled():
        return box["cancelled"]
    try:
        resp = T.urlopen(request(srv), 5.0, is_cancelled=is_cancelled)
        for _ in resp:
            box["chunks"] += 1
            if box["chunks"] >= 2:
                box["cancelled"] = True
        record("C_stream_cancel", False, "completed without cancellation")
    except T.HTTPStreamCancelled:
        record("C_stream_cancel", box["chunks"] >= 2 and box["cancelled"],
               "cancelled after %d chunks, trigger reached=%s" % (box["chunks"], box["chunks"] >= 1))
    except Exception as e:
        record("C_stream_cancel", False, "wrong exc %r chunks=%d" % (e, box["chunks"]))
    srv.shutdown()

    # D: timeout=10 与共享 deadline=0.5 的区分（C4 传输层边界）
    srv = pick_port(); srv.serve_mode = "delay_body"; threading.Thread(target=srv.serveforever, daemon=True).start()
    t0 = time.monotonic()
    try:
        resp = T.urlopen(request(srv), 10.0, deadline=T.DeadlineLike if False else None)
        resp.read()
        record("D_shared_deadline", False, "completed")
    except Exception as e:
        record("D_shared_deadline", False, "unexpected %r (missing deadline obj)" % e)
    # 真正用共享 Deadline：借 llm_refine.Deadline
    from intelligence.services.llm_refine import Deadline
    t0 = time.monotonic()
    try:
        resp = T.urlopen(request(srv), 10.0, deadline=Deadline.from_timeout(0.5))
        resp.read()
        record("D_shared_deadline", False, "completed against 2.9s server")
    except T.HTTPDeadlineExceeded:
        elapsed = time.monotonic() - t0
        record("D_shared_deadline", elapsed < 1.5,
               "stopped at %.2fs (timeout=10, shared=0.5, server=2.9)" % elapsed)
    except Exception as e:
        record("D_shared_deadline", False, "wrong exc %r" % e)
    srv.shutdown()

    # E: 停滞父进程——父进程不读 stdout，worker Timer 到点 os._exit(124)
    srv = pick_port(); srv.serve_mode = "delay_body"; threading.Thread(target=srv.serveforever, daemon=True).start()
    try:
        resp = T.urlopen(request(srv), 3.0)
        time.sleep(4.0)  # 父进程停滞，超过 deadline
        rc = resp.process.poll()
        record("E_stalled_parent", rc == 124,
               "worker self-exit returncode=%r (expect 124)" % rc)
        resp.close()
    except Exception as e:
        record("E_stalled_parent", False, "unexpected %r" % e)
    srv.shutdown()

    # F: 已过期 deadline —— 不应 spawn 子进程
    from intelligence.services.llm_refine import Deadline as D2
    events = []
    try:
        T.urlopen(request(srv), 10.0, deadline=D2.from_timeout(0.0), observer=lambda n, **k: events.append(n))
        record("F_expired_no_spawn", False, "no exception")
    except T.HTTPDeadlineExceeded:
        record("F_expired_no_spawn", "spawn_started" not in events,
               "events=%r (no spawn)" % events)
    except Exception as e:
        record("F_expired_no_spawn", False, "wrong exc %r" % e)
    srv.shutdown()

    # G: C6 子进程启动计入预算 —— 0.6s 预算 vs 1.5s 才返回 header
    srv = pick_port(); srv.serve_mode = "delay_headers"; threading.Thread(target=srv.serveforever, daemon=True).start()
    events = []
    t0 = time.monotonic()
    try:
        resp = T.urlopen(request(srv), 0.6, observer=lambda n, **k: events.append(n))
        resp.read()
        record("G_startup_in_budget", False, "completed under 0.6s against 1.5s server")
    except T.HTTPDeadlineExceeded:
        elapsed = time.monotonic() - t0
        record("G_startup_in_budget", "spawn_started" in events and "request_sent" in events and elapsed < 1.2,
               "events=%r elapsed=%.2f (spawn+send inside 0.6s budget)" % (events, elapsed))
    except Exception as e:
        record("G_startup_in_budget", False, "wrong exc %r" % e)
    srv.shutdown()


if __name__ == "__main__":
    run_subcases()
