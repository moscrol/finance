"""Probe 1 v2 — C1/C6 (corrected T6a: use measured timeout slice, not timeout=0).

Subcases:
  T1 fast success        : full body + observer events (positive control).
  T2 trickle deadline    : HTTPDeadlineExceeded AFTER headers (mid-body trigger).
  T3 cancelled stream    : HTTPStreamCancelled after cancellation flips mid-stream.
  T4 stalled parent      : parent stops reading; worker timer exit 124; close reaps.
  T5 startup in budget   : 0.01s deadline with fast server fails before request.
  T6a timeout slice 0.4s, deadline +10s -> fail ~0.4s.
  T6b timeout 10s, deadline +0.5s      -> fail ~0.5s (C4 boundary at transport).
Exit nonzero on any failure.
"""
import os
import sys
import time
import urllib.request

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.abspath(os.path.join(HERE, "..", "..", "..", "..", "..", "candidate"))
sys.path.insert(0, os.path.join(ROOT, "intelligence"))
sys.path.insert(0, HERE)

import fake_server
from services import llm_http_transport as T
from services.llm_refine import Deadline

P_FAST, P_TRICK = 26001, 26002
results = []


def req(port):
    return urllib.request.Request(
        f"http://127.0.0.1:{port}/v1/chat/completions",
        data=b"{}", headers={"Content-Type": "application/json"}, method="POST")


def record(name, ok, detail):
    results.append((name, ok, detail))
    print(f"[{'PASS' if ok else 'FAIL'}] {name}: {detail}", flush=True)


def main():
    fast = fake_server.start(P_FAST, "fast_json")
    trickle = fake_server.start(P_TRICK, "trickle_sse", interval=0.05)

    events = []
    try:
        t0 = time.monotonic()
        with T.urlopen(req(P_FAST), 5, deadline=Deadline.from_timeout(5),
                       observer=lambda e, **k: events.append(e)) as r:
            body = r.read()
        ok = bool(body) and all(e in events for e in ("spawn_started", "spawned", "request_sent", "headers", "eof"))
        record("T1_fast_success", ok, f"events={events} bytes={len(body)} elapsed={time.monotonic()-t0:.2f}s")
    except Exception as e:
        record("T1_fast_success", False, f"unexpected {type(e).__name__}: {e}")

    events = []
    try:
        with T.urlopen(req(P_TRICK), 30, deadline=Deadline.from_timeout(0.6),
                       observer=lambda e, **k: events.append(e)) as r:
            r.read()
        record("T2_trickle_deadline", False, "stream completed unexpectedly")
    except T.HTTPDeadlineExceeded:
        record("T2_trickle_deadline", "headers" in events, f"HTTPDeadlineExceeded events={events}")
    except Exception as e:
        record("T2_trickle_deadline", False, f"wrong exc {type(e).__name__}: {e} events={events}")

    state = {"cancel": False}
    events = []
    try:
        with T.urlopen(req(P_TRICK), 30, deadline=Deadline.from_timeout(30),
                       is_cancelled=lambda: state["cancel"],
                       observer=lambda e, **k: events.append(e)) as r:
            r.open()
            for line in r:
                state["cancel"] = True  # flip mid-stream after first line
        record("T3_cancelled_stream", False, "completed without cancellation")
    except T.HTTPStreamCancelled:
        record("T3_cancelled_stream", "headers" in events, f"HTTPStreamCancelled events={events}")
    except Exception as e:
        record("T3_cancelled_stream", False, f"wrong exc {type(e).__name__}: {e} events={events}")

    proc = {}
    try:
        t0 = time.monotonic()
        r = T.urlopen(req(P_TRICK), 30, deadline=Deadline.from_timeout(0.7))
        proc["p"] = r.process
        r.open()
        time.sleep(1.5)  # parent stalled past deadline
        r.close()
        elapsed = time.monotonic() - t0
        rc = proc["p"].poll()
        record("T4_stalled_parent_reap", rc is not None and elapsed < 3.5,
               f"elapsed={elapsed:.2f}s returncode={rc} (124 expected: timer os._exit)")
    except Exception as e:
        record("T4_stalled_parent_reap", False, f"unexpected {type(e).__name__}: {e}")
        if "p" in proc:
            try:
                proc["p"].kill()
            except Exception:
                pass

    cons = fast.stats["connections"]
    try:
        with T.urlopen(req(P_FAST), 5, deadline=Deadline.from_timeout(0.01)) as r:
            r.read()
        record("T5_startup_in_budget", False, "completed despite 0.01s budget")
    except T.HTTPDeadlineExceeded:
        new_cons = fast.stats["connections"] - cons
        record("T5_startup_in_budget", new_cons == 0,
               f"HTTPDeadlineExceeded before request (new_connections={new_cons}) => spawn counted in budget")
    except Exception as e:
        record("T5_startup_in_budget", False, f"wrong exc {type(e).__name__}: {e}")

    try:
        t0 = time.monotonic()
        with T.urlopen(req(P_TRICK), 0.4, deadline=Deadline.from_timeout(10)) as r:
            r.read()
        record("T6a_timeout_slice", False, "completed despite 0.4s slice")
    except T.HTTPDeadlineExceeded:
        dt = time.monotonic() - t0
        record("T6a_timeout_slice", dt < 2.0, f"fail at {dt:.2f}s with slice=0.4s")
    except Exception as e:
        record("T6a_timeout_slice", False, f"wrong exc {type(e).__name__}: {e}")

    try:
        t0 = time.monotonic()
        with T.urlopen(req(P_TRICK), 10, deadline=Deadline.from_timeout(0.5)) as r:
            r.read()
        record("T6b_deadline_wins", False, "completed despite 0.5s deadline")
    except T.HTTPDeadlineExceeded:
        dt = time.monotonic() - t0
        record("T6b_deadline_wins", dt < 2.0, f"fail at {dt:.2f}s; 10s slice did not extend 0.5s deadline")
    except Exception as e:
        record("T6b_deadline_wins", False, f"wrong exc {type(e).__name__}: {e}")

    fast.shutdown(); trickle.shutdown()
    fails = [n for n, ok, _ in results if not ok]
    print(f"SUBCASES total={len(results)} failed={len(fails)} {fails}")
    sys.exit(1 if fails else 0)


if __name__ == "__main__":
    main()
