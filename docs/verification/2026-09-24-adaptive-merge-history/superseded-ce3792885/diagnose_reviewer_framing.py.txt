"""Host reproduction of reviewer fixture errors; never amends their verdict."""
import hashlib
import json
import os
from pathlib import Path
import subprocess

ROOT=Path(__file__).resolve().parent
OUT=ROOT/'reviewer-framing'
OUT.mkdir(exist_ok=False)
SOURCE=Path('/Users/a77/.finance-runtime/reviews/pr868-glm-qc-20260924-0024/quality/work/probes/probe_c1_transport_v2.py')
raw=SOURCE.read_bytes()
text=raw.decode()
anchor='        self.send_header("Content-Type", "application/json")\n        self.send_header("Content-Length", str(len(body)))'
assert text.count(anchor)==1
length_fixed=text.replace(anchor,'        self.send_header("Content-Type", "application/json")\n        self.send_header("Content-Length", str(100 if mode == "trickle" else len(body)))')
assert length_fixed.count('self.wfile.write(b"x" * 10)')==1
framing_fixed=length_fixed.replace('self.wfile.write(b"x" * 10)','self.wfile.write(b"x" * 9 + b"\\n")')
rows=[]
env={k:os.environ[k] for k in ('HOME','PATH','LANG') if k in os.environ}
env['PYTHONDONTWRITEBYTECODE']='1'
for label,body in [('original',text),('length_only',length_fixed),('length_and_lines',framing_fixed)]:
 path=OUT/(label+'.py')
 path.write_text(body)
 with (OUT/(label+'.log.txt')).open('xb') as f:
  proc=subprocess.run(['/Users/a77/finance-workspace-private/.venv-workbench/bin/python','-B',str(path)],env=env,stdout=f,stderr=subprocess.STDOUT,timeout=35)
 rows.append({'case':label,'exit_code':proc.returncode,'source_sha256':hashlib.sha256(path.read_bytes()).hexdigest()})
assert SOURCE.read_bytes()==raw
record={'scope':'host fixture diagnosis, NOT independent QC; old CHANGES_REQUIRED retained','reviewer_source':str(SOURCE),'original_sha256':hashlib.sha256(raw).hexdigest(),'results':rows,'old_source_unchanged':True,'model_requests':0,'runtime_edited':False}
(OUT/'result.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps(record))
assert [r['exit_code'] for r in rows]==[1,1,0]
