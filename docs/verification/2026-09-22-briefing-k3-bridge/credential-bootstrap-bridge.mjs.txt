import childProcess from "node:child_process";
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { createHash } from "node:crypto";

const ROOT = path.dirname(fileURLToPath(import.meta.url));
const BRIDGE_DIR = "/Users/a77/devin-sub2api-bridge";
const MAX_REQUESTS = 40;
const BUDGET_SECONDS = 1200;

export function bootstrap(pi) {
  const out = fs.realpathSync(process.env.REVIEW_OUTPUT_DIR || "/invalid-output");
  if (!out.startsWith(ROOT + path.sep)) throw new Error("output is outside review root");
  const recordPath = path.join(out, "credential.json");
  const started = Date.now();
  const record = {
    status: "RESERVED",
    started_at: new Date(started).toISOString(),
    provider: "mirasim-kimi-bridge",
    model: "kimi-k3",
    route: "http://127.0.0.1:18790/v1 -> http://127.0.0.1:8080",
    cached_only_in_process_memory: true,
    prices_verified: false,
    model_requests_admitted_at_bootstrap: 0,
  };
  fs.writeFileSync(recordPath, JSON.stringify(record, null, 2) + "\n", { flag: "wx", mode: 0o600 });
  try {
    const token = childProcess.execFileSync("/bin/zsh", ["-lc",
      `cd ${BRIDGE_DIR} && set -a && . ./.env && set +a && printf '%s' "$BRIDGE_KEY"`,
    ], {
      encoding: "utf8",
      timeout: 35000,
      maxBuffer: 65536,
      stdio: ["ignore", "pipe", "pipe"],
      env: { HOME: "/Users/a77", PATH: "/opt/homebrew/bin:/usr/bin:/bin", LANG: "C.UTF-8" },
    }).trim();
    if (!token) throw new Error("bridge credential is empty");
    Object.assign(record, {
      status: "READY",
      route: "http://127.0.0.1:18790/v1 -> http://127.0.0.1:8080",
      credential_sha256: createHash("sha256").update(token).digest("hex"),
    });
    fs.writeFileSync(recordPath, JSON.stringify(record, null, 2) + "\n");
    pi.registerProvider("mirasim-kimi-bridge", {
      baseUrl: "http://127.0.0.1:18790/v1",
      api: "openai-completions",
      apiKey: token,
      models: [{
        id: "kimi-k3",
        name: "Kimi K3 via local bridge",
        reasoning: true,
        thinkingLevelMap: { xhigh: "xhigh", max: "max" },
        input: ["text"],
        contextWindow: 200000,
        maxTokens: 16384,
        cost: { input: 0, output: 0, cacheRead: 0, cacheWrite: 0 },
        compat: { supportsDeveloperRole: false, supportsReasoningEffort: true },
      }],
    });
    let admitted = 0;
    pi.on("before_provider_request", (event, ctx) => {
      try {
        if (ctx.model?.provider !== "mirasim-kimi-bridge" || ctx.model?.id !== "kimi-k3" ||
            event.payload?.model !== "kimi-k3" || admitted >= MAX_REQUESTS ||
            Date.now() - started >= BUDGET_SECONDS * 1000) {
          throw new Error("model request admission denied");
        }
        fs.appendFileSync(path.join(out, "request-admissions.jsonl"), JSON.stringify({
          admission: admitted + 1,
          timestamp: new Date().toISOString(),
          provider: "mirasim-kimi-bridge",
          model: "kimi-k3",
          route: "127.0.0.1:18790",
        }) + "\n", { mode: 0o600 });
        admitted += 1;
      } catch {
        process.stderr.write("Review request admission blocked; no retry or fallback.\n");
        process.exit(75);
      }
    });
  } catch (error) {
    Object.assign(record, {
      status: "BLOCKED",
      error_type: error?.name || "Error",
      resolver_exit: Number.isInteger(error?.status) ? error.status : null,
      resolver_signal: error?.signal || null,
      stderr_sha256: error?.stderr ? createHash("sha256").update(error.stderr).digest("hex") : null,
    });
    fs.writeFileSync(recordPath, JSON.stringify(record, null, 2) + "\n");
    throw new Error("bridge credential bootstrap blocked; details intentionally omit credentials");
  }
}

export default function (pi) {
  try {
    bootstrap(pi);
  } catch {
    process.stderr.write("Review bridge credential bootstrap failed before model dispatch.\n");
    process.exit(70);
  }
}
