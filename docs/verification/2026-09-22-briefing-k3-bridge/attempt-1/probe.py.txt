def build_labels_db(path: Path, rows, with_gap=False) -> None:
    con = duckdb.connect(str(path))
    try:
        con.execute("""CREATE TABLE history_teaching_labels (
            entity_type VARCHAR NOT NULL, entity_id VARCHAR NOT NULL, trade_date DATE NOT NULL,
            label VARCHAR NOT NULL, value_num DOUBLE, value_text VARCHAR,
            label_version VARCHAR NOT NULL, framework_version VARCHAR NOT NULL,
            status VARCHAR NOT NULL DEFAULT 'ok', status_reason VARCHAR, computed_at TIMESTAMP NOT NULL,
            PRIMARY KEY (entity_type, entity_id, trade_date, label))""")
        con.execute("""CREATE TABLE history_teaching_gaps (trade_date DATE NOT NULL, gap_kind VARCHAR NOT NULL,
            missing_cols VARCHAR, framework_version VARCHAR NOT NULL, status VARCHAR NOT NULL DEFAULT 'gap',
            status_reason VARCHAR, computed_at TIMESTAMP NOT NULL, PRIMARY KEY (trade_date, gap_kind))""")
        for r in rows:
            con.execute("INSERT INTO history_teaching_labels VALUES (?,?,?,?,?,?,?,?,?,?,?)", list(r))
        if with_gap:
            con.execute("INSERT INTO history_teaching_gaps VALUES (?, 'tf.narrative_events', NULL, 'fw-test', 'gap', 'narrative_source_missing', ?)", [LANDING, "2026-09-18 15:30:00"])
    finally:
        con.close()


def default_rows(computed_at="2026-09-18 15:30:00"):
    rows = [STAGE_ROW]
    for field in sorted(EXPECT):
        rows.append(briefing_row(f"tf.{field}", float(EXPECT[field]) if not isinstance(EXPECT[field], float) else EXPECT[field], computed_at))
    for field in MERGE_NULL_FIELDS:  # builder writes explicit NULL rows; river object drops them
        rows.append(briefing_row(f"tf.{field}", None, computed_at))
    return rows


def build_wiki(root: Path, projection_text: str) -> None:
    p = root / "raw/theme-radar/opinion-store/briefing-tier-events.jsonl"
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(projection_text, encoding="utf-8")


def run_case(name: str, tmp: Path, market: Path, wiki: Path, labels: Path, date: str, expect: set[int]) -> int:
    argv = ["verify", "--db-path", str(market), "--labels-db", str(labels),
            "--kb-wiki", str(wiki), "--briefing-date", date, "--entity", "测试板块"]
    old = sys.argv
    sys.argv = argv
    try:
        code = main()
    finally:
        sys.argv = old
    verdict = "OK" if code in expect else "UNEXPECTED"
    print(f"===== CASE {name}: exit={code} expect={sorted(expect)} -> {verdict}", file=sys.stderr)
    return code


def main_probe() -> int:
    import shutil
    tmp = Path("/Users/a77/.finance-runtime/reviews/briefing-k3-bridge-20260922/tmp-finance/probe")
    shutil.rmtree(tmp, ignore_errors=True)
    tmp.mkdir(parents=True, exist_ok=True)
    failures = []

    def check(name, got, want):
        ok = got in want
        print(f"[{'PASS' if ok else 'FAIL'}] {name}: exit={got} expected={sorted(want)}")
        if not ok:
            failures.append(name)

    # ---- positive baseline
    market = tmp / "market.duckdb"; labels = tmp / "labels.duckdb"; wiki = tmp / "wiki"
    build_market_db(market)
    build_wiki(wiki, json.dumps(PROJECTION_ROW, ensure_ascii=False) + "\n")
    build_labels_db(labels, default_rows(), with_gap=True)
    check("positive_full_consumption", run_case("positive", tmp, market, wiki, labels, "2026-09-16", {0}), {0})

    # ---- negatives on the same positive baseline
    # 1. label value mismatch
    rows = default_rows()
    rows = [r if r[3] != "tf.briefing_tier1_items" else r[:4] + (9.0,) + r[5:] for r in rows]
    build_labels_db(tmp / "l1.duckdb", rows)
    check("neg_label_value_mismatch", run_case("neg_mismatch", tmp, market, wiki, tmp / "l1.duckdb", "2026-09-16", {1}), {1})

    # 2. missing label row (table exists, one field absent)
    rows = [r for r in default_rows() if r[3] != "tf.briefing_lag_days"]
    build_labels_db(tmp / "l2.duckdb", rows)
    check("neg_missing_label_row", run_case("neg_missing", tmp, market, wiki, tmp / "l2.duckdb", "2026-09-16", {1}), {1})

    # 3. duplicate labels (dup rows in a nos-PK table must not silently PASS)
    con = duckdb.connect(str(tmp / "l3.duckdb"))
    con.execute("CREATE TABLE history_teaching_labels (entity_type VARCHAR, entity_id VARCHAR, trade_date DATE, label VARCHAR, value_num DOUBLE, value_text VARCHAR, label_version VARCHAR, framework_version VARCHAR, status VARCHAR, status_reason VARCHAR, computed_at TIMESTAMP)")
    for r in default_rows():
        con.execute("INSERT INTO history_teaching_labels VALUES (?,?,?,?,?,?,?,?,?,?,?)", list(r))
    # duplicate one label with the SAME value (weak dup) and one with a DIFFERENT value (strong dup)
    dup_same = briefing_row("tf.briefing_tier2_items", 0.0, "2026-09-18 15:30:00")
    dup_diff = briefing_row("tf.briefing_tier3_items", 7.0, "2026-09-18 15:30:00")
    con.execute("INSERT INTO history_teaching_labels VALUES (?,?,?,?,?,?,?,?,?,?,?)", list(dup_same))
    con.execute("INSERT INTO history_teaching_labels VALUES (?,?,?,?,?,?,?,?,?,?,?)", list(dup_diff))
    con.close()
    check("neg_duplicate_labels", run_case("neg_dup", tmp, market, wiki, tmp / "l3.duckdb", "2026-09-16", {1}), {1})

    # 4. NULL value_num where expected value is 0 (NULL must not equal 0)
    rows = default_rows()
    rows = [r if r[3] != "tf.briefing_market_confirmed" else r[:4] + (None,) + r[5:] for r in rows]
    build_labels_db(tmp / "l4.duckdb", rows)
    check("neg_null_not_zero", run_case("neg_null", tmp, market, wiki, tmp / "l4.duckdb", "2026-09-16", {1}), {1})

    # 5. late teaching objects must be filtered in strict slice (positive: live shows them,
    #    strict drops them; verify still passes because the briefing row itself is on time).
    late = default_rows(computed_at="2026-09-18 15:30:00")
    late.append(("market", "market", LANDING, "tf.dragon_count", 3.0, None, "v1", "fw-test", "ok", None, "2026-09-20 09:00:00"))
    build_labels_db(tmp / "l5.duckdb", late, with_gap=True)
    check("pos_late_teaching_filtered", run_case("pos_late", tmp, market, wiki, tmp / "l5.duckdb", "2026-09-16", {0}), {0})

    # 6. briefing label rows computed_at AFTER the landing day while projection recorded_at is on time
    #    => object recorded_at > landing; verify's backdate guard should FAIL (river object would
    #    leak through strict filter otherwise).
    backdated = default_rows(computed_at="2026-09-21 09:00:00")
    backdated[0] = STAGE_ROW
    build_labels_db(tmp / "l6.duckdb", backdated)
    check("neg_backdated_labels", run_case("neg_backdated", tmp, market, wiki, tmp / "l6.duckdb", "2026-09-16", {1}), {1})

    # 7. missing rows for a requested briefing_date in projection
    check("neg_no_projection_rows", run_case("neg_norows", tmp, market, wiki, labels, "2026-09-15", {1}), {1})

    # 8. missing labels db file
    check("neg_missing_labels_db", run_case("neg_nolabels", tmp, market, wiki, tmp / "absent.duckdb", "2026-09-16", {1}), {1})

    # 9. invalid projection JSON
    badwiki = tmp / "wiki2"
    build_wiki(badwiki, "{\n")
    check("neg_invalid_projection_json", run_case("neg_badjson", tmp, market, badwiki, labels, "2026-09-16", {1}), {1})

    print("PROBE_FAILURES:", failures if failures else "none")
    return 1 if failures else 0


if __name__ == "__main__":
    raise SystemExit(main_probe())
