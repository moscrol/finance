def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git(tree, *args):
    return subprocess.check_output(["git", "-C", str(tree), *args])


def save(path, data):
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n")


receipts = {}
for folder in ("finance-review", "kb-review", "finance-review-2"):
    output = ROOT / folder
    receipt = json.loads((output / "execution.json").read_text())
    assert receipt["complete"] and receipt["all_candidate_and_author_trees_unchanged"]
    assert receipt["inputs_unchanged"] and receipt["model_errors"] == []
    for name, expected in receipt["inputs_sha256"].items():
        assert digest(ROOT / name) == expected, name
    for name, expected in receipt["sha256"].items():
        assert digest(output / name) == expected, (folder, name)
    for protected in (receipt["pid"],):
        try:
            import os
            os.kill(protected, 0)
        except ProcessLookupError:
            pass
        else:
            raise RuntimeError(f"Review process is still present: {protected}")
    verdict_path = output / "verdict.json"
    verdict = json.loads(verdict_path.read_text()) if verdict_path.exists() else None
    receipts[folder] = {

for axis, destination in DESTINATIONS.items():
    target = destination / RELPATH
    target.mkdir(parents=True, exist_ok=True)
    manifest = []
    for source, relative in common + axis_files[axis]:
        src = ROOT / source
        dst = target / relative
        dst.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(src, dst)
        manifest.append({"path": relative, "bytes": dst.stat().st_size, "sha256": digest(dst), "source": str(src)})
    save(target / "coordinator-qc.json", summary)
    manifest.append({"path": "coordinator-qc.json", "bytes": (target / "coordinator-qc.json").stat().st_size,
                     "sha256": digest(target / "coordinator-qc.json"), "source": "coordinator verification, not reviewer verdict"})
    for entry in manifest:
        data = (target / entry["path"]).read_bytes()
        assert not re.search(rb"eyJ[A-Za-z0-9_-]{12,}\.[A-Za-z0-9_-]{12,}\.[A-Za-z0-9_-]{12,}", data)
        assert not re.search(rb"-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----\r?\n[A-Za-z0-9+/=]{20,}", data)
        assert len(data) < 100000
    save(target / "manifest.json", {"axis": axis, "candidate_revision": REVISIONS[axis], "files": manifest})
    print(json.dumps({"axis": axis, "files": len(manifest), "destination": str(target), "status": summary["coordinator_acceptance"]}))
