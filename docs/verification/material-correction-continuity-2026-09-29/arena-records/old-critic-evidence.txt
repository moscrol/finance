#!/usr/bin/env python3
"""Isolated, non-production GLM q8-only accounting-basis critic pilot.

Run from the frozen 774cfbd1a repository with FORESIGHT_BUILTIN_LLM_API_KEY
and FORESIGHT_BUILTIN_LLM_BASE_URL provided by the existing runtime launcher.
One invocation makes at most one model request and refuses to overwrite output.
Never reads or writes a production user, market database or service.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import time

from openai import OpenAI

BASE = Path("docs/verification/answer-capability-2026-09-29/answers")
QUESTION = json.loads((BASE / "review-15.json").read_text())["question"]
Q8 = QUESTION[QUESTION.rfind("\n8. ") + 1:].strip()
RED = {}
for filename in ("review-15.json", "review-18.json"):
    record = json.loads((BASE / filename).read_text())
    answer = record["public_answer"]
    RED[record["id"]] = answer[answer.index("## q8"):answer.index("## 证据边界")].strip()
CASES = {
    "1": (RED["review-15"], "red"),
    "2": ("净利润不能直接扣现金资本开支；应检查15倍股权利润估值是否已合理反映未来增长、再投资回报与融资假设。", "green"),
    "3": (RED["review-18"], "red"),
    "4": ("不要问净利润是否已扣除资本开支；应核对增长、回报、融资与15倍估值假设，不能从给定股权价值机械再扣现金投资。", "green"),
}
SYSTEM = (
    "你只审阅财务口径，不判断买卖建议，也不替作者改写答案。"
    "判断待审文本是否声称、暗示或要求检查：会计净利润已经扣除或应先扣现金资本开支；"
    "或者从给定的股权价值机械再扣现金投资。这是红色错口径。"
    "正确区分：会计净利润与投资现金流是不同量；可核对利润倍数是否合理计入未来增长、"
    "再投资回报和融资条件，但这不等于从净利润直接扣现金资本开支。"
    "注意否定句：明确说‘不要问净利润是否已扣资本开支’不是红色。"
    "如果没有相关说法或无法判定，输出unknown，不能为了绿而编造。"
    "只输出一个JSON对象，字段 verdict(red/green/unknown)、quote(从待审文本逐字复制的关键短句)、reason(一句话)。"
)


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("case", choices=CASES)
    parser.add_argument("--outdir", type=Path, required=True)
    args = parser.parse_args()
    case_text, expected = CASES[args.case]
    prompt = "原第8问：\n" + Q8 + "\n\n待审文本：\n" + case_text
    key = os.environ.get("FORESIGHT_BUILTIN_LLM_API_KEY")
    base_url = os.environ.get("FORESIGHT_BUILTIN_LLM_BASE_URL")
    if not key or not base_url:
        print("blocked: runtime model credential/endpoint missing (not shown)")
        return 2
    args.outdir.mkdir(parents=True, exist_ok=True)
    output = args.outdir / ("case-" + args.case + ".json")
    if output.exists():
        print("blocked: evidence already exists")
        return 3
    messages = [{"role": "system", "content": SYSTEM}, {"role": "user", "content": prompt}]
    request = json.dumps({"model": "glm-5.3", "messages": messages, "max_tokens": 1500, "temperature": 0}, ensure_ascii=False, sort_keys=True).encode()
    data = {"case": args.case, "expected": expected, "request_sha256": hashlib.sha256(request).hexdigest(), "question_sha256": hashlib.sha256(QUESTION.encode()).hexdigest(), "case_text_sha256": hashlib.sha256(case_text.encode()).hexdigest(), "model_requested": "glm-5.3"}
    try:
        client = OpenAI(api_key=key, base_url=base_url, timeout=105, max_retries=0)
        started = time.monotonic()
        resp = client.chat.completions.create(model="glm-5.3", messages=messages, max_tokens=1500, temperature=0)
        data["elapsed_seconds"] = round(time.monotonic() - started, 3)
        data["model_reported"] = resp.model
        data["response_id"] = resp.id
        data["usage"] = resp.usage.model_dump() if resp.usage else None
        raw = resp.choices[0].message.content or ""
        data["raw_response"] = raw
        data["raw_response_sha256"] = hashlib.sha256(raw.encode()).hexdigest()
        try:
            parsed = json.loads(raw)
            data["parsed"] = parsed
            data["matches_predeclared"] = parsed.get("verdict") == expected and isinstance(parsed.get("quote"), str) and parsed["quote"] in case_text
        except (ValueError, TypeError, AttributeError):
            data["parsed"] = None
            data["matches_predeclared"] = False
    except Exception as exc:
        # Do not persist an exception string: a transport library might include
        # a credential-bearing URL/header. A failed request is not retried.
        data["error_type"] = type(exc).__name__
        data["matches_predeclared"] = False
    output.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")
    print("saved", output, "case", args.case, "reported_model", data.get("model_reported"), "verdict", (data.get("parsed") or {}).get("verdict"), "ok", data["matches_predeclared"], "error_type", data.get("error_type"))
    return 0 if data["matches_predeclared"] and data.get("model_reported") == "glm-5.3" else 1


if __name__ == "__main__":
    raise SystemExit(main())
