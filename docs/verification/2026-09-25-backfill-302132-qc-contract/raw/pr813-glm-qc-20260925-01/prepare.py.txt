"""Prepare two isolated focused reviews, reusing the bounded controller unchanged."""
import ast
import hashlib
import json
from pathlib import Path
import shutil
import subprocess

ROOT = Path(__file__).resolve().parent
OLD = ROOT.parent / 'pr813-glm-qc-20260924-02'
CAND = ROOT / 'candidate'
AUTHOR = ROOT.parent / 'backfill-302132-0923/forward-02/tree'
REV = '3c5b3c9a6f0c1fe9401c424bc5ac396cf46fbc59'
BASE = '4cc15e703f81bce8abadee00f68caacdb0c72b4d'
PY = '/Users/a77/finance-workspace-private/.venv-workbench/bin/python'
FILES = ['market_feature_store/sync/repair_backfill_stock_history.py', 'scripts/verify_302132_backfill_acceptance.py', 'tests/test_repair_backfill_stock_history.py']

def save(path, obj):
    path.write_text(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')

def git(*args):
    return subprocess.check_output(['git', '-C', str(AUTHOR), *args], text=True).strip()

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def main():
    assert not CAND.exists() and git('rev-parse', 'HEAD') == REV and not git('status', '--porcelain')
    refs = dict(line.split()[::-1] for line in git('ls-remote', 'gitea', 'refs/heads/main', 'refs/heads/fix/backfill-main-ready-0921').splitlines())
    assert refs['refs/heads/fix/backfill-main-ready-0921'] == REV
    subprocess.run(['git', '-C', str(AUTHOR), 'worktree', 'add', '--detach', str(CAND), REV], check=True)
    manifest = {'revision': REV, 'baseline': BASE, 'source': str(OLD), 'controller_api_changes': False, 'files': {}}
    for name in ('glm_credential.py', 'glm_review_shim.py', 'bridge_stage.py', 'report_packet.py'):
        shutil.copyfile(OLD / name, ROOT / name)
        manifest['files'][name] = {'source_sha256': digest(OLD / name), 'sha256': digest(ROOT / name)}
    symbols = []
    for name in FILES:
        symbols.append('\n## ' + name)
        for node in ast.walk(ast.parse((CAND / name).read_text())):
            if isinstance(node, (ast.FunctionDef, ast.ClassDef)):
                symbols.append(f'{node.name}: {node.lineno}-{node.end_lineno}')
    claims = (ROOT.parent / 'pr813-k3-qc-20260924-01/spec/inputs/claims.md').read_text()
    for axis in ('spec', 'quality'):
        dst = ROOT / axis
        work = dst / 'work'
        for name in ('work/tmp', 'work/users', 'work/probes', 'pi-config', 'inputs'):
            (dst / name).mkdir(parents=True, exist_ok=True)
        for name in ('review.mjs', 'pi_review_protocol.mjs', 'gateway.mjs', 'file_tool.py', 'provider-template.json', 'run_control.py', 'run_stage.py', 'pi-config/settings.json'):
            shutil.copyfile(OLD / 'spec' / name, dst / name)
            manifest['files'][axis + '/' + name] = {'source_sha256': digest(OLD / 'spec' / name), 'sha256': digest(dst / name)}
        for name in ('auth.json', 'models-store.json'):
            save(dst / 'pi-config' / name, {})
        for name in ('tools.sb', 'sandbox_preflight.mjs'):
            text = (OLD / 'spec' / name).read_text().replace(str(OLD / 'spec'), str(dst)).replace(str(OLD), str(ROOT))
            (dst / name).write_text(text)
        config = json.loads((OLD / 'spec/config.json').read_text())
        config.update(tree=str(CAND), author=str(AUTHOR), axis=axis, revision=REV, baseline=BASE)
        save(dst / 'config.json', config)
        claim = 'C2' if axis == 'spec' else 'C3'
        (dst / 'inputs/claims.md').write_text(f'Only {claim} is in scope for this isolated axis. Every other claim MUST be out_of_scope. Current main {refs["refs/heads/main"]} is not integrated. Full-candidate approval is not sought.\n\n' + claims)
        (dst / 'inputs/source-map.md').write_text('\n'.join(symbols) + '\n')
        focus = ('Challenge _guard source cutoff, adjusted filtering, complete/partial gap and frozen parquet hash rejection. Use _fixture and _spec helpers as synthetic setup if helpful, not their tests/assertions. Source PK is (trade_date, stock_ts_code), not adjusted: change a row to qfq rather than creating an impossible duplicate. BackfillSpec is frozen; dataclasses.replace is available. Do not test decimal arithmetic or the whole CLI.' if axis == 'spec' else 'Challenge the ACTUAL external scripts/verify_302132_backfill_acceptance.py::_data_checks (or its main acceptance entry), NOT only writer _accept or arithmetic helpers. Establish one passing baseline and assert acceptance catches changes to an outside-window row: amount, timestamp, deletion, insertion, duplicate multiplicity; include another-stock mutation if feasible. To simulate corrupt duplicate rows, a clone-only CREATE TABLE AS SELECT can deliberately remove PK constraints, but disclose this. Use tests/test_repair_backfill_stock_history.py::_build_e2e_artifacts for synthetic setup if helpful; read its exact return shape. Author fixture reuse is allowed but own mutation assertions are required. No production-copy run is authorized.')
        common = f'''Independent {axis} review of PR #813, only {claim}. Chinese concise summary. Candidate {CAND}, revision {REV}, baseline {BASE}.
Only writable root {work}; Python {PY}. Use absolute paths. Read inputs/claims.md and inputs/source-map.md under {dst} first, then precise code regions. {focus}
No git/network/live vendors/credentials/production DB/deployments/candidate writes. Synthetic DuckDB only inside work. Candidate docs and other axis denied; never nest sandbox-exec. Do not read old reviewer reports. Existing test helper construction may be reused, not its assertions or result claims; distinguish fixture reuse from author-test execution.
Limits: explore <=17 requests, execute <=17, report exactly 1; each stage600s, request/command120s. Deliver early. One direct bash command per call; no semicolons/pipelines/newline chains. Prefix commands with OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1. All probe files MUST be {work}/probes/*.py. On correction use a NEW filename under probes; NEVER rm/overwrite earlier evidence or use probes_v2/.
Call deliver_stage exactly once, ALONE: {{"result": <OBJECT>}}, not a JSON string. Omit controller identity stage/axis/revision/baseline/complete. Keep result <4500 characters. Delivery is not PASS. A proposed correction never counts as an executed passing test.
Final report required fields: verdict, claims, findings, reviewer_probe_counts, author_test_counts, positive_control, limits.
claims MUST have exactly C1..C7, each {{"id":"C#","status":"verified"|"not_verified"|"out_of_scope","evidence":"observed references or explicit missing evidence"}}. The ONLY legal statuses are verified, not_verified, out_of_scope; verified_with_limits is INVALID. Only {claim} is in scope; use not_verified when the full claim remains incomplete even if named subcases passed.
Counts schema: reviewer_probe_counts={{"executed":INT,"passed":INT,"failed":INT,"errors":INT,"skipped":INT}}. Use the LAST actually executed probe suite, matching its XML testcase count; disclose previous attempts separately in limits. author_test_counts uses the same five fields, zeros if not run. Intentional control is separate, never included in reviewer or author counts.
positive_control={{"status":"OBSERVED_EXPECTED_FAILURE"|"NOT_RUN","classification":"probe_bug","evidence":"command log reference"}}.
verdict is PASS_WITH_LIMITS, CHANGES_REQUIRED, or BLOCKED_INCOMPLETE_EVIDENCE. Concrete defects need candidate file:line, trigger and observed wrong result. A fixture failure is not a product defect. No approval outside measured scope, no merge or production permission.
'''
        explore = f'''Explore: no execution. Write ONE short self-contained pytest probe file by request 8, using the available fixture helpers to avoid rebuilding a large database fixture. Import tests by importlib.util.spec_from_file_location and register in sys.modules before exec_module. Write {work}/EXPLORE.md with exact probe and XML commands and known limits. Then deliver_stage with probe_files (existing absolute paths under probes), claims_examined, limits, next_stage_commands. Do not claim any tests ran.
'''
        execute = f'''Execute: first read {work}/phase-context/explore/parsed.json and {work}/EXPLORE.md.
Your FIRST bash command MUST be: OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 {PY} -B {work}/positive_control.py
Expected exit1 and intentional probe_bug assertion; observe and record separately, then continue. Syntax-check each probe before pytest using ast.parse (do not use py_compile, which writes caches). Run pytest with --confcutdir={work} -p no:cacheprovider --basetemp={work}/tmp/pytest --junitxml={work}/probe-results-v1.xml; use v2/v3 new paths for subsequent attempts. Never overwrite XML. No whole author suite; author tests optional and separately counted.
Fix only genuine probe errors, new file per version; unexecuted repairs remain unverified. Reserve enough requests to write {work}/EXECUTE.md and deliver_stage with probe_results, author_test_results, positive_control, findings, limits. If blocked, deliver truthful missing evidence, not imagined passes. Raw logs and XML are authoritative.
'''
        report = f'''Report: only one request, no tools except deliver_stage. The immutable packet supplies raw logs, probe code and XML. Call deliver_stage with all seven required final fields above. The claim status enum is STRICTLY verified/not_verified/out_of_scope. For {claim}, if any stated requirement is untested, status=not_verified with exact passed subset in evidence; do not invent verified_with_limits. Count actual XML cases, not planned files, and never repeat a prior-stage assertion unsupported by raw logs. If any probe error remains unresolved or the intentional control was not observed, verdict=BLOCKED_INCOMPLETE_EVIDENCE. Empty findings means no defect established within measured cases, not global absence of defects.
'''
        for stage, text in [('explore', explore), ('execute', execute), ('report', report)]:
            (dst / f'prompt-{stage}.md').write_text(common + text)
        for name in ('config.json', 'tools.sb', 'sandbox_preflight.mjs', 'prompt-explore.md', 'prompt-execute.md', 'prompt-report.md'):
            manifest['files'][axis + '/' + name] = {'sha256': digest(dst / name)}
    save(ROOT / 'prepare-manifest.json', manifest)
    save(ROOT / 'plan.json', {'revision': REV, 'baseline': BASE, 'observed_main': refs['refs/heads/main'],
        'axes': {'spec': ['C2'], 'quality': ['C3']}, 'provider': 'glm-direct-review/glm-5.3',
        'authorization': 'User said 推进; 2026-09-23 preference permits existing GLM when K3 unavailable. New bounded batch, not an automatic retry of sealed failures.',
        'request_cap': 78, 'stage_caps': {'gateway': 4, 'explore': 17, 'execute': 17, 'report': 1},
        'stage_seconds': 600, 'request_seconds': 120, 'concurrency': 1, 'automatic_retries': 0,
        'stop_axis_on_invalid_delivery': True, 'axes_independent': True, 'merge_authorized': False, 'production_authorized': False})
    print(json.dumps({'root': str(ROOT), 'revision': REV, 'request_cap': 78}))

if __name__ == '__main__':
    main()
