def sha(data):
    return hashlib.sha256(data).hexdigest()

def statuses(items):
    result = {}
    for case in items:
        ident = case.get('classname', '') + '::' + case.get('name', '')
        assert ident not in result, ident
        status = 'passed'
        for child in ('failure', 'error', 'skipped'):
            element = case.find(child)
            if element is not None:
                status = child + ':' + element.get('type', '') + ':' + element.get('message', '')
        result[ident] = status
    return result
