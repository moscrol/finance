def dump(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():

    assert subprocess.check_output(['git', '-C', str(AUTHOR), 'rev-parse', 'HEAD'], text=True).strip() == REV
    assert not subprocess.check_output(['git', '-C', str(AUTHOR), 'status', '--porcelain'], text=True).strip()

    for name in FILES:

        (inp / 'source-map.md').write_text('\n'.join(symbols) + '\n')
        for rel in ['verified-closeout.json', 'archive-verification.json', 'focused-receipt.json', 'python.command.json',
                    'python-receipts/gate-47iUwvcM/pytest.json', 'scope.command.json', 'scope.log.txt',
                    'frontend/frontend.json', 'registry-complete/receipt.json', 'rehearsal/summary.json',
                    'rehearsal/acceptance.json', 'rehearsal/amount-control.json']:
            shutil.copyfile(EVIDENCE / rel, inp / rel.replace('/', '--'))
        shutil.copyfile(DOCS / 'docs/handoffs/2026-09-24-backfill-302132-current-main-ready.md', inp / 'author-handoff.md')
        shutil.copyfile(ROOT.parent / 'backfill-302132-0923/production-authorization-draft.md', inp / 'production-authorization-draft.md')
        common = f'''You are the independent {axis} reviewer for #75, reviewing #83 / PR #813 only. You do not know the other axis result. Return Chinese summaries. Do not fix product code.

For external pytest probes use --confcutdir={dst}/work -p no:cacheprovider --basetemp={dst}/work/tmp/pytest; import modules from candidate using PYTHONPATH or sys.path.
