def git(tree, *args):
    return subprocess.check_output(['git', '-C', str(tree), *args], text=True, env={**os.environ, 'GIT_OPTIONAL_LOCKS': '0'}).strip()


def digest(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

    return {'revision': git(tree, 'rev-parse', 'HEAD'), 'status': git(tree, 'status', '--porcelain', '--untracked-files=all')}
