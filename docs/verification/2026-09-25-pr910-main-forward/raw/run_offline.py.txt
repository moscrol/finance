Never invokes a model, a review controller, a merge or a deployment. Full suites
are not approximated by these selected targets; resource observations remain separate.
"""
from datetime import datetime, timezone
import hashlib
import importlib
import json
import os
from pathlib import Path
import subprocess
import sys
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parent
PY = '/Users/a77/fwp-wt-pi-research/.venv-workbench/bin/python'
BASE = '1751e21e0fd30642e0b223604b64b30e38c46f41'
SPECS = {
    '910': {'tree': '/Users/a77/fwp-wt-pr910-main-0925', 'revision': '4ad2cb42b0c58d9b3997a435568eea5e549bccdb',
            'preview_tree': '7e9b4629af7efb6e45d66b21647a525e512b077f',
            'targets': ['tests/test_pi_review_repair.py::' + name for name in (
                'test_content_binding_rejects_malformed_or_forged_submissions',
                'test_probe_validation_checks_canonical_files_without_rewriting',
                'test_prepare_preserves_sealed_inputs_and_refuses_reuse',
                'test_prepare_fails_closed_on_changed_archive')]},
    '911': {'tree': '/Users/a77/fwp-wt-pr911-main-0925', 'revision': '86da22fda1f8fa02c71cd5d8759ee55a65e6fd37',
            'preview_tree': 'c7ceec198d114f108e9a22b726fcc00f9a418872',
            'targets': ['intelligence/tests/conformance/test_research_chain.py',
                        'intelligence/tests/test_workbench_research_chain.py']},
}
sys.path.insert(0, SPECS['910']['tree'])
run = importlib.import_module('scripts.review_probes.run_extraction_mutations')._run_logged_command
fingerprint = importlib.import_module('scripts.check_test_receipt')._fingerprint
observe = importlib.import_module('scripts.review_probes.gate_resources').observe
assert sys.executable == PY and fingerprint() == '66726d345bf37ce5'


def save(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def git(tree, *args):
    return subprocess.check_output(['git', '-C', str(tree), *args], text=True).strip()


with (ROOT / 'started.json').open('x') as stream:
    json.dump({'pid': os.getpid(), 'started_at': datetime.now(timezone.utc).isoformat(),
               'python': PY, 'script_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
               'paid_authorization': False, 'model_requests': 0}, stream)
for pr, spec in SPECS.items():
    tree = Path(spec['tree'])
    out = ROOT / pr
    out.mkdir()
    assert git(tree, 'rev-parse', 'HEAD') == spec['revision']
    assert git(tree, 'status', '--porcelain') == ''
    assert git(tree, 'rev-parse', 'HEAD^{tree}') == spec['preview_tree']
    env = {k: os.environ[k] for k in ('HOME', 'PATH', 'LANG', 'TMPDIR') if k in os.environ}
    env.update(FWP_WORKBENCH_PYTHON=PY, PYTHONDONTWRITEBYTECODE='1', PYTHONPATH=str(tree),
               FWP_TEST_RECEIPT_PATH=str(out / 'pytest.json'), FWP_TEST_RECEIPT_DIR=str(out / 'receipts'),
               FORESIGHT_LLM_KEYCHAIN='0', FORESIGHT_USERS_DIR=str(out / 'users'))
    report = {**spec, 'baseline': BASE, 'python': PY, 'dependency_fingerprint': fingerprint(),
              'steps': [], 'model_requests': 0, 'full_scope_executed': False,
              'resource_observation': observe(tree, minimum_free=20 * 1024**3)}
    save(out / 'summary.json', report)
    commands = [
        ('doctor', [PY, 'scripts/workspace.py', 'doctor']),
        ('ruff', [PY, '-m', 'ruff', 'check', '.']),
        ('diff-check', ['git', 'diff', '--check', BASE + '...HEAD']),
        ('registry-parse', [PY, 'scripts/build_registry.py', 'check-parseability']),
        ('registry-check', [PY, 'scripts/build_registry.py', 'check']),
        ('registry-tables', [PY, 'scripts/build_registry.py', 'backfill-tables', '--check']),
        ('registry-views', [PY, 'scripts/build_registry.py', 'generate-views', '--check']),
        ('ledger-crosswalk', [PY, 'scripts/audit_ledger_spec_crosswalk.py']),
        ('targeted', [PY, '-B', '-m', 'pytest', '-q', '-p', 'no:cacheprovider',
                      '--basetemp=' + str(out / 'pytest-tmp'), '--junitxml=' + str(out / 'targeted.xml'),
                      *spec['targets']]),
        ('receipt-check', [PY, 'scripts/check_test_receipt.py', str(out / 'pytest.json'),
                           '--expect-revision', spec['revision'], '--base-drift-max', '0', '--main-ref', BASE,
                           '--require-target', spec['targets'][0]]),
    ]
    for label, command in commands:
        result = run(tree, out, label, command, env, timeout=180)
        report['steps'].append({'label': label, 'exit': result['exit'], 'status': result['status']})
        save(out / 'summary.json', report)
        if result['exit'] != 0 or result['status'] != 'completed':
            raise SystemExit('failed without retry: ' + pr + '/' + label)
    cases = list(ET.parse(out / 'targeted.xml').getroot().iter('testcase'))
    report['counts'] = {'executed': len(cases), 'failures': sum(len(c.findall('failure')) for c in cases),
                        'errors': sum(len(c.findall('error')) for c in cases),
                        'skipped': sum(len(c.findall('skipped')) for c in cases)}
    report['dirty_after'] = git(tree, 'status', '--porcelain')
    assert report['dirty_after'] == ''
    report['status'] = 'SCOPED_CHECKS_COMPLETE_NOT_RELEASE_READY'
    save(out / 'summary.json', report)
    print(json.dumps({'pr': pr, 'revision': spec['revision'], 'counts': report['counts']}), flush=True)
