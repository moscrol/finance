def save(path, value):
    temporary = path.with_suffix(path.suffix + '.tmp')
    temporary.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
    temporary.replace(path)


def identity(pr):
    tree = Path(f'/Users/a77/fwp-wt-pr{pr}-main-0925')
    assert helpers.git(tree, 'rev-parse', 'HEAD') == REVISIONS[pr]
    assert not helpers.git(tree, 'status', '--porcelain')
    return tree


def step(pr, label, command, *, timeout=180, heavy=False, ports=(), receipt=None):
    tree = identity(pr)
    out = OUT / pr
    out.mkdir(exist_ok=True)
    if heavy:
        deadline = time.monotonic() + 600
        samples = []
        while True:
            sample = observe(tree, minimum_free=20 * 1024**3)
            samples.append(sample)
            save(out / f'{label}.admission.json', samples)
            if sample['admitted']:
                break
            if time.monotonic() >= deadline:
                raise RuntimeError(f'resource wait expired before {pr}/{label}')
            time.sleep(min(30, max(0, deadline - time.monotonic())))
    for port in ports:
        with socket.socket() as sock:
            sock.bind(('127.0.0.1', port))
    env = {key: os.environ[key] for key in ('PATH', 'HOME', 'LANG', 'TMPDIR') if key in os.environ}
    env.update(FWP_WORKBENCH_PYTHON=PY, PYTHONPATH=str(tree), PYTHONDONTWRITEBYTECODE='1',
               FORESIGHT_LLM_KEYCHAIN='0', FORESIGHT_USERS_DIR=str(out / 'isolated-users'),
               FINANCE_DEPLOY_LEDGER=str(out / 'deploy-ledger.jsonl'),
               FWP_TEST_RECEIPT_DIR=str(out / 'receipts'))
    if receipt:
        env['FWP_TEST_RECEIPT_PATH'] = str(receipt)
    result = helpers._run_logged_command(tree, out, label, command, env, timeout=timeout)
    identity(pr)
    row = {'pr': pr, 'label': label, 'exit': result['exit'], 'status': result['status']}
    report['steps'].append(row)
    save(OUT / 'summary.json', report)
    print(json.dumps(row), flush=True)
    return result['exit'] == 0 and result['status'] == 'completed'
