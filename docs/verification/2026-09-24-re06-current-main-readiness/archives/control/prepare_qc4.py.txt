def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()

def git(*args: str) -> str:
    return subprocess.check_output(["git", "-C", REPO, *args], text=True).strip()

def replace_text(text: str) -> str:
    for before, after in [
        (OLD_ROOT, NEW_ROOT),
        (OLD_AUTHOR, NEW_AUTHOR),
        (OLD_REV, REV),
        (OLD_BASE, BASE),
        ("qc-k3-03", "qc-k3-04"),
        ("re06-ready-refresh-0923", "re06-ready-current-0924"),
        ("baseline/re06-ready-refresh-0923", BRANCH),
    ]:
        text = text.replace(before, after)
    return text

def copy_relocated(source: Path, target: Path) -> None:
    target.parent.mkdir(parents=True, exist_ok=True)
    if source.suffix in {".py", ".mjs", ".sb", ".json", ".md", ".tsx"}:
        target.write_text(replace_text(source.read_text()))
    else:
        shutil.copy2(source, target)
