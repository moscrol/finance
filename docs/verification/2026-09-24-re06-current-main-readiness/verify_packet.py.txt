"""Verify archival bytes, identities, counts, links and terminal state, not product quality."""
import ast
import hashlib
import json
from pathlib import Path
import re
import runpy
import sys
import xml.etree.ElementTree as ET
import markdown

ROOT = Path('/Users/a77/.finance-runtime/reviews/re06-deploy-readiness-20260923-01')
REPO = Path('/Users/a77/fwp-wt-closeout-workorders-0922')
DOC = REPO / 'docs/verification/2026-09-24-re06-current-main-readiness'
helper = runpy.run_path(str(DOC / 'archive_evidence.py.txt'))
assert hashlib.sha256((DOC / 'archive_evidence.py.txt').read_bytes()).hexdigest() == json.loads((DOC / 'archive-helper-sha256.json').read_text())['sha256']
results = []
for folder in sorted((DOC / 'archives').iterdir()):
    manifest = json.loads((folder / 'manifest.json').read_text())
    count = helper['verify'](folder)
    assert manifest['revision'] == 'b027194f1039692b8c26cacf9310619d633c54b7'
    for name, item in manifest['files'].items():
        raw = helper['decode'](folder, item)
        assert raw == (Path(manifest['source_root']) / name).read_bytes(), name
        if name.endswith('.py'):
            ast.parse(raw, filename=name)
    results.append({'archive': folder.name, 'files': count,
                    'manifest_sha256': hashlib.sha256((folder / 'manifest.json').read_bytes()).hexdigest()})
def archived(package, name):
    folder = DOC / 'archives' / package
    manifest = json.loads((folder / 'manifest.json').read_text())
    return json.loads(helper['decode'](folder, manifest['files'][name]))
identity = archived('control', 'identity-03.json')
report = archived('control', 'readiness-report-0924.json')
closure = archived('control', 'closure-0924.json')
assert identity['revision'] == report['identity']['revision'] == closure['revision']
assert identity['baseline'] == '3bb81b9638f97b4773ce0f338df3a505b7c0162f'
assert report['status'] == 'NOT_READY' and not report['mergeable'] and not report['deployable']
assert report['requests'] == {'prior': 147, 'new': 0, 'total': 147, 'ceiling': 218, 'remaining': 71, 'shipping_prior_separate': 73}
assert closure['complete'] and not closure['automatic_followups']
assert closure['actual_candidate_tests'] == closure['actual_candidate_builds'] == closure['actual_model_requests'] == 0
assert not closure['push_pr_merge_deploy_production_writes'] and not closure['worktrees_deleted']
for name in ['qc-current-continuation-0924.json', 'engineering-wait-0924.json']:
    state = archived('control', name)
    assert state['complete'] and state['status'] == 'BLOCKED_RESOURCE_GATE'
    assert not any(row['allowed'] for row in state['resource_admissions'])
assert not archived('control', 'engineering-wait-0924.json')['acceptance_run_started']
assert archived('control', 'qc-current-continuation-0924.json')['steps'] == []
assert not (ROOT / 'acceptance-05/receipt.json').exists()
for path in DOC.rglob('*'):
    if not path.is_file():
        continue
    raw = path.read_bytes()
    assert not raw or (raw.endswith(b'\n') and not raw.endswith(b'\n\n')), path
    assert not re.search(rb'[ \t]+$', raw, re.M), path
    assert len(raw) < 500000, path
    assert path.suffix not in {'.py', '.mjs', '.ts', '.tsx', '.sb', '.sh', '.duckdb', '.db', '.sqlite'}, path
    if path.suffix == '.json':
        json.loads(raw)
for path in [DOC / 'README.md', DOC / 'spec/CLAIMS.md', DOC / 'RELEASE-GATES.md',
             REPO / 'docs/handoffs/2026-09-24-re06-current-main-resource-block.md']:
    dom = ET.fromstring('<root>' + markdown.markdown(path.read_text(), extensions=['tables', 'fenced_code']) + '</root>')
    for link in dom.iter('a'):
        target = link.get('href', '').split('#', 1)[0]
        if target and '://' not in target:
            assert (path.parent / target).exists(), (path, target)
claims_dom = ET.fromstring('<root>' + markdown.markdown((DOC / 'spec/CLAIMS.md').read_text(), extensions=['tables']) + '</root>')
claims = {''.join(row[0].itertext()): ''.join(row[-1].itertext()) for row in claims_dom.findall('.//tbody/tr')}
assert claims == report['current_claims'] == {f'C{i}': 'not_verified' for i in range(1, 11)}
handoff = REPO / 'docs/handoffs/inflight/docs-closeout-workorders-0922.md'
assert handoff.stat().st_size <= 3000
summary = {'status': 'PASS', 'scope': 'Byte/identity/accounting/terminal-state integrity only; not independent semantic approval',
           'revision': identity['revision'], 'archives': results,
           'original_files_verified': sum(result['files'] for result in results),
           'requests_prior': 147, 'requests_new': 0, 'requests_remaining': 71,
           'candidate_tests': 0, 'candidate_builds': 0, 'candidate_model_requests': 0,
           'handoff_bytes': handoff.stat().st_size, 'model_requests': 0, 'pytest_runs': 0}
if '--save' in sys.argv:
    (DOC / 'verify_packet.py.txt').write_bytes(Path(__file__).read_bytes())
    (DOC / 'verification.json').write_text(json.dumps(summary, indent=2) + '\n')
print(json.dumps(summary))
