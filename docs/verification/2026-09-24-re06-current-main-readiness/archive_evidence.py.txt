"""Reversible text evidence archive; never archive fixtures or credentials."""
import base64
import gzip
import hashlib
import json
from pathlib import Path
import re


def digest(data):
    return hashlib.sha256(data).hexdigest()


def decode(destination, item):
    paths=[destination/p for p in item['artifacts']]
    if item['encoding']=='identity':
        assert len(paths)==1
        return paths[0].read_bytes()
    assert item['encoding']=='gzip-base64-chunks'
    pieces=[json.loads(p.read_text()) for p in paths]
    compressed=b''.join(base64.b64decode(p['data'],validate=True) for p in pieces)
    assert digest(compressed)==item['compressed_sha256']
    return gzip.decompress(compressed)


def archive(source, destination, revision, names, credential_fixtures=None):
    source,destination=Path(source),Path(destination)
    assert not (destination/'manifest.json').exists()
    destination.mkdir(parents=True,exist_ok=True)
    credential_fixtures=credential_fixtures or {}
    manifest={'schema':'reversible-evidence-v1','source_root':str(source),'revision':revision,'nonsecret_pattern_exemptions':credential_fixtures,'files':{}}
    for name in sorted(set(names)):
        relative=Path(name)
        assert not relative.is_absolute() and '..' not in relative.parts
        original=source/relative
        assert not original.is_symlink() and original.resolve().is_relative_to(source.resolve())
        raw=original.read_bytes()
        raw.decode('utf-8')
        assert not re.search(rb'-----BEGIN (?:RSA |OPENSSH |EC )?PRIVATE KEY-----(?:\r?\n|\\n)[ \t]*[A-Za-z0-9+/]{16,}',raw),f'Private-key material: {name}'
        matches=re.findall(rb'(?<![A-Za-z0-9])sk-(?:ant-)?[A-Za-z0-9_-]{24,}|eyJ[A-Za-z0-9_-]{12,}\.eyJ[A-Za-z0-9_-]{12,}\.[A-Za-z0-9_-]{12,}',raw)
        assert all(digest(match) in credential_fixtures for match in matches),f'Credential-pattern review required: {name}'
        item={'sha256':digest(raw),'bytes':len(raw),'artifacts':[]}
        canonical = not raw or (raw.endswith(b'\n') and not raw.endswith(b'\n\n') and b'\r' not in raw and not re.search(rb'[ \t]+$',raw,re.M))
        if relative.suffix in {'.jsonl','.log'} or '.log.' in relative.name or len(raw)>400000 or not canonical:
            compressed=gzip.compress(raw,mtime=0)
            item.update(encoding='gzip-base64-chunks',compressed_sha256=digest(compressed))
            chunks=[compressed[n:n+131072] for n in range(0,len(compressed),131072)]
            for index,chunk in enumerate(chunks,1):
                target=relative.with_name(relative.name+f'.part{index:03}.json')
                path=destination/target
                assert not path.exists()
                path.parent.mkdir(parents=True,exist_ok=True)
                path.write_text(json.dumps({'source':name,'raw_sha256':item['sha256'],'raw_bytes':len(raw),
                    'chunk':index,'chunks':len(chunks),'encoding':'gzip-base64-chunk',
                    'data':base64.b64encode(chunk).decode('ascii')},indent=2)+'\n')
                item['artifacts'].append(str(target))
        else:
            target=relative.with_name(relative.name+'.txt') if relative.suffix in {'.py','.mjs','.sb','.ts','.tsx','.sh'} else relative
            path=destination/target
            assert not path.exists()
            path.parent.mkdir(parents=True,exist_ok=True)
            path.write_bytes(raw)
            item.update(encoding='identity',artifacts=[str(target)])
        assert decode(destination,item)==raw
        manifest['files'][name]=item
    for name,item in manifest['files'].items():
        assert digest((source/name).read_bytes())==item['sha256'],f'Changed during archival: {name}'
    (destination/'manifest.json').write_text(json.dumps(manifest,ensure_ascii=False,indent=2)+'\n')
    return {'files':len(manifest['files']),'raw_bytes':sum(item['bytes'] for item in manifest['files'].values()),
            'encoded_files':sum(len(item['artifacts']) for item in manifest['files'].values())}


def verify(destination):
    destination=Path(destination)
    manifest=json.loads((destination/'manifest.json').read_text())
    for item in manifest['files'].values():
        raw=decode(destination,item)
        assert len(raw)==item['bytes'] and digest(raw)==item['sha256']
    return len(manifest['files'])
