"""Materialize fixed-revision source inputs; exclude author conclusions."""
import hashlib
import json
from pathlib import Path
import subprocess
ROOT = Path(__file__).resolve().parent
config = json.loads((ROOT / 'config.json').read_text())
def git(*args):
    return subprocess.check_output(['git', '-C', config['tree'], *args], text=True)
assert git('rev-parse', 'HEAD').strip() == config['revision']
assert not git('status', '--porcelain').strip()
inputs = ROOT / 'inputs'
inputs.mkdir(exist_ok=False)
(inputs / 'source.diff').write_text(git('diff', config['baseline'], config['revision'], '--', 'intelligence/'))
files = git('diff', '--name-only', config['baseline'], config['revision'], '--', 'intelligence/').splitlines()
(inputs / 'identity.json').write_text(json.dumps({**config, 'changed_source_sha256': {p: hashlib.sha256((Path(config['tree']) / p).read_bytes()).hexdigest() for p in files}}, indent=2) + '\n')
claims = Path('/Users/a77/fwp-wt-closeout-workorders-0922/docs/verification/2026-09-23-re06-timer-scope/spec/CLAIMS.md').read_text()
rows = [line for line in claims.splitlines() if line.startswith('| C')]
assert len(rows) == 10
(inputs / 'claims.md').write_text('# Author claims, not evidence\n\nPaths below are relative to intelligence/.\n\n| ID | Claim | Source locator | Independent state |\n|---|---|---|---|\n' + '\n'.join(rows) + '\n')
print(json.dumps({'revision': config['revision'], 'source_files': len(files), 'claims': len(rows)}))
