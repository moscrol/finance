def now():
    return datetime.now(timezone.utc).isoformat()

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def identity():
    def git(*args):
        return subprocess.check_output(['git', '-C', str(TREE), *args], text=True).strip()
    return {'revision': git('rev-parse', 'HEAD'), 'status': git('status', '--porcelain')}

record = {'revision': REVISION, 'baseline': BASELINE, 'tree': str(TREE), 'started_at': now(),
          'runner_sha256': sha(Path(__file__)), 'complete': False, 'leaves': {},
          'resource_admissions': [], 'identity_before': identity(),
          'model_requests_authorized': False, 'production_writes_authorized': False,
          'ports': [29501,29504]}
assert record['identity_before'] == {'revision': REVISION, 'status': ''}
(ROOT / 'receipts').mkdir(exist_ok=False)

def save():
    tmp = ROOT / 'receipt.json.tmp'
    tmp.write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
    tmp.replace(ROOT / 'receipt.json')

def admit(leaf):
    deadline = time.monotonic() + 300
    while True:
        ps = subprocess.check_output(['ps', '-axo', 'pid=,command='], text=True)
        processes = [line.strip() for line in ps.splitlines() if '-m pytest' in line and ('/Python ' in line or '/python ' in line)]
        load = list(os.getloadavg())
        free = shutil.disk_usage(ROOT).free / 1024**3
        allowed = load[0] <= 8 and len(processes) <= 2 and free >= 8
        obs = {'leaf': leaf, 'at': now(), 'load': load, 'pytest_processes': len(processes), 'free_disk_gib': round(free,3), 'allowed': allowed}
        record['resource_admissions'].append(obs)
        save()
        if allowed:
            assert identity() == {'revision': REVISION, 'status': ''}
            return True
        if time.monotonic() >= deadline:
            record['leaves'][leaf] = {'status': 'BLOCKED_RESOURCE_GATE'}
            save()
            return False
        time.sleep(10)

def command(name, args, timeout=1800):
    log = ROOT / (name + '.log.txt')
    start = time.monotonic()
    timed_out = False
    with log.open('xb') as stream:
        proc = subprocess.Popen(args, cwd=TREE, env=ENV, stdout=stream, stderr=subprocess.STDOUT, start_new_session=True)
        try:
            code = proc.wait(timeout=timeout)
        except subprocess.TimeoutExpired:
            timed_out = True
            os.killpg(proc.pid, signal.SIGTERM)
            try:
                code = proc.wait(timeout=10)
            except subprocess.TimeoutExpired:
                os.killpg(proc.pid, signal.SIGKILL)
                code = proc.wait()
    result = {'command': args, 'exit_code': code, 'timed_out': timed_out, 'elapsed_seconds': round(time.monotonic()-start,3),
              'log': log.name, 'log_sha256': sha(log), 'log_bytes': log.stat().st_size}
    print(json.dumps({'step': name, **result}), flush=True)
    return result
