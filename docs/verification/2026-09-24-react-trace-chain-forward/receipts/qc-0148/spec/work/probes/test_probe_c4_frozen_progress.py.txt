"""Independent C4 probe: frozen nested Mappings normalize for progress without mutation.

Unknown objects must stay rejected rather than being stringified into a query key.
"""
from __future__ import annotations

import json
from types import MappingProxyType

import pytest

from intelligence.runtime.research_progress import normalize_query


_INNER = {"field": "stage_day", "op": "eq", "value": 3}


def test_c4_frozen_nested_mapping_records_progress_without_mutation():
    mutable_inner = dict(_INNER)
    query = MappingProxyType(
        {"dataset": "market_daily", "filters": MappingProxyType(dict(mutable_inner))}
    )
    snapshot = json.dumps(mutable_inner, sort_keys=True)

    first = normalize_query(query)
    second = normalize_query(query)

    assert isinstance(first, str) and first  # a real progress key, not an error marker
    assert first == second  # deterministic for repeated identical calls
    # Freezing must not change the key: same arguments as plain dicts.
    assert first == normalize_query(
        {"dataset": "market_daily", "filters": dict(_INNER)}
    )
    # The nested dict the proxy wrapped is untouched, and the proxy is still read-only.
    assert json.dumps(mutable_inner, sort_keys=True) == snapshot
    with pytest.raises(TypeError):
        query["dataset"] = "tampered"
    assert query["dataset"] == "market_daily"


@pytest.mark.parametrize("value", [object(), {1, 2}, object()])
def test_c4_unsupported_objects_are_rejected_not_stringified(value):
    with pytest.raises(
        TypeError, match=f"Object of type {type(value).__name__} is not JSON serializable"
    ):
        normalize_query(
            MappingProxyType(
                {"filters": MappingProxyType({"value": value}), "dataset": "market_daily"}
            )
        )
