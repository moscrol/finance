"""Independent C3 probe: a local validation failure feeds back into the same episode.

Reuses the pre-existing `_frame`/`_context`/`_finish_turn` episode fixtures and a
throwaway duckdb store; the scenario (good query, invalid query, finish) is mine.
"""
from __future__ import annotations

import json
from dataclasses import replace
from threading import Event
from uuid import uuid4

import duckdb
import pytest

from intelligence.runtime.agent_episode import ContinuousAgentEpisode
from intelligence.services import episode_tools
from intelligence.services.agent_runtime import ModelToolCall, ModelTurn
from intelligence.tests.test_agent_episode import _context, _finish_turn, _frame

_GOOD = {
    "dataset": "market_daily",
    "dimensions": ["trade_date"],
    "metrics": ["index_return_pct"],
    "time_range": {"start": "2026-07-21", "end": "2026-07-21"},
}
_INVALID = {
    "dataset": "market_daily",
    "dimensions": ["trade_date"],
    "metrics": ["definitely_not_a_metric"],
    "time_range": {"start": "2026-07-21", "end": "2026-07-21"},
}


def _setup(tmp_path, *, max_steps=3):
    path = tmp_path / "db" / "market_feature_store.duckdb"
    path.parent.mkdir()
    with duckdb.connect(str(path)) as con:
        con.execute(
            "CREATE TABLE fact_market_daily "
            "(trade_date DATE, stage_day INTEGER, sh_index_pct_chg DOUBLE)"
        )
        con.execute("INSERT INTO fact_market_daily VALUES ('2026-07-21', 3, 1.25)")
    frame = _frame()
    context = replace(
        _context(frame, max_steps=max_steps, allowed_capabilities=("finance_query",)),
        trace_parent_id=uuid4().hex,
    )
    registry = episode_tools.build_episode_registry(
        frame, context, finance_root=tmp_path, knowledge_wiki=tmp_path / "wiki",
    )
    return frame, context, registry, path


class _Scripted:
    """Good query first, invalid query second, then finish."""

    def __init__(self, *, cancel=None):
        self.cancel = cancel
        self.turns: list[list[dict]] = []

    def complete(self, *, messages, tools, timeout):
        self.turns.append([dict(m) for m in messages])
        tool_messages = [m for m in messages if m.get("role") == "tool"]
        if not tool_messages:
            return ModelTurn(
                "", (ModelToolCall("probe-good", "finance_query", _GOOD),), "probe", ""
            )
        if len(tool_messages) == 1:
            if self.cancel is not None:
                self.cancel.set()
            return ModelTurn(
                "", (ModelToolCall("probe-bad", "finance_query", _INVALID),), "probe", ""
            )
        return _finish_turn(
            status="partial", draft="Evidence from the good query is retained.",
            hashes=(), gap="Diagnostic delivery only.",
        )


def test_c3_feedback_stays_in_episode_and_keeps_prior_evidence(tmp_path):
    frame, context, registry, path = _setup(tmp_path)
    frozen = path.read_bytes()
    model = _Scripted()
    result = ContinuousAgentEpisode(model).run(
        task_frame=frame, context=context, registry=registry,
    )
    assert result.stop_reason == "model_finish"
    assert result.usage.tool_calls == 2  # no extra slot minted for the repair loop
    feedback = [
        json.loads(m["content"])
        for m in model.turns[-1]
        if m.get("role") == "tool" and m.get("tool_call_id") == "probe-bad"
    ]
    assert len(feedback) == 1
    payload = feedback[0]
    assert payload["ok"] is False
    assert "结构化查询参数无效" in payload["observation"]
    assert "重试提示" in payload["observation"]
    assert not payload["evidence"]
    assert len(result.evidence) == 1  # the good query's evidence survived
    assert path.read_bytes() == frozen  # a rejected query leaves the store untouched


def test_c3_exhausted_tool_budget_stops_after_feedback_without_dispatch(tmp_path):
    frame, context, registry, _ = _setup(tmp_path, max_steps=1)


class _InvalidFirst(_Scripted):
    def complete(self, *, messages, tools, timeout):
        tool_messages = [m for m in messages if m.get("role") == "tool"]
        if not tool_messages:
            return ModelTurn(
                "", (ModelToolCall("probe-bad", "finance_query", _INVALID),), "probe", ""
            )
        self.turns.append([dict(m) for m in messages])
        pytest.fail("no dispatch may follow once the tool budget is exhausted")


def test_c3_budget_one_invalid_call_only(tmp_path):
    frame, context, registry, _ = _setup(tmp_path, max_steps=1)
    model = _InvalidFirst()
    result = ContinuousAgentEpisode(model).run(
        task_frame=frame, context=context, registry=registry,
    )
    assert result.usage.tool_calls == 1
    assert not result.evidence


def test_c3_cancellation_after_feedback_prevents_dispatch(tmp_path):
    frame, context, registry, _ = _setup(tmp_path)
    cancelled = Event()
    model = _Scripted(cancel=cancelled)
    result = ContinuousAgentEpisode(model, is_cancelled=cancelled.is_set).run(
        task_frame=frame, context=context, registry=registry,
    )
    assert result.stop_reason == "cancelled"
    assert result.usage.tool_calls == 1
