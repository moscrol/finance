"""Real loopback relay roundtrip; upstream HTTPS is replaced only in this process."""
import http.client
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
import json
from pathlib import Path
import threading
from unittest.mock import patch

from glm_review_shim import Ledger, ReviewServer

ROOT = Path(__file__).resolve().parent
seen = []


class FakeUpstream(BaseHTTPRequestHandler):
    def log_message(self, *_args):
        pass

    def do_POST(self):
        seen.append(json.loads(self.rfile.read(int(self.headers['Content-Length']))))
        body = b'{"choices":[{"message":{"role":"assistant","content":"OK"}}]}'
        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Content-Length', str(len(body)))
        self.end_headers()
        self.wfile.write(body)


fake = ThreadingHTTPServer(('127.0.0.1', 0), FakeUpstream)
ledger = Ledger(ROOT/'shim-wire-ledger', 1)
relay = ReviewServer(0, 443, ledger, 3, 20)
threads = [threading.Thread(target=server.serve_forever) for server in [fake, relay]]
for thread in threads:
    thread.start()
payload = {'model':'glm-5.3','messages':[{'role':'user','content':'offline fixture'}],
           'thinking':{'type':'enabled','clear_thinking':False},'reasoning_effort':'low','max_tokens':8192}
checks = []
try:
    def connection(host, port, timeout):
        assert host == 'open.bigmodel.cn' and port == 443
        return http.client.HTTPConnection('127.0.0.1', fake.server_port, timeout=timeout)

    with patch('glm_review_shim.http.client.HTTPSConnection', side_effect=connection):
        for label, changes, expected in [
            ('missing_thinking', {'thinking':None},422),
            ('nominal_off', {'thinking':{'type':'disabled'}},422),
            ('default_max_effort', {'reasoning_effort':None},422),
            ('wrong_token_field', {'max_completion_tokens':8192},422),
            ('excess_tokens', {'max_tokens':8193},422),
            ('native_low', {},200),
        ]:
            client = http.client.HTTPConnection('127.0.0.1',relay.server_port,timeout=4)
            client.request('POST','/v1/chat/completions',json.dumps({**payload,**changes}),{'Content-Type':'application/json'})
            response = client.getresponse()
            response.read()
            client.close()
            assert response.status == expected, (label,response.status)
            if expected == 422:
                assert not seen and ledger.state['requests'] == ledger.state['dispatched'] == 0
            checks.append({'label':label,'status':response.status})
        assert seen == [payload]
finally:
    for server in [relay,fake]:
        server.shutdown()
        server.server_close()
    for thread in threads:
        thread.join()
    ledger.close()
assert ledger.state['dispatched'] == ledger.state['completed'] == 1
assert ledger.state['shutdown_complete'] and ledger.state['active'] == 0
receipt = {'status':'PASS','model_requests':0,'fake_upstream_requests':len(seen),'checks':checks,
           'original_body_preserved':True,'rejected_before_quota_or_external_io':True}
with (ROOT/'shim-wire-control.json').open('x') as stream:
    json.dump(receipt,stream,indent=2)
    stream.write('\n')
print(json.dumps(receipt))
