from intelligence.runtime.research_progress import (
    ResearchProgressTracker,
    ToolCallDigest,
    normalize_query,
)


def test_frozen_nested_mapping_normalizes():
    frozen = MappingProxyType({
        "query": MappingProxyType({"goals": ("g1", "g2")}),
    })
    out = normalize_query(frozen)
    assert "g1" in out and "g2" in out
    # no mutation possible; original still read-only structure
    assert len(frozen) == 1


def test_unknown_object_still_rejected():
    class Weird:
        pass
    try:
        normalize_query(MappingProxyType({"query": Weird()}))
    except TypeError:
        pass
    else:
        raise AssertionError("unknown object must be rejected, not coerced")


def test_tracker_records_digest_without_error():
    tracker = ResearchProgressTracker()
    tracker.record_call(ToolCallDigest(
        tool="finance_query",
        query=MappingProxyType({"query": "总市值 前十"}),
        status="new", new_evidence=2, total_evidence=5,
    ))
    batch = tracker.close_batch()
    assert batch is not None
    assert batch.calls[0].new_evidence == 2
