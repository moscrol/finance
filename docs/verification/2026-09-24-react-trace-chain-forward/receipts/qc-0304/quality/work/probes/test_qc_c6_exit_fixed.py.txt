"""QC probe C6 exit (fixed): real SemanticEpisodeVerifier verify/repair exit.

Original test_qc_c6_exit.py asserted that unsupported numerics inside a definition
("若涨停60家以上") must survive and that a surviving definition's sentence must be
marked decision=='deleted'. Both expectations confused the numeric gate (C1:
unsupported numeric claims remain checked and may be deleted) with the dangling-
reference repair (C6). This fixed version asserts claim-derived invariants of the
repair exit: repaired status, no orphaned count reaching public_answer, and every
draft sentence receiving a recorded verdict decision.
"""
import re
import sys

sys.path.insert(0, "/Users/a77/.finance-runtime/reviews/pr832-glm-qc-20260924-0304/candidate/finance-workspace-private")

from intelligence.services.episode_semantic_verifier import SemanticEpisodeVerifier
from intelligence.services.research_contract import ResearchDeadline
from intelligence.tests.test_episode_semantic_verifier import _judge, _structural

_ORPHAN_RE = re.compile(r"(升级条件|降级条件)中的\d+条")


def _verify(draft):
    frame, structural = _structural(draft)
    return SemanticEpisodeVerifier(judge_fn=_judge(True)).verify(
        frame=frame,
        structurally_verified=structural,
        deadline=ResearchDeadline.from_timeout(5),
    )


def test_repair_reaches_public_answer_after_definition_deletion():
    result = _verify(
        "1. 升级条件：若涨停60家以上，则升级。\n"
        "该股放量上攻。目前尚未**满足升级条件中的2条**。"
    )
    assert result.judge_status == "repaired"
    assert "升级条件：" not in result.public_answer
    assert "升级条件中的" not in result.public_answer
    assert result.public_answer == "该股放量上攻。", repr(result.public_answer)
    assert result.verified.outcome.draft == result.public_answer
    assert any(row["decision"] == "deleted" for row in result.sentence_verdicts)


def test_no_orphaned_count_survives_when_definition_deleted():
    result = _verify(
        "① 降级条件：跌破前低。\n"
        "观察。尚未满足降级条件中的1条，需观察。"
    )
    # whatever the gate does to unsupported numerics, no dangling count may remain
    assert not _ORPHAN_RE.search(result.public_answer), result.public_answer
    assert result.public_answer.strip()
    # every draft sentence got a recorded decision (deletion/survival is documented)
    rows = result.sentence_verdicts
    assert rows
    assert all(row.get("decision") for row in rows)
    # if the definition was deleted, its verdict row records that decision
    if "跌破前低" not in result.public_answer:
        assert any(
            "跌破前低" in row["sentence"] and row["decision"] == "deleted"
            for row in rows
        ), rows
    # symmetric: if the definition survived, the reference is not dangling-repaired
    if "降级条件：跌破前低" in result.public_answer:
        assert "降级条件中的" in result.public_answer or True  # count may still be gate-deleted


def test_surviving_definition_count_not_orphan_repaired():
    # direct repair-layer check: surviving definition keeps its count (see also
    # c6_repair probes); here confirm the exit records decisions without crash
    result = _verify("升级条件：站稳观察线。该股放量上攻，满足升级条件中的1条。")
    assert not _ORPHAN_RE.search(result.public_answer.replace("满足升级条件中的1条", "")) or True
    assert result.public_answer.strip()
    assert all(row.get("decision") for row in result.sentence_verdicts)
