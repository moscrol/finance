"""C6 probe v2: fixture fixes only.
- _rejected now uses the product's _CONDITION_DEFINITION_RE (definitions may carry
  list prefixes), instead of naive startswith.
- the survivor test's precondition now counts ALL definition sentences in `before`
  (there are three, not two). Expectations derived from claims.md C6 unchanged."""

from intelligence.services.episode_semantic_verifier import (
    SemanticEpisodeVerifier,
    _numbered_sentences,
    _repair_dangling_condition_references,
    _CONDITION_DEFINITION_RE,
)
from intelligence.services.research_contract import ResearchDeadline
from intelligence.tests.test_episode_semantic_verifier import _judge, _structural


DEFS = "升级条件：站稳20日线。降级条件：跌破前低。"


def _rejected(text):
    return tuple(
        int(i["index"])
        for i in _numbered_sentences(text)
        if _CONDITION_DEFINITION_RE.match(str(i["text"]))
    )


def test_repair_runs_after_sentence_deletion_and_surviving_definition_blocks_it():
    before = DEFS + "\n降级条件：风险可控。已满足降级条件中的1条，需观察。"
    rejected = _rejected(before)
    assert rejected == (1, 2, 3)  # three definition sentences in `before`
    draft = "降级条件：风险可控。已满足降级条件中的1条，需观察。"
    out = _repair_dangling_condition_references(
        draft, before=before, rejected_sentence_indexes=rejected
    )
    # surviving definition (降级条件：风险可控) keeps its count
    assert "已满足降级条件中的1条" in out


def test_deleted_only_definition_removes_count_keeps_unrelated_prose():
    before = "升级条件：若涨停60家以上，则升级。\n- 观察一。\r\n- 量能配合，满足升级条件中的2条，需跟踪。\r\n"
    rejected = _rejected(before)
    assert rejected == (1,)
    draft = "- 观察一。\r\n- 量能配合，满足升级条件中的2条，需跟踪。\r\n"
    out = _repair_dangling_condition_references(
        draft, before=before, rejected_sentence_indexes=rejected
    )
    assert "升级条件中的2条" not in out
    assert "量能配合" in out and "需跟踪" in out and "观察一" in out
    assert "\r\n" in out


def test_circled_and_ordered_definition_prefixes_are_recognized():
    for prefix in ("① ", "一、", "（2）", "10. "):
        before = f"{prefix}升级条件：若放量，则升级。\n满足升级条件中的2条，量能配合。"
        rejected = _rejected(before)
        assert rejected == (1,), prefix
        draft = "满足升级条件中的2条，量能配合。"
        out = _repair_dangling_condition_references(
            draft, before=before, rejected_sentence_indexes=rejected
        )
        assert "升级条件中的" not in out, prefix
        assert "量能配合" in out, prefix


def test_non_definition_deleted_sentence_does_not_authorize_removal():
    before = "观察到升级条件：无关正文。\n满足升级条件中的2条，需观察。"
    assert _rejected(before) == ()  # prose, not a definition
    draft = "满足升级条件中的2条，需观察。"
    out = _repair_dangling_condition_references(
        draft, before=before, rejected_sentence_indexes=(1,)
    )
    assert out == draft


def test_real_verify_exit_delivers_repaired_public_answer():
    judge = _judge(True)
    frame, structural = _structural(
        "升级条件：若涨停60家以上，则升级。\n该股放量上攻。目前尚未**满足升级条件中的2条**。"
    )
    result = SemanticEpisodeVerifier(judge_fn=judge).verify(
        frame=frame,
        structurally_verified=structural,
        deadline=ResearchDeadline.from_timeout(5),
    )
    assert result.judge_status == "repaired"
    assert "满足升级条件中的2条" not in result.public_answer
    assert result.public_answer == "该股放量上攻。"
    assert result.verified.outcome.draft == result.public_answer
    assert any(r["decision"] == "deleted" and "60家" in r["sentence"]
               for r in result.sentence_verdicts)
