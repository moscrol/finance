"""C5 probe v2: unit pagination table now emulates the real read_history_result
slicing (preview/returned_count) before calling _result, matching production path.
Original probe preserved; fix is fixture-only."""

from tests.test_history_model_projection import _payload, _project, REF
from intelligence.services.historical_research.episode import _result


def test_unit_page_offsets_and_navigation():
    rows = [{"entity_code": "A.FP", "trade_date": f"2026-08-0{i}"} for i in (3, 4)]
    for offset, limit, nxt, trunc in ((0, 1, 1, True), (0, 2, None, False),
                                      (1, 1, None, True), (5, 1, None, False)):
        # real paging path slices preview and recomputes returned_count/truncated
        # (episode.py read_history_result rows[offset:offset+limit]) before _result.
        page = rows[offset:offset + limit]
        payload = dict(
            _payload(rows),
            preview=page,
            returned_count=len(page),
            truncated=len(page) < len(rows),
        )
        model, details = _project(_result(payload, result_ref=REF, offset=offset))
        scope = next(row for row in details if "total_matched" in row)
        assert scope["offset"] == offset
        assert scope["next_offset"] == nxt
        assert scope["truncated"] is trunc
        rendered = "null" if nxt is None else str(nxt)
        assert f'"next_offset":{rendered}' in model["observation"]
        samples = {row["sample"] for row in details if "entity_code" in row}
        assert samples == set(range(offset, min(offset + limit, 2)))
