from intelligence.services.episode_semantic_verifier import (  # noqa: F401
    _repair_dangling_condition_references,
)


def test_increment_exists_and_removes_dangling_count():
    before = "升级条件：若涨停60家以上，则升级。\n满足升级条件中的2条，量能配合。"
    draft = "满足升级条件中的2条，量能配合。"
    out = _repair_dangling_condition_references(
        draft, before=before, rejected_sentence_indexes=(1,))
    assert "升级条件中的2条" not in out
    assert "量能配合" in out


def test_history_pagination_increment_exists():
    from intelligence.services.historical_research.episode import _result

    payload = {"query_id": "q", "operation": "compute_history", "status": "research_only",
               "total_matched": 3, "returned_count": 1, "truncated": True,
               "preview": [{"entity_code": "A.FP"}], "rows": []}
    result = _result(payload, result_ref="ref", offset=2)
    assert '"offset":2' in result.observation
    assert '"next_offset":null' in result.observation  # 2+1 < 3? no -> None... 2+1=3 not <3
