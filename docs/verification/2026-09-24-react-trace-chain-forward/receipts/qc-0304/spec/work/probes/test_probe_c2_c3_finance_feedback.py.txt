"""C2+C3 probe: validation failures give safe structured diagnostics; a local tool
validation failure feeds back into the same episode keeping evidence/budget.

Borrows author fixtures (_setup/_request from test_finance_query_repair_feedback),
but expectations below are derived from claims.md C2/C3, not copied.
"""

from threading import Event
from uuid import uuid4

from intelligence.services import episode_tools, finance_query
from intelligence.services.agent_runtime import ModelToolCall, ModelTurn
from intelligence.runtime.agent_episode import ContinuousAgentEpisode
from intelligence.tests.test_agent_episode import _finish_turn
from intelligence.tests.test_finance_query_repair_feedback import _request, _setup


def test_validation_error_diagnostic_is_structured_and_prose_free():
    prose = "DROP TABLE /secret/path 99% fabricated"
    arguments = _request(metrics=[prose])
    result = episode_tools._finance_query_failure_result(
        finance_query.FinanceQuerySpec.from_arguments(arguments),
        finance_query.FinanceQueryValidationError(f"not a metric: {prose}"),
    )
    assert result.observation == ""
    assert not result.evidence
    assert len(result.diagnostics) == 1
    d = result.diagnostics[0]
    assert d.code == "invalid_query"
    assert "secret" not in d.message and "DROP TABLE" not in d.message
    assert "99%" not in d.message
    assert result.trace.status == "parse_error"


def test_invalid_query_via_registry_keeps_db_untouched(tmp_path):
    frame, context, registry, path = _setup(tmp_path)
    before = path.read_bytes()
    result = registry.execute(
        "finance_query", _request(), context=context, step_id=uuid4().hex)
    assert result.trace.status == "parse_error"
    assert not result.evidence
    assert "dimension" in result.observation or result.diagnostics
    assert path.read_bytes() == before


class _FeedbackModel:
    """Invalid call first, then repaired call, then finish."""

    def __init__(self):
        self.turns = 0

    def complete(self, *, messages, tools, timeout):
        tool_msgs = [m for m in messages if m.get("role") == "tool"]
        if not tool_msgs:
            return ModelTurn("", (ModelToolCall("q1", "finance_query", _request()),),
                             "scripted", "")
        if len(tool_msgs) == 1:
            feedback = tool_msgs[0]["content"]
            assert "dimension" in feedback or "market_daily" in feedback
            return ModelTurn(
                "",
                (ModelToolCall("q2", "finance_query",
                               _request(dimensions=["trade_date", "stage_day"],
                                        metrics=["index_return_pct"])),),
                "scripted", "",
            )
        assert any('"evidence"' in m["content"] or "1.25" in m["content"]
                   for m in tool_msgs[1:])
        return _finish_turn(
            status="partial", draft="反馈链路验证。",
            hashes=(), gap="仅验证修复反馈交付，不验证研究质量。",
        )


def test_repair_feedback_stays_in_same_episode_without_extra_budget(tmp_path):
    frame, context, registry, _ = _setup(tmp_path, max_steps=2)
    model = _FeedbackModel()
    result = ContinuousAgentEpisode(model).run(
        task_frame=frame, context=context, registry=registry)
    assert result.stop_reason == "model_finish"
    assert result.usage.tool_calls == 2  # invalid + repaired, no extra slot minted
    events = [e for e in result.events if e.kind == "tool_result"]
    assert len(events) == 2


def test_cancellation_after_feedback_stops_further_dispatch(tmp_path):
    frame, context, registry, _ = _setup(tmp_path)
    cancelled = Event()

    class _CancelModel(_FeedbackModel):
        def complete(self, *, messages, tools, timeout):
            turn = super().complete(messages=messages, tools=tools, timeout=timeout)
            if len([m for m in messages if m.get("role") == "tool"]) == 1:
                cancelled.set()
            return turn

    result = ContinuousAgentEpisode(
        _CancelModel(), is_cancelled=cancelled.is_set).run(
        task_frame=frame, context=context, registry=registry)
    assert result.stop_reason == "cancelled"
    assert result.usage.tool_calls <= 2
