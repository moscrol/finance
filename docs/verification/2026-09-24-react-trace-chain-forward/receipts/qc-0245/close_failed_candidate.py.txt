        'verdict':'CHANGES_REQUIRED_C6_ORDERED_DEFINITION_AND_BLOCKED_INDEPENDENT_COVERAGE',
        'model_requests_in_batch':56,'model_requests_added':0,'current_authorization_dispatched_total':124,
        'model_report':'Spec BLOCKED with C6 not_verified; Quality never started.',
        'observed':['4 gateway + 19 explore + 20 execute + 13 report requests completed; no provider failed/cancelled requests.',
                    'Every actual relay admission carried enabled thinking, low effort and native max_tokens <=8192.',
                    'Spec final report appropriately refuses C6 independent verification but incorrectly attributes every C6 failure solely to its synthetic grammar.',
                    'Host reproduction through SemanticEpisodeVerifier.verify proves ordered 1./1)/(1) definition deletion leaves orphaned count in public_answer.',
                    'Host helper reproduction proves a surviving ordered definition is missed and its live count is wrongly removed.'],
        'host_diagnosis':str(ROOT/'host-c6-diagnostic.json'),
        'limits':['Host reproduction uses an author structural fixture; it is not a new independent signoff.',

engineering = ROOT.parent/'pr832-glm-qc-20260924-0148/engineering'
state = json.loads((engineering/'status.json').read_text())
assert state['status'] == 'WAIT_RESOURCE' and not state['steps']
pid = 2056
command = subprocess.check_output(['ps','-p',str(pid),'-o','command='],text=True).strip()
assert 'pr832-glm-qc-20260924-0148/run_engineering.py' in command
os.kill(pid,signal.SIGTERM)
for _ in range(100):
    check = subprocess.run(['ps','-p',str(pid),'-o','state='],capture_output=True,text=True)
    if check.returncode or check.stdout.strip() == 'Z':
        break
    time.sleep(0.1)
