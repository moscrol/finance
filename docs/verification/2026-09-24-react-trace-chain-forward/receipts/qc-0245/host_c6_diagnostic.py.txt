"""Host-only boundary diagnosis, never a replacement reviewer execution."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

ROOT = Path(__file__).resolve().parent
TREE = ROOT/'candidate/finance-workspace-private'
sys.path.insert(0,str(TREE))
from intelligence.services.episode_semantic_verifier import (
    SemanticEpisodeVerifier, _drop_rejected_sentences, _numbered_sentences,
    _repair_dangling_condition_references,
)
from intelligence.services.research_contract import ResearchDeadline
from intelligence.tests.test_episode_semantic_verifier import _judge, _structural

before_sha = subprocess.check_output(['git','-C',str(TREE),'rev-parse','HEAD'],text=True).strip()
rows = []
for prefix in ['', '- ', '1. ', '1) ', '(1) ']: 
    source = prefix+'升级条件：若涨停60家以上，则升级。\n该股放量上攻。目前尚未**满足升级条件中的2条**。'
    sentences = _numbered_sentences(source)
    draft = _drop_rejected_sentences(source, (1,))
    repaired = _repair_dangling_condition_references(draft,before=source,rejected_sentence_indexes=(1,))
    frame, structural = _structural(source)
    judge = _judge(True)
    result = SemanticEpisodeVerifier(judge_fn=judge).verify(frame=frame,structurally_verified=structural,deadline=ResearchDeadline.from_timeout(5))
    rows.append({'prefix':prefix,'sentences':sentences,'drop_result':draft,'helper_result':repaired,
                 'public_answer':result.public_answer,'judge_status':result.judge_status,
                 'sentence_verdicts':result.sentence_verdicts})
retained = '1. 升级条件：量能改善。\n满足升级条件中的1条，需观察。'
source = '升级条件：若涨停60家以上，则升级。\n'+retained
rows.append({'kind':'ordered_survivor','source':source,'draft':retained,
             'result':_repair_dangling_condition_references(retained,before=source,rejected_sentence_indexes=(1,))})
assert before_sha == subprocess.check_output(['git','-C',str(TREE),'rev-parse','HEAD'],text=True).strip()
assert not subprocess.check_output(['git','-C',str(TREE),'status','--porcelain'])
record = {'classification':'HOST_PRODUCT_DIAGNOSIS_NOT_INDEPENDENT_REVIEW','model_requests':0,'revision':before_sha,
          'author_fixture_used':True,'source_sha256':hashlib.sha256((TREE/'intelligence/services/episode_semantic_verifier.py').read_bytes()).hexdigest(),'rows':rows}
with (ROOT/'host-c6-diagnostic.json').open('x') as stream:
    json.dump(record,stream,ensure_ascii=False,indent=2,default=str)
    stream.write('\n')
print(json.dumps(record,ensure_ascii=False,default=str))
os.sys.exit(0)
