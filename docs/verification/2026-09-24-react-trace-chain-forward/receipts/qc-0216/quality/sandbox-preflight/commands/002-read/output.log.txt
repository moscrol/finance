def test_sqlite_wal_and_temp(tmp_path):
    parent = tmp_path / "nested"
    parent.mkdir(parents=True)
    with sqlite3.connect(parent / "control.sqlite3") as con:
        assert con.execute("PRAGMA journal_mode=WAL").fetchone() == ("wal",)
        con.execute("CREATE TABLE controls (value INTEGER)")
        con.execute("INSERT INTO controls VALUES (81)")
        assert con.execute("SELECT value FROM controls").fetchone() == (81,)
