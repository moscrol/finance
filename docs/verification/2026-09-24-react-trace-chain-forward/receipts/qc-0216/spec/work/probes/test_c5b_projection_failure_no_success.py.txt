"""PR832 spec probe 4 (C5b): projection failure cannot yield a successful read.

Source order in episode.py execute_query: ``result = _result(...)`` precedes
``context.history_results.append(metadata)``.  These tests pin the behavioral
half: ``_model_projection`` runs inside ``_result`` before any ToolRunResult
exists, so a projection failure cannot produce a success result to register.
"""
from __future__ import annotations

import pytest

from intelligence.services.historical_research import episode as hist_episode
from intelligence.services.historical_research.episode import _result


def _payload() -> dict:
    return {
        "query_id": "qid-c5b",
        "operation": "find_analogues",
        "status": "ok",
        "total_matched": 40,
        "returned_count": 20,
        "truncated": True,
        "preview": [{"entity_code": "801010.SI"}],
    }


def test_projection_failure_raises_out_of_result(monkeypatch: pytest.MonkeyPatch):
    def boom(payload, result_ref):
        raise RuntimeError("projection failed")

    monkeypatch.setattr(hist_episode, "_model_projection", boom)
    with pytest.raises(RuntimeError):
        _result(_payload(), result_ref="ref-x", offset=5)


def test_result_injects_offset_and_next_offset_before_projection(
    monkeypatch: pytest.MonkeyPatch,
):
    seen = {}

    def recorder(payload, result_ref):
        seen["payload"] = payload
        seen["result_ref"] = result_ref
        return [("历史观察样本 x", '{"sample": 5}', None)]

    monkeypatch.setattr(hist_episode, "_model_projection", recorder)
    res = _result(_payload(), result_ref="ref-y", offset=5)
    # Unconditional: the projection must see the page coordinates.
    assert seen["payload"]["offset"] == 5
    assert seen["payload"]["next_offset"] == 25
    assert seen["result_ref"] == "ref-y"
    assert res.trace.status == "success"
    assert res.telemetry["projected_evidence_count"] == 1
    assert res.evidence[0].internal_locator == "ref-y"


def test_final_page_injection_is_visible_to_projection(monkeypatch: pytest.MonkeyPatch):
    seen = {}

    def recorder(payload, result_ref):
        seen["payload"] = payload
        return []

    monkeypatch.setattr(hist_episode, "_model_projection", recorder)
    payload = dict(_payload(), total_matched=25, returned_count=20)
    res = _result(payload, result_ref="ref-z", offset=5)
    assert seen["payload"]["next_offset"] is None
    assert res.telemetry["next_offset"] is None
    assert res.telemetry["offset"] == 5
