"""Resolve the existing official GLM key in memory; --emit is for owned pipes only."""
import argparse
import json
from pathlib import Path
import shlex
import subprocess
import sys

parser = argparse.ArgumentParser()
parser.add_argument('--emit', action='store_true')
args = parser.parse_args()
values = {}
for line in Path('/Users/a77/.local/bin/start-finance-workbench').read_text().splitlines():
    if not line.startswith('export '):
        continue
    for item in shlex.split(line, comments=True)[1:]:
        name, sep, value = item.partition('=')
        if sep:
            values[name] = value
expression = values.get('FORESIGHT_BUILTIN_LLM_API_KEY', '')
assert expression == '$(security find-generic-password -a a77 -s finance-workbench-glm -w 2>/dev/null || true)'
assert values.get('FORESIGHT_BUILTIN_LLM_BASE_URL') == 'https://open.bigmodel.cn/api/coding/paas/v4'
result = subprocess.run(['/usr/bin/security', 'find-generic-password', '-a', 'a77', '-s', 'finance-workbench-glm', '-w'],
                        capture_output=True, timeout=15)
if result.returncode or not result.stdout.strip():
    sys.exit('Existing GLM credential unavailable')
key = result.stdout.strip()
assert b'\n' not in key and b'\r' not in key
if args.emit:
    sys.stdout.buffer.write(key)
else:
    print(json.dumps({'status': 'READY', 'source': 'existing launcher Keychain reference',
                      'base_url': values['FORESIGHT_BUILTIN_LLM_BASE_URL'], 'key_logged': False}))
