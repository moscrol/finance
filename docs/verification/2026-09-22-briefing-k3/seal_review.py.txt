def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def execute(name, cwd, argv):
    with (QC / f'{name}.log').open('xb') as log:
        result = subprocess.run(argv, cwd=cwd, env=ENV, stdout=log, stderr=subprocess.STDOUT, timeout=180)
    return {'name': name, 'cwd': str(cwd), 'argv': argv, 'exit_code': result.returncode,
            'log': str(QC / f'{name}.log'), 'sha256': digest(QC / f'{name}.log')}
