"""Exercise host admission plus sandboxed Python and UI children, no model calls."""
import ast
import hashlib
import json
from pathlib import Path
import subprocess
import time

ROOT=Path(__file__).resolve().parent
CONFIG=json.loads((ROOT/'config.json').read_text())
for name in ['run_probe_v3.py','run_stage_v3.py','stage_admission_v3.py','check_execution_v3.py']:
    ast.parse((ROOT/name).read_text())
boundary=json.loads((ROOT/'stage-boundary-preflight-v3-0925.json').read_text())
assert boundary['status']=='PASS'
assert all(hashlib.sha256((ROOT/name).read_bytes()).hexdigest()==value for name,value in boundary['source_sha256'].items())
out=ROOT/('execution-boundary-preflight-v3-'+str(time.time_ns()))
out.mkdir()
env={'HOME':str(ROOT/'work'),'PATH':'/Users/a77/.finance-runtime/reviews/pi-closeout-execution-20260924/re06-a30e-locked/toolchain/bin:/opt/homebrew/bin:/usr/bin:/bin:/usr/sbin:/sbin','LANG':'C.UTF-8',
     'TMPDIR':str(ROOT/'work/tmp'),'PYTHONDONTWRITEBYTECODE':'1','PYTHONNOUSERSITE':'1',
     'FORESIGHT_USERS_DIR':str(ROOT/'work/users')}
record={'status':'RUNNING','real_model_requests':0,'independent_candidate_tests':0,'runs':[]}
for role,total,passed in [('controller_preflight',10,9),('ui_controller_preflight',2,1),('controller_author_collection',None,0)]:
    command=[CONFIG['python'],'-I',str(ROOT/'run_probe_v3.py'),'controller',role]
    log=out/(role+'.log.txt')
    with log.open('xb') as stream:
        result=subprocess.run(command,cwd=ROOT,env=env,stdout=stream,stderr=subprocess.STDOUT,timeout=120)
    markers=[json.loads(line.removeprefix('EXECUTION_RECEIPT ')) for line in log.read_text().splitlines() if line.startswith('EXECUTION_RECEIPT ')]
    record['runs'].append({'role':role,'exit_code':result.returncode,'log':str(log),'markers':markers})
    (out/'receipt.json').write_text(json.dumps(record,indent=2)+'\n')
    if result.returncode==75:
        record['status']='BLOCKED_RESOURCE_GATE'; break
    assert result.returncode==(0 if role=='controller_author_collection' else 1) and len(markers)==1
    receipt=Path(markers[0]['path'])
    assert hashlib.sha256(receipt.read_bytes()).hexdigest()==markers[0]['sha256']
    data=json.loads(receipt.read_text()); counts=data['counts']
    assert data['source_kind']=='host_controller' and data['blocked'] is None
    if role=='controller_author_collection':
        assert counts and counts['collected']>0 and counts['passed']==counts['failed']==counts['tests']==0
    else:
        assert counts and counts['collected']==total and counts['passed']==passed and counts['failed']==1
    assert counts['errors']==counts['skipped']==counts['xfailed']==0
    child_log=Path(data['log']).read_text().lower()
    assert 'unhandled error' not in child_log and 'websocket server error' not in child_log
    assert data['command'][:3]==['/usr/bin/sandbox-exec','-f',str(receipt.parent/'child.sb')]
    assert data['child_policy_sha256']==hashlib.sha256((receipt.parent/'child.sb').read_bytes()).hexdigest()
    assert data['io_helper_sha256']==hashlib.sha256((ROOT/'review_io_sandbox.py').read_bytes()).hexdigest()
    assert not (ROOT/'controller-denied-write.txt').exists()
else:
    record['status']='PASS'
    record['expected_host_control_failures']=2
    record['not_the_independent_qc_positive_control']=True
    files=['run_probe_v3.py','review_v3.mjs','probe_execution_menu.mjs','run_stage_v3.py','stage_admission_v3.py','check_execution_v3.py','tools-v3.sb','resource_pytest.py','review_io_sandbox.py','review_artifact_writer.py','file_tool.py','inputs/ui.vitest-v3.config.mjs','inputs/qc_userspace_isolation.py']
    record['source_sha256']={name:hashlib.sha256((ROOT/name).read_bytes()).hexdigest() for name in files}
    with (ROOT/'execution-boundary-preflight-v3.json').open('x') as stream: json.dump(record,stream,indent=2)
(out/'receipt.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps({'status':record['status'],'out':str(out),'real_model_requests':0}))
raise SystemExit(0 if record['status']=='PASS' else 75)
