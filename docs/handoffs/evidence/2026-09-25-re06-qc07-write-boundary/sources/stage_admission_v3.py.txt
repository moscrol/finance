"""Fail closed on absent deliverables before invoking the credential-bearing runner."""
import ast
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess
import sys
import time
ROOT=Path(__file__).resolve().parent
stage,group=sys.argv[1:3]
assert stage in {'execute','report'} and group in {'timer','e2','consent'}
check_only=sys.argv[3:]==['--check-only']
prepare_only=sys.argv[3:]==['--prepare-only']
assert len(sys.argv)==3 or check_only or (prepare_only and stage=='execute')
config=json.loads((ROOT/'config.json').read_text())
assert stage+'/'+group in config['allowed_stage_groups'], 'Stage outside current bounded slice'
previous='explore' if stage=='execute' else 'execute'
work=ROOT/'work'/group
reasons=[]
files={}
intake=config.get('prior_explore',{}).get(group) if stage=='execute' else None
prior_path=Path(intake['execution']) if intake else ROOT/f'{previous}-{group}-v3'/'execution.json'
if intake and hashlib.sha256(prior_path.read_bytes()).hexdigest()!=intake['sha256']:
    reasons.append('PRIOR_EXPLORE_CHANGED')
if not prior_path.is_file():
    reasons.append('MISSING_PREVIOUS_STAGE')
else:
    prior=json.loads(prior_path.read_text())
    if prior.get('status')!='STAGE_COMPLETE' or prior.get('revision')!=config['revision']:
        reasons.append('PREVIOUS_STAGE_NOT_COMPLETE')
required=['EXPLORE.md','probes/test_reviewer.py','probes/test_positive_control.py'] if stage=='execute' else ['EXECUTE.md','EXECUTE.json','host-execution-check.json']
if stage=='execute' and group=='timer':
    required.append('probes/activity-reviewer.test.tsx')
if stage=='execute' and group=='consent':
    required.extend(['transaction-review.md','transaction-review.json'])
for name in required:
    host_check=name=='host-execution-check.json'
    path=ROOT/'stage-execution-checks-v3'/f'{group}.json' if host_check else work/name
    boundary=ROOT/'stage-execution-checks-v3' if host_check else work
    if path.is_symlink() or not path.resolve().is_relative_to(boundary.resolve()):
        reasons.append('UNSAFE_ARTIFACT:'+name)
        continue
    if not path.is_file() or not path.stat().st_size:
        reasons.append('MISSING_ARTIFACT:'+name)
        continue
    data=path.read_bytes()
    files[name]=hashlib.sha256(data).hexdigest()
    if path.suffix=='.py':
        try:
            module=ast.parse(data)
            assert any(isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef)) and n.name.startswith('test_') for n in ast.walk(module))
        except (SyntaxError,AssertionError):
            reasons.append('NO_PARSEABLE_PROBE_TESTS')
    if name=='host-execution-check.json':
        check=json.loads(data)
        if check.get('complete_execution_evidence') is not True or check.get('revision')!=config['revision']:
            reasons.append('EXECUTION_EVIDENCE_NOT_COMPLETE')
if intake:
    prior_manifest=json.loads((Path(intake['work'])/'manifest.json').read_text())
    for name, expected in prior_manifest['artifacts_sha256'].items():
        if files.get(name)!=expected:
            reasons.append('PRIOR_EXPLORE_ARTIFACT_CHANGED:'+name)
record={'at':datetime.now(timezone.utc).isoformat(),'stage':stage,'group':group,'revision':config['revision'],
        'allowed':not reasons,'reasons':reasons,'artifact_sha256':files,'credential_access_before_check':False,
        'model_requests_before_check':0,'runner_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'semantic_candidate_verdict':None,'check_only':check_only,'prepare_only':prepare_only}
folder=ROOT/'stage-admissions'
folder.mkdir(exist_ok=True)
with (folder/f'{stage}-{group}-{time.time_ns()}.json').open('x') as stream:
    json.dump(record,stream,indent=2)
print(json.dumps(record),flush=True)
if reasons:
    raise SystemExit(75)
if not check_only:
    if stage=='execute':
        frozen=ROOT/'inputs'/('frozen-explore-'+group+'-v3')
        frozen.mkdir(exist_ok=bool(intake))
        for name, expected in files.items():
            data=(work/name).read_bytes()
            assert hashlib.sha256(data).hexdigest()==expected
            destination=frozen/name
            if intake:
                assert not destination.is_symlink() and hashlib.sha256(destination.read_bytes()).hexdigest()==expected
            else:
                destination.parent.mkdir(parents=True,exist_ok=True)
                with destination.open('xb') as stream:
                    stream.write(data)
        helpers=['run_probe_v3.py','resource_pytest.py','review_io_sandbox.py','tools-v3.sb','inputs/ui.vitest-v3.config.mjs','inputs/qc_userspace_isolation.py']
        helper_hashes={name:hashlib.sha256((ROOT/name).read_bytes()).hexdigest() for name in helpers}
        binding={'revision':config['revision'],'baseline':config['baseline'],'group':group,'artifacts_sha256':files,'execution_tools_sha256':helper_hashes,'prior_explore':intake}
        with (frozen/'execution-manifest.json').open('x') as stream:
            json.dump(binding,stream,indent=2)
    if not prepare_only:
        raise SystemExit(subprocess.call([sys.executable,str(ROOT/'run_stage_v3.py'),stage,group]))
