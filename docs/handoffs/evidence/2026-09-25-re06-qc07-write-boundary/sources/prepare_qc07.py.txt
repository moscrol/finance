"""Create QC07 inputs only; never execute or reuse QC06 evidence."""
import hashlib
import json
from pathlib import Path
import shutil
import subprocess

ROOT = Path(__file__).resolve().parent
OLD = ROOT / 'qc-execution-final-06'
QC = ROOT / 'qc-evidence-isolated-07'
TOOLS = Path('/Users/a77/fwp-wt-re06-closeout-refresh-0925/scripts/review_probes')
assert not QC.exists()
config = json.loads((OLD / 'config.json').read_text())
config['tree'] = config['tree'].replace(str(OLD), str(QC))
config['prior_explore']['e2']['work'] = str(QC / 'inputs/frozen-explore-e2-v3')
# This slice reserves eleven requests at most; unstarted groups are not admitted.
config['allowed_stage_groups'] = ['execute/e2', 'report/e2']
config['request_ceiling'] = 218
config['prior_real_requests'] = 186
config['slice_max_requests'] = 11
subprocess.run(['git', '-C', config['author'], 'worktree', 'add', '--detach', config['tree'], config['revision']], check=True)
records = {}
def copy(source, destination, translate=True):
    data = source.read_bytes()
    result = data.decode().replace(str(OLD), str(QC)).encode() if translate else data
    destination.parent.mkdir(parents=True, exist_ok=True)
    with destination.open('xb') as stream:
        stream.write(result)
    records[str(destination.relative_to(QC))] = {'source': str(source), 'source_sha256': hashlib.sha256(data).hexdigest(), 'prepared_sha256': hashlib.sha256(result).hexdigest()}

names = [
    'credential-bootstrap.mjs', 'gateway.mjs', 'run_gateway.py',
    'review_v3.mjs', 'probe_execution_menu.mjs', 'run_stage_v3.py', 'run_probe_v3.py', 'stage_admission_v3.py',
    'check_execution_v3.py', 'file_tool.py', 'tools-v3.sb',
    'stage_boundary_preflight_v3.mjs', 'execution_boundary_preflight_v3.py', 'artifact_checkpoint_preflight.mjs',
    'probe_menu_preflight.mjs', 'probe_transport_preflight.mjs', 'provider-template.json',
    'resource_pytest.py', 'verify_transaction_inventory.py',
    'inputs/ui.vitest-v3.config.mjs', 'inputs/qc_userspace_isolation.py', 'inputs/claims.md',
    'pi-config/settings.json', 'prompt-execute-e2-v3.md', 'prompt-report-e2-v3.md',
]
for name in names:
    copy(OLD / name, QC / name)
for source in (OLD / 'inputs').glob('source-*-*'):
    copy(source, QC / 'inputs' / source.name)
for source in (OLD / 'work/controller-v3/probes').iterdir():
    if source.is_file():
        copy(source, QC / 'work/controller-v3/probes' / source.name)
for name in ('review_artifact_writer.py', 'review_io_sandbox.py'):
    copy(TOOLS / name, QC / name, False)
manifest = json.loads((OLD / 'inputs/frozen-explore-e2-v3/manifest.json').read_text())
for name, expected in manifest['artifacts_sha256'].items():
    source = OLD / 'inputs/frozen-explore-e2-v3' / name
    assert hashlib.sha256(source.read_bytes()).hexdigest() == expected
    for destination in (QC / 'inputs/frozen-explore-e2-v3' / name, QC / 'work/e2' / name):
        copy(source, destination, False)
copy(OLD / 'inputs/frozen-explore-e2-v3/manifest.json', QC / 'inputs/frozen-explore-e2-v3/manifest.json', False)
for folder in ('work/tmp', 'work/users', 'work/timer/probes', 'work/consent/probes', 'scratch', 'evidence'):
    (QC / folder).mkdir(parents=True, exist_ok=True)
for name in ('auth.json', 'models.json'):
    (QC / 'pi-config' / name).write_text('{}\n')
(QC / 'config.json').write_text(json.dumps(config, indent=2) + '\n')
subprocess.run(['cp', '-cR', str(ROOT / 'frontend-tree/intelligence/webapp/node_modules'), str(Path(config['tree']) / 'intelligence/webapp')], check=True)
for folder in ('inputs', 'work'):
    (QC / folder / 'node_modules').symlink_to('../candidate/finance-workspace-private/intelligence/webapp/node_modules')
(QC / 'preparation.json').write_text(json.dumps({'scope': 'e2 slice only; timer/consent not admitted', 'prior_real_requests': 186, 'ceiling': 218, 'max_new_requests': 11, 'automatic_retries': 0, 'dynamic_verdicts_inherited': False, 'sources': records}, indent=2) + '\n')
print(QC)
